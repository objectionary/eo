34  /**
35   * Test case for {@link MjMerge}.
36   *
37   * <p>What the merge does to a program is described by the packs in
38   * {@code merge-packs}: each one carries the files of a program, the packages to
39   * merge, and the XPaths the merged XMIR must satisfy, plus the text the
40   * generated Java must hold, both of the object and of its tests, so the
41   * behaviour is stated as the program a human would write rather than as XMIR
42   * by hand. What is left here are the mechanics
43   * no EO source can express: where the merged XMIR is pointed to from, what
44   * stops being compiled apart, and what the mojo refuses to do at all.</p>
45   *
46   * @since 0.68.0
47   */
48  @ExtendWith(MktmpResolver.class)
49  final class MjMergeTest {
50  
51      /**
52       * Temp directory, injected into every test instance, since a parameterized
53       * test cannot also take one as an argument.
54       */
55      @Mktmp
56      private Path dir;

58      @ParameterizedTest
59      @ClasspathSource(value = "org/eolang/maven/merge-packs/", glob = "**.yaml")
60      void mergesTheProgramOfAPack(final String yaml) throws IOException {
61          final Xtory pack = new XtSticky(new XtYaml(yaml));
62          MatcherAssert.assertThat(
63              "every XPath and every text of the pack must match what the merge wrote, but some didnt",
64              MjMergeTest.unmatched(pack, MjMergeTest.spliced(pack, this.dir)),
65              Matchers.empty()
66          );
67      }
68  
69      @Test
70      void pointsTheObjectAtTheMergedXmir(@Mktmp final Path temp) throws Exception {
71          final FakeMaven maven = MjMergeTest.merged(temp);
72          MatcherAssert.assertThat(
73              "the object must be transpiled from the merged XMIR and not from the parsed one",
74              maven.foreignTojos().find("foo").xmir().toString(),
75              Matchers.endsWith(
76                  Paths.get(String.format("%s/foo.xmir", maven.dirName("merge"))).toString()
77              )
78          );
79      }
80  
81      @Test
82      void takesTheMergedMemberAwayFromTranspiling(@Mktmp final Path temp) throws Exception {
83          MatcherAssert.assertThat(
84              "a member that now lives inside its package object must not be transpiled apart",
85              MjMergeTest.merged(temp)
86                  .execute(MjTranspile.class)
87                  .result()
88                  .keySet(),
89              Matchers.not(Matchers.hasItem(Matchers.containsString("EObar.java")))
90          );
91      }

93      @Test
94      void leavesAMergedMemberAloneWhenMergedAgain(@Mktmp final Path temp) throws Exception {
95          MatcherAssert.assertThat(
96              "a member already inside its object must not arrive there a second time, since MjTranspile merges too",
97              new XMLDocument(
98                  MjMergeTest.merged(temp)
99                      .execute(MjMerge.class)
100                     .foreignTojos()
101                     .find("foo")
102                     .xmir()
103             ).nodes("/object/o[@name='foo']/o[@name='bar']").size(),
104             Matchers.equalTo(1)
105         );
106     }

108     @ParameterizedTest
109     @ValueSource(strings = {"foo", "foo.gap"})
110     void leavesAPackageWithoutAnObjectAlone(final String pkg) throws Exception {
111         final FakeMaven maven = new FakeMaven(this.dir).withProgram(
112             MjMergeTest.program(String.format("+package %s", pkg), "", "[] > bar", "  42 > @"),
113             String.format("%s.bar", pkg),
114             String.format("%s/bar.eo", pkg.replace('.', '/'))
115         );
116         maven.execute(new PpMerge());
117         try (Stream<Path> kids = Files.list(maven.targetPath())) {
118             MatcherAssert.assertThat(
119                 "a package no object is named after has nothing to merge into, so nothing may be written",
120                 kids.anyMatch(kid -> kid.getFileName().toString().endsWith("-merge")),
121                 Matchers.is(false)
122             );
123         }
124     }

126     @Test
127     void refusesToMergeAMemberOverAnExistingAttribute(@Mktmp final Path temp) throws Exception {
128         MatcherAssert.assertThat(
129             "a name held by both the object and its member must stop the build and name both",
130             MjMergeTest.root(
131                 Assertions.assertThrows(
132                     IllegalStateException.class,
133                     () -> new FakeMaven(temp).withProgram(
134                         MjMergeTest.program("[n] > foo", "  42 > bar", "  n > @"),
135                         "foo",
136                         "foo.eo"
137                     ).withProgram(
138                         MjMergeTest.program("+package foo", "", "[] > bar", "  42 > @"),
139                         "foo.bar",
140                         "foo/bar.eo"
141                     ).execute(new PpMerge())
142                 )
143             ),
144             Matchers.stringContainsInOrder("bar", "foo")
145         );
146     }

148     @ParameterizedTest
149     @ValueSource(strings = {"gap", "gap.deeper"})
150     void refusesAnUnreachableMember(final String gap) {
151         final String pkg = String.format("foo.%s", gap);
152         final String member = String.format("%s.bar", pkg);
153         MatcherAssert.assertThat(
154             "a missing intermediate object must stop the merge and name the unreachable member",
155             MjMergeTest.root(
156                 Assertions.assertThrows(
157                     IllegalStateException.class,
158                     () -> new FakeMaven(this.dir).withProgram(
159                         MjMergeTest.program("[n] > foo", "  n > @"), "foo", "foo.eo"
160                     ).withProgram(
161                         MjMergeTest.program(
162                             String.format("+package %s", pkg), "", "[] > bar", "  42 > @"
163                         ),
164                         member,
165                         String.format("%s.eo", member.replace('.', '/'))
166                     ).execute(new PpMerge())
167                 )
168             ),
169             Matchers.stringContainsInOrder(member, pkg, "foo")
170         );
171     }

173     private static FakeMaven spliced(final Xtory pack, final Path temp) throws IOException {
174         final FakeMaven maven = new FakeMaven(temp);
175         for (final Map.Entry<?, ?> source : MjMergeTest.sources(pack).entrySet()) {
176             final String name = source.getKey().toString();
177             maven.withProgram(source.getValue().toString(), MjMergeTest.identifier(name), name);
178         }
179         maven.execute(new PpMerge());
180         if (!MjMergeTest.asked(pack, "java").isEmpty()
181             || !MjMergeTest.asked(pack, "tests").isEmpty()) {
182             maven.execute(MjTranspile.class);
183         }
184         return maven;
185     }

187     private static Collection<String> unmatched(final Xtory pack, final FakeMaven maven)
188         throws IOException {
189         final Collection<String> failed = new ArrayList<>(0);
190         for (final Object key : pack.map().keySet()) {
191             if (!Arrays.asList("eo", "xmir", "java", "tests", "absent").contains(key)) {
192                 failed.add(String.format("unknown key: %s", key));
193             }
194         }
195         failed.addAll(
196             MjMergeTest.unmerged(pack, maven.targetPath().resolve(maven.dirName("merge")))
197         );
198         failed.addAll(MjMergeTest.untranspiled(pack, maven));
199         return failed;
200     }
201 
202     private static Collection<String> unmerged(final Xtory pack, final Path base)
203         throws IOException {
204         final Collection<String> failed = new ArrayList<>(0);
205         for (final Map.Entry<?, ?> entry : MjMergeTest.asked(pack, "xmir").entrySet()) {
206             final String name = entry.getKey().toString();
207             final Path file = base.resolve(name.replace(".eo", ".xmir"));
208             if (Files.exists(file)) {
209                 failed.addAll(
210                     MjMergeTest.absent(new XMLDocument(file), name, (List<?>) entry.getValue())
211                 );
212             } else {
213                 failed.add(String.format("no merged XMIR for %s", name));
214             }
215         }
216         for (final Object name : MjMergeTest.listed(pack, "absent")) {
217             if (Files.exists(base.resolve(name.toString().replace(".eo", ".xmir")))) {
218                 failed.add(String.format("%s was merged while it should not be", name));
219             }
220         }
221         return failed;
222     }
223 
224     private static Collection<String> untranspiled(final Xtory pack, final FakeMaven maven)
225         throws IOException {
226         final Collection<String> failed = new ArrayList<>(0);
227         for (final String kind : Arrays.asList("java", "tests")) {
228             for (final Map.Entry<?, ?> entry : MjMergeTest.asked(pack, kind).entrySet()) {
229                 final String name = entry.getKey().toString();
230                 final Path file = MjMergeTest.generated(maven, name, kind);
231                 if (Files.exists(file)) {
232                     failed.addAll(
233                         MjMergeTest.missing(
234                             Files.readString(file), name, (List<?>) entry.getValue()
235                         )
236                     );
237                 } else {
238                     failed.add(String.format("nothing generated for %s under %s", name, kind));
239                 }
240             }
241         }
242         return failed;
243     }
244 
245     private static Path generated(final FakeMaven maven, final String name, final String kind)
246         throws IOException {
247         final Path base;
248         final String mark;
249         if ("tests".equals(kind)) {
250             base = maven.generatedPath().getParent().resolve("generated-test-sources");
251             mark = "Test";
252         } else {
253             base = maven.generatedPath();
254             mark = "";
255         }
256         return base.resolve("org").resolve("eolang").resolve(
257             String.format(
258                 "%sEO%s.java", mark, name.replace(".eo", "").replace("/", "$EO")
259             )
260         );
261     }

263     private static Map<?, ?> asked(final Xtory pack, final String key) {
264         return (Map<?, ?>) pack.map().getOrDefault(key, Collections.emptyMap());
265     }

267     private static List<?> listed(final Xtory pack, final String key) {
268         return (List<?>) pack.map().getOrDefault(key, Collections.emptyList());
269     }

271     private static Collection<String> missing(
272         final String java, final String about, final List<?> texts
273     ) {
274         final Collection<String> failed = new ArrayList<>(0);
275         for (final Object text : texts) {
276             if (!java.contains(text.toString())) {
277                 failed.add(String.format("%s: %s", about, text));
278             }
279         }
280         return failed;
281     }

283     private static Map<?, ?> sources(final Xtory pack) {
284         return (Map<?, ?>) pack.map().get("eo");
285     }

287     private static String identifier(final String name) {
288         return name.replace(".eo", "").replace('/', '.');
289     }
290 
291     private static Collection<String> absent(
292         final XML document, final String about, final List<?> xpaths
293     ) {
294         final Collection<String> failed = new ArrayList<>(0);
295         for (final Object xpath : xpaths) {
296             if (document.nodes(xpath.toString()).isEmpty()) {
297                 failed.add(String.format("%s: %s", about, xpath));
298             }
299         }
300         return failed;
301     }

303     private static FakeMaven merged(final Path temp) throws IOException {
304         return new FakeMaven(temp).withProgram(
305             MjMergeTest.program("[n] > foo", "  n > @"),
306             "foo",
307             "foo.eo"
308         ).withProgram(
309             MjMergeTest.program("+package foo", "", "[] > bar", "  42 > @"),
310             "foo.bar",
311             "foo/bar.eo"
312         ).execute(new PpMerge());
313     }

315     private static String root(final Throwable thrown) {
316         Throwable cause = thrown;
317         while (cause.getCause() != null) {
318             cause = cause.getCause();
319         }
320         return cause.getMessage();
321     }

323     private static String program(final String... lines) {
324         return String.join(System.lineSeparator(), lines);
