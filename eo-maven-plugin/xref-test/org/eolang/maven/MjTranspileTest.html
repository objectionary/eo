913     private static boolean intact(final Path file) {
914         final String java = new UncheckedText(new TextOf(file)).asString();
915         return MjTranspileTest.balanced(java, '{', '}')
916             && MjTranspileTest.balanced(java, '(', ')')
917             && (!MjTranspileTest.filename(file).startsWith("EOmain")
918                 || java.contains("class EOmain"));
919     }

921     private static boolean balanced(final String text, final char open, final char close) {
922         int depth = 0;
923         boolean valid = true;
924         for (int idx = 0; idx < text.length(); ++idx) {
925             final char chr = text.charAt(idx);
926             if (chr == open) {
927                 depth = depth + 1;
928             } else if (chr == close) {
929                 depth = depth - 1;
930             }
931             if (depth < 0) {
932                 valid = false;
933                 break;
934             }
935         }
936         return valid && depth == 0;
937     }
