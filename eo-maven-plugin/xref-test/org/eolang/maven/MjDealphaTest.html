31      void namesArgumentAfterVoidItLandsIn(@Mktmp final Path temp) throws IOException {
32          final FakeMaven maven = new FakeMaven(temp).withProgram(
33              String.join(

46                  )
47              ),
48              XhtmlMatchers.hasXPath("/object/o/o[@name='φ']/o[@as='v🌵bar']")

435     private static XMLDocument picked(final Path temp, final String right)
436         throws IOException {
437         final FakeMaven maven = new FakeMaven(temp).withProgram(
438             String.join(
439                 System.lineSeparator(),
440                 "[] > app",
441                 "  pick > yes",
442                 "    bool",
443                 "      [left right]",
444                 "        left > @",
445                 "  pick > @",
446                 "    bool",
447                 "      [left right]",
448                 "        right > @",
449                 "  [flag] > pick",
450                 "    box.code.eq 5 > @",
451                 "    flag.if > box",
452                 "      os.status 7",
453                 String.format("      os.state %s", right)
454             )
455         ).withProgram(
456             String.join(
457                 System.lineSeparator(),
458                 "[] > os",
459                 "  [cde] > status",
460                 "    cde > code",
461                 "  [cd] > state",
462                 "    cd > code"
463             ),
464             "os",
465             "os.eo"
466         ).withProgram(
467             String.join(
468                 System.lineSeparator(),
469                 "[fork] > bool",
470                 "  fork > if"
471             ),
472             "bool",
473             "bool.eo"
474         ).withProgram(
475             String.join(
476                 System.lineSeparator(),
477                 "[@] > number",
478                 "  [x] > eq",
479                 "    x > @"
480             ),
481             "number",
482             "number.eo"
483         ).withProgram("[@] > bytes", "bytes", "bytes.eo");
484         return new XMLDocument(
485             maven.execute(new PpDealpha()).result().get(
486                 String.format("target/%s/foo/x/main.xmir", maven.dirName("dealpha"))
487             )
488         );
