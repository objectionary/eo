5   package org.eolang.maven;
6   
7   import java.net.InetSocketAddress;
8   import java.net.Proxy;
9   import org.apache.maven.settings.Settings;
10  import org.hamcrest.MatcherAssert;
11  import org.hamcrest.Matchers;
12  import org.junit.jupiter.api.Assertions;
13  import org.junit.jupiter.api.Test;
14  
15  /**
16   * Test case for {@link Proxies}.
17   *
18   * @since 0.62.0
19   */
20  final class ProxiesTest {
21  
22      @Test
23      void translatesActiveProxy() {
24          MatcherAssert.assertThat(
25              "Active proxy of settings must keep its address",
26              ProxiesTest.only(ProxiesTest.settings("prox.eolang.org", 8431, true)).address(),
27              Matchers.equalTo(
28                  new Proxy(
29                      Proxy.Type.HTTP,
30                      InetSocketAddress.createUnresolved("prox.eolang.org", 8431)
31                  )
32              )
33          );
34      }
35  
36      @Test
37      void skipsInactiveProxy() {
38          MatcherAssert.assertThat(
39              "Inactive proxy of settings must be dropped",
40              new Proxies(ProxiesTest.settings("dead.eolang.org", 3129, false)).value(),
41              Matchers.emptyArray()
42          );
43      }
44  
45      @Test
46      void staysEmptyWithoutProxies() {
47          MatcherAssert.assertThat(
48              "Settings without proxies must give no proxies at all",
49              new Proxies(new Settings()).value(),
50              Matchers.emptyArray()
51          );
52      }
53  
54      @Test
55      void refusesAProxyThatSpeaksSomethingElse() {
56          final Settings settings = ProxiesTest.settings("socks.eolang.org", 1080, true);
57          settings.getProxies().get(0).setProtocol("socks5");
58          MatcherAssert.assertThat(
59              "the protocol must be named, since only an HTTP proxy can be used here",
60              Assertions.assertThrows(
61                  IllegalStateException.class,
62                  () -> new Proxies(settings).value(),
63                  "a proxy nothing can talk to must not be taken for one that works"
64              ).getMessage(),
65              Matchers.containsString("socks5")
66          );
67      }
68  
69      private static MvnProxy only(final Settings settings) {
70          return new Proxies(settings).value()[0];
71      }
72  
73      private static Settings settings(final String host, final int port, final boolean active) {
74          final org.apache.maven.settings.Proxy proxy = new org.apache.maven.settings.Proxy();
75          proxy.setHost(host);
76          proxy.setPort(port);
77          proxy.setActive(active);
78          final Settings settings = new Settings();
79          settings.addProxy(proxy);
80          return settings;
81      }
82  }
