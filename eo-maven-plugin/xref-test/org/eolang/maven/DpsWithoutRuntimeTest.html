5   package org.eolang.maven;
6   
7   import org.cactoos.list.ListOf;
8   import org.hamcrest.MatcherAssert;
9   import org.hamcrest.Matchers;
10  import org.junit.jupiter.api.Test;
11  
12  /**
13   * Test case for {@link DpsWithoutRuntime}.
14   *
15   * @since 0.29
16   */
17  final class DpsWithoutRuntimeTest {
18  
19      @Test
20      void removesTheRuntimeOfEo() {
21          MatcherAssert.assertThat(
22              "the EO runtime must be taken out of the list, but it wasnt",
23              new DpsWithoutRuntime(
24                  () -> new ListOf<>(
25                      new Dep()
26                          .withGroupId("org.eolang")
27                          .withArtifactId("eo-runtime")
28                          .withVersion("0.30.0")
29                  ).iterator()
30              ),
31              Matchers.emptyIterable()
32          );
33      }
34  
35      @Test
36      void keepsTheClassifiedArtifactOfTheRuntime() {
37          MatcherAssert.assertThat(
38              "a classified artifact of the runtime must stay, but it didnt",
39              new DpsWithoutRuntime(
40                  () -> new ListOf<>(
41                      new Dep()
42                          .withGroupId("org.eolang")
43                          .withArtifactId("eo-runtime")
44                          .withClassifier("tests")
45                          .withVersion("0.30.0")
46                  ).iterator()
47              ),
48              Matchers.iterableWithSize(1)
49          );
50      }
51  
52      @Test
53      void keepsTheRuntimeOfSomebodyElse() {
54          MatcherAssert.assertThat(
55              "a dependency of another group named eo-runtime must stay, but it didnt",
56              new DpsWithoutRuntime(
57                  () -> new ListOf<>(
58                      new Dep()
59                          .withGroupId("com.example")
60                          .withArtifactId("eo-runtime")
61                          .withVersion("1.0.0")
62                  ).iterator()
63              ),
64              Matchers.iterableWithSize(1)
65          );
66      }
67  }
