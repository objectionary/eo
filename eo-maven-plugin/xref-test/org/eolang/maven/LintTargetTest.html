5   package org.eolang.maven;
6   
7   import com.jcabi.xml.XMLDocument;
8   import com.yegor256.Mktmp;
9   import com.yegor256.MktmpResolver;
10  import java.io.File;
11  import java.nio.file.Path;
12  import org.hamcrest.MatcherAssert;
13  import org.hamcrest.Matchers;
14  import org.junit.jupiter.api.Test;
15  import org.junit.jupiter.api.extension.ExtendWith;
16  
17  /**
18   * Test cases for {@link LintTarget}.
19   *
20   * @since 0.64
21   */
22  @ExtendWith(MktmpResolver.class)
23  final class LintTargetTest {
24  
25      @Test
26      void namesTheTargetAfterTheObjectTheDocumentDeclares(@Mktmp final Path temp) {
27          MatcherAssert.assertThat(
28              "the target must be named after the object of the document, but it wasnt",
29              new LintTarget(
30                  new XMLDocument("<object><o name='bar'/></object>"),
31                  temp.resolve("bar.xmir"),
32                  "foo.x.bar"
33              ).under(temp).toString(),
34              Matchers.endsWith(String.format("bar.%s", MjAssemble.XMIR))
35          );
36      }
37  
38      @Test
39      void namesTheTargetAfterTheCatalogWhenTheDocumentDeclaresNoObject(@Mktmp final Path temp) {
40          MatcherAssert.assertThat(
41              "a document that declares no object must be named by the catalog, not abort the goal",
42              new LintTarget(
43                  new XMLDocument("<object><metas/></object>"),
44                  temp.resolve("main.xmir"),
45                  "foo.x.main"
46              ).under(temp).toString(),
47              Matchers.endsWith(
48                  String.format("foo%sx%smain.%s", File.separator, File.separator, MjAssemble.XMIR)
49              )
50          );
51      }
52  }
