37      @Test
38      void pullsSuccessfully(@Mktmp final Path temp) throws IOException {
39          final FakeMaven maven = new FakeMaven(temp);
40          maven.foreignTojos()
41              .add(this.stdout())
42              .withVersion("*.*.*");
43          maven.with("skip", false).execute(MjPull.class);
44          MatcherAssert.assertThat(
45              "PullMojo should have pulled stdout object, but didn't",
46              MjPullTest.exists(temp, this.stdout()),
47              Matchers.is(true)
48          );
49      }

51      @Test
52      void pullsFromProbes(@Mktmp final Path temp) throws IOException {
53          new FakeMaven(temp).withProgram(
54              String.format("+package foo.x%n"),
55              "[] > main",
56              "  Q.stdout > @",
57              "    \"I am 18 years old\""
58              )
59              .with("objectionary", new ScalarOf<>(() -> new OyRemote(new ChRemote("master"))))
60              .execute(new PpPull());
61          MatcherAssert.assertThat(
62              "PullMojo should have pulled from probes, but it didn't",
63              MjPullTest.exists(temp, this.stdout()),
64              Matchers.is(true)
65          );
66      }

68      @Test
69      void pullsUsingOfflineHashFile(@Mktmp final Path temp) throws IOException {
70          new Saved(
71              new ResourceOf("org/eolang/maven/commits/tags.txt"),
72              temp.resolve("tags.txt")
73          ).value();
74          final FakeMaven maven = new FakeMaven(temp);
75          maven.foreignTojos()
76              .add(this.stdout())
77              .withVersion("*.*.*");
78          maven.with("skip", false).with(
79              "hash",
80              new ChCached(new ChText(temp.resolve("tags.txt"), "master"))
81              )
82              .execute(MjPull.class);
83          MatcherAssert.assertThat(
84              "PullMojo should have pulled using offline hash file, but it didn't",
85              new MnCsv(maven.foreignPath()).read().iterator().next().get("hash"),
86              Matchers.equalTo("mmmmmmm")
87          );
88      }

124     @Test
125     void doesNotPullInOfflineMode(@Mktmp final Path tmp) throws IOException {
126         final FakeMaven maven = new FakeMaven(tmp)
127             .withHelloWorld()
128             .with("offline", true);
129         final Map<String, Path> result = maven.execute(new PpPull()).result();
130         final String stdout = "org/eolang/stdout.eo";
131         final String string = "org/eolang/string.eo";
132         final String dir = maven.dirName("pull");
133         MatcherAssert.assertThat(
134             String.format(
135                 "%s folder should not contain %s and %s file, but it did",
136                 dir,
137                 stdout,
138                 string
139             ),
140             result,
141             Matchers.allOf(
142                 Matchers.not(Matchers.hasKey(String.format("%s/%s", dir, stdout))),
143                 Matchers.not(Matchers.hasKey(String.format("%s/%s", dir, string)))
144             )
145         );
146     }

148     @Test
149     void skipsAlreadyPulled(@Mktmp final Path temp) throws IOException {
150         final FakeMaven maven = new FakeMaven(temp)
151             .withHelloWorld()
152             .execute(new PpPull());
153         final Path path = maven.result().get(
154             String.format("target/%s/bytes.%s", maven.dirName("pull"), MjAssemble.EO)
155         );
156         final long mtime = path.toFile().lastModified();
157         maven.execute(MjPull.class);
158         MatcherAssert.assertThat(
159             "PullMojo must skip pulling if source was already downloaded",
160             path.toFile().lastModified(),
161             Matchers.is(mtime)
162         );
163     }

165     @Test
166     void savesPulledResultsToCache(@Mktmp final Path temp) throws IOException {
167         final Path cache = temp.resolve("cache");
168         final CommitHash hash = new ChCached(
169             new ChNarrow(
170                 new ChRemote("master")
171             )
172         );
173         new FakeMaven(temp)
174             .withHelloWorld()
175             .with("cache", cache.toFile())
176             .with("hash", hash)
177             .execute(new PpPull());
178         MatcherAssert.assertThat(
179             "Pulled results must be saved to cache",
180             cache.resolve(Pulling.CACHE)
181                 .resolve(FakeMaven.pluginVersion())
182                 .resolve(hash.value())
183                 .resolve("bytes.eo")
184                 .toFile(),
185             FileMatchers.anExistingFile()
186         );
187     }

189     @Test
190     void getsAlreadyPulledResultsFromCache(@Mktmp final Path temp) throws Exception {
191         final Path cache = temp.resolve("cache");
192         final String hash = "abcdef1";
193         final String cached = String.format("[] > just-something%n");
194         new Saved(
195             cached,
196             cache
197                 .resolve(Pulling.CACHE)
198                 .resolve(FakeMaven.pluginVersion())
199                 .resolve(hash)
200                 .resolve("io/stdout.eo")
201         ).value();
202         Files.setLastModifiedTime(
203             cache.resolve(
204                 Paths
205                     .get(Pulling.CACHE)
206                     .resolve(FakeMaven.pluginVersion())
207                     .resolve(hash)
208                     .resolve("io/stdout.eo")
209             ),
210             FileTime.fromMillis(System.currentTimeMillis() + 50_000)
211         );
212         final FakeMaven maven = new FakeMaven(temp).withProgram(
213             String.join(
214                 System.lineSeparator(),
215                 "+package foo.x",
216                 "",
217                 "[] > main",
218                 "  Q.io.stdout > @"
219             )
220         ).with("hash", new CommitHash.ChConstant(hash))
221             .with("cache", cache.toFile());
222         maven.execute(new PpPull());
223         MatcherAssert.assertThat(
224             "PullMojo should take source from cache, but it does not",
225             new TextOf(
226                 Files.readAllBytes(
227                     temp.resolve(
228                         String.format(
229                             "target/%s/io/stdout.%s",
230                             maven.dirName("pull"),
231                             MjAssemble.EO
232                         )
233                     )
234                 )
235             ).asString(),
236             Matchers.is(cached)
237         );
238     }

244     private static boolean exists(final Path temp, final String source) throws IOException {
245         return Files.exists(temp.resolve("target").resolve(MjPullTest.path(temp, source)));
246     }
247 
248     private static Path path(final Path temp, final String source) throws IOException {
249         return new Place(source).make(
250             Paths.get(
251                 new Seqdir(temp.resolve("target"), 2)
252                     .dirs().once("pull").getFileName().toString()
253             ),
254             "eo"
255         );
