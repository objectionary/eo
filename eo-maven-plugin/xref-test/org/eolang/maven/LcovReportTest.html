5   package org.eolang.maven;
6   
7   import java.util.LinkedHashMap;
8   import java.util.Map;
9   import org.hamcrest.MatcherAssert;
10  import org.hamcrest.Matchers;
11  import org.junit.jupiter.api.Test;
12  
13  /**
14   * Test case for {@link LcovReport}.
15   *
16   * @since 0.75.0
17   */
18  final class LcovReportTest {
19  
20      @Test
21      void rendersOneFileWithAMixOfHitAndMissedLines() {
22          final Map<Integer, Integer> lines = new LinkedHashMap<>(2);
23          lines.put(5, 2);
24          lines.put(9, 0);
25          final Map<String, Map<Integer, Integer>> files = new LinkedHashMap<>(1);
26          files.put("foo.eo", lines);
27          final StringBuilder expected = new StringBuilder(32);
28          for (final String line : new String[] {
29              "SF:foo.eo", "DA:5,2", "DA:9,0", "LH:1", "LF:2", "end_of_record",
30          }) {
31              expected.append(line).append('\n');
32          }
33          MatcherAssert.assertThat(
34              "the rendered text must carry SF/DA/LH/LF for the one file, but it didnt",
35              new LcovReport(files).text(),
36              Matchers.equalTo(expected.toString())
37          );
38      }
39  
40      @Test
41      void rendersAnEmptyRecordForAFileWithoutLines() {
42          final Map<String, Map<Integer, Integer>> files = new LinkedHashMap<>(1);
43          files.put("foo.eo", new LinkedHashMap<>(0));
44          final StringBuilder expected = new StringBuilder(32);
45          for (final String line : new String[] {
46              "SF:foo.eo", "LH:0", "LF:0", "end_of_record",
47          }) {
48              expected.append(line).append('\n');
49          }
50          MatcherAssert.assertThat(
51              "a file with no lines must still be rendered as a record of its own, but it wasnt",
52              new LcovReport(files).text(),
53              Matchers.equalTo(expected.toString())
54          );
55      }
56  
57      @Test
58      void ignoresAFileWithoutLinesInThePercentage() {
59          final Map<Integer, Integer> lines = new LinkedHashMap<>(2);
60          lines.put(1, 1);
61          lines.put(2, 0);
62          final Map<String, Map<Integer, Integer>> files = new LinkedHashMap<>(2);
63          files.put("a.eo", lines);
64          files.put("b.eo", new LinkedHashMap<>(0));
65          MatcherAssert.assertThat(
66              "a file with nothing to instrument must not drag the percentage down, but it did",
67              new LcovReport(files).covered(),
68              Matchers.closeTo(50.0, 0.001)
69          );
70      }
71  
72      @Test
73      void computesTheCoveredPercentageAcrossFiles() {
74          final Map<Integer, Integer> first = new LinkedHashMap<>(2);
75          first.put(1, 1);
76          first.put(2, 0);
77          final Map<Integer, Integer> second = new LinkedHashMap<>(2);
78          second.put(1, 3);
79          second.put(2, 0);
80          final Map<String, Map<Integer, Integer>> files = new LinkedHashMap<>(2);
81          files.put("a.eo", first);
82          files.put("b.eo", second);
83          MatcherAssert.assertThat(
84              "two hit lines out of four total must be 50 percent, but it wasnt",
85              new LcovReport(files).covered(),
86              Matchers.closeTo(50.0, 0.001)
87          );
88      }
89  
90      @Test
91      void reportsFullCoverageWhenThereIsNothingToInstrument() {
92          MatcherAssert.assertThat(
93              "an empty report has nothing uncovered, so it must read as fully covered, but it didnt",
94              new LcovReport(new LinkedHashMap<>(0)).covered(),
95              Matchers.closeTo(100.0, 0.001)
96          );
97      }
98  }
