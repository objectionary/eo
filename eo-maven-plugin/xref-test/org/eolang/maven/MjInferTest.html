22   * Test cases for {@link MjInfer}.

26  @ExtendWith(MktmpResolver.class)
27  final class MjInferTest {
28  
29      @Test
30      void writesTableOfWholeProgram(@Mktmp final Path temp) throws IOException {
31          final FakeMaven maven = new FakeMaven(temp).withProgram(
32              String.join(
33                  System.lineSeparator(),
34                  "[] > app",
35                  "  [] > t",
36                  "    [] > next",
37                  ""
38              )
39          );
40          maven.execute(MjParse.class).execute(MjInfer.class);
41          MatcherAssert.assertThat(
42              "the facts must say what the innermost formation has, but they dont",
43              new XMLDocument(
44                  maven.targetPath()
45                      .resolve(maven.dirName("inference"))
46                      .resolve("facts.xml")
47              ),
48              XhtmlMatchers.hasXPath("/facts/objects/type[@id='Φ.app.t']/attr[@name='next']")
49          );
50      }
51  
52      @Test
53      void writesAnswersBesideFacts(@Mktmp final Path temp) throws IOException {
54          final FakeMaven maven = new FakeMaven(temp).withProgram(
55              String.join(
56                  System.lineSeparator(),
57                  "[] > app",
58                  "  [] > t",
59                  "    [] > next",
60                  ""
61              )
62          );
63          maven.execute(MjParse.class).execute(MjInfer.class);
64          MatcherAssert.assertThat(
65              "the answers must name what each place is, but they dont",
66              new XMLDocument(
67                  maven.targetPath()
68                      .resolve(maven.dirName("inference"))
69                      .resolve("answers.xml")
70              ),
71              XhtmlMatchers.hasXPath("/answers/o[@loc='Φ.app.t']/is[@loc='Φ.app.t']")
72          );
73      }
74  
75      @Test
76      void readsOnlyTheXmirOfARegisteredSource(@Mktmp final Path temp) throws IOException {
77          final FakeMaven maven = new FakeMaven(temp).withProgram(
78              String.join(System.lineSeparator(), "[] > app", "")
79          );
80          maven.execute(MjParse.class);
81          Files.writeString(
82              maven.targetPath().resolve(maven.dirName("parse")).resolve("bee.xmir"),
83              new EoSyntax(
84                  String.join(System.lineSeparator(), "[] > bee", "")
85              ).parsed().toString()
86          );
87          maven.execute(MjInfer.class);
88          MatcherAssert.assertThat(
89              "an XMIR that no tojo names must stay out of the facts, but it didnt",
90              new XMLDocument(
91                  maven.targetPath()
92                      .resolve(maven.dirName("inference"))
93                      .resolve("facts.xml")
94              ).nodes("/facts/objects/type[@id='Φ.bee']"),
95              Matchers.empty()
96          );
97      }
98  
99      @Test
100     void writesFactsForModuleWithoutSources(@Mktmp final Path temp) throws IOException {
101         final FakeMaven maven = new FakeMaven(temp);
102         maven.execute(MjParse.class).execute(MjInfer.class);
103         MatcherAssert.assertThat(
104             "the facts must be written for a module with no sources, but they werent",
105             new XMLDocument(
106                 maven.targetPath()
107                     .resolve(maven.dirName("inference"))
108                     .resolve("facts.xml")
109             ),
110             XhtmlMatchers.hasXPath("/facts/calls[not(*)]")
111         );
112     }
113 
114     @Test
115     void takesCanonOfRuntimeWhenUnset(@Mktmp final Path temp) throws IOException {
116         final FakeMaven maven = new FakeMaven(temp)
117             .withProgram(String.join(System.lineSeparator(), "[] > app", "  number 5 > five", ""))
118             .withProgram("[@] > bytes", "bytes")
119             .withProgram("[@] > number", "number")
120             .withProgram("[@] > string", "string");
121         maven.execute(MjParse.class).execute(MjInfer.class);
122         MatcherAssert.assertThat(
123             "a number must hold bytes and nothing else, as in the canon of eo-runtime, but it didnt",
124             new XMLDocument(
125                 maven.targetPath()
126                     .resolve(maven.dirName("inference"))
127                     .resolve("answers.xml")
128             ),
129             XhtmlMatchers.hasXPath(
130                 "/answers/o[@loc='Φ.number.φ' and count(*)=1]/is[@loc='Φ.bytes']"
131             )
132         );
133     }
134 
135     @Test
136     void takesCanonOfRuntimeWhenEmpty(@Mktmp final Path temp) throws IOException {
137         final FakeMaven maven = new FakeMaven(temp)
138             .withProgram(String.join(System.lineSeparator(), "[] > app", "  number 5 > five", ""))
139             .withProgram("[@] > bytes", "bytes")
140             .withProgram("[@] > number", "number")
141             .withProgram("[@] > string", "string")
142             .with("canon", List.of())
143             .with("roots", List.of())
144             .with("scalars", List.of());
145         maven.execute(MjParse.class).execute(MjInfer.class);
146         MatcherAssert.assertThat(
147             "empty lists, which Maven gives when nothing is set, must mean the canon of eo-runtime, but they didnt",
148             new XMLDocument(
149                 maven.targetPath()
150                     .resolve(maven.dirName("inference"))
151                     .resolve("answers.xml")
152             ),
153             XhtmlMatchers.hasXPath(
154                 "/answers/o[@loc='Φ.number.φ' and count(*)=1]/is[@loc='Φ.bytes']"
155             )
156         );
157     }

159     @Test
160     void keepsPreparedXmirOfProgram(@Mktmp final Path temp) throws IOException {
161         final FakeMaven maven = new FakeMaven(temp).withProgram(
162             String.join(
163                 System.lineSeparator(),
164                 "[x] > box",
165                 "  x.lid.hinge > @",
166                 ""
167             )
168         );
169         maven.execute(MjParse.class).execute(MjInfer.class);
170         MatcherAssert.assertThat(
171             "the prepared XMIR must have one object per name of a chain, but it doesnt",
172             new XMLDocument(
173                 maven.targetPath()
174                     .resolve(maven.dirName("pre-inference"))
175                     .resolve("foo")
176                     .resolve("x")
177                     .resolve("main.xmir")
178             ),
179             XhtmlMatchers.hasXPath(
180                 "//o[@base='.hinge']/o[@base='.lid']/o[@base='.v🌵x']/o[@base='ξ']"
181             )
182         );
183     }
