5   package org.eolang.maven;
6   
7   import com.jcabi.matchers.XhtmlMatchers;
8   import com.jcabi.xml.XMLDocument;
9   import com.yegor256.Mktmp;
10  import com.yegor256.MktmpResolver;
11  import java.io.IOException;
12  import java.nio.file.Files;
13  import java.nio.file.Path;
14  import org.eolang.parser.EoSyntax;
15  import org.hamcrest.MatcherAssert;
16  import org.hamcrest.Matchers;
17  import org.junit.jupiter.api.Test;
18  import org.junit.jupiter.api.extension.ExtendWith;
19  
20  /**
21   * Test cases for {@link MjInfer}.
22   *
23   * @since 0.67.0
24   */
25  @ExtendWith(MktmpResolver.class)
26  final class MjInferTest {
27  
28      @Test
29      void writesTableOfWholeProgram(@Mktmp final Path temp) throws IOException {
30          final FakeMaven maven = new FakeMaven(temp).withProgram(
31              String.join(
32                  System.lineSeparator(),
33                  "[] > app",
34                  "  [] > t",
35                  "    [] > next",
36                  ""
37              )
38          );
39          maven.execute(MjParse.class).execute(MjInfer.class);
40          MatcherAssert.assertThat(
41              "the facts must say what the innermost formation has, but they dont",
42              new XMLDocument(
43                  maven.targetPath()
44                      .resolve(maven.dirName("inference"))
45                      .resolve("facts.xml")
46              ),
47              XhtmlMatchers.hasXPath("/facts/objects/type[@id='Φ.app.t']/attr[@name='next']")
48          );
49      }
50  
51      @Test
52      void writesAnswersBesideFacts(@Mktmp final Path temp) throws IOException {
53          final FakeMaven maven = new FakeMaven(temp).withProgram(
54              String.join(
55                  System.lineSeparator(),
56                  "[] > app",
57                  "  [] > t",
58                  "    [] > next",
59                  ""
60              )
61          );
62          maven.execute(MjParse.class).execute(MjInfer.class);
63          MatcherAssert.assertThat(
64              "the answers must name what each place is, but they dont",
65              new XMLDocument(
66                  maven.targetPath()
67                      .resolve(maven.dirName("inference"))
68                      .resolve("answers.xml")
69              ),
70              XhtmlMatchers.hasXPath("/answers/o[@loc='Φ.app.t']/is[@loc='Φ.app.t']")
71          );
72      }
73  
74      @Test
75      void readsOnlyTheXmirOfARegisteredSource(@Mktmp final Path temp) throws IOException {
76          final FakeMaven maven = new FakeMaven(temp).withProgram(
77              String.join(System.lineSeparator(), "[] > app", "")
78          );
79          maven.execute(MjParse.class);
80          Files.writeString(
81              maven.targetPath().resolve(maven.dirName("parse")).resolve("bee.xmir"),
82              new EoSyntax(
83                  String.join(System.lineSeparator(), "[] > bee", "")
84              ).parsed().toString()
85          );
86          maven.execute(MjInfer.class);
87          MatcherAssert.assertThat(
88              "an XMIR that no tojo names must stay out of the facts, but it didnt",
89              new XMLDocument(
90                  maven.targetPath()
91                      .resolve(maven.dirName("inference"))
92                      .resolve("facts.xml")
93              ).nodes("/facts/objects/type[@id='Φ.bee']"),
94              Matchers.empty()
95          );
96      }
97  
98      @Test
99      void writesFactsForModuleWithoutSources(@Mktmp final Path temp) throws IOException {
100         final FakeMaven maven = new FakeMaven(temp);
101         maven.execute(MjParse.class).execute(MjInfer.class);
102         MatcherAssert.assertThat(
103             "the facts must be written for a module with no sources, but they werent",
104             new XMLDocument(
105                 maven.targetPath()
106                     .resolve(maven.dirName("inference"))
107                     .resolve("facts.xml")
108             ),
109             XhtmlMatchers.hasXPath("/facts/calls[not(*)]")
110         );
111     }
112 
113     @Test
114     void keepsPreparedXmirOfProgram(@Mktmp final Path temp) throws IOException {
115         final FakeMaven maven = new FakeMaven(temp).withProgram(
116             String.join(
117                 System.lineSeparator(),
118                 "[x] > box",
119                 "  x.lid.hinge > @",
120                 ""
121             )
122         );
123         maven.execute(MjParse.class).execute(MjInfer.class);
124         MatcherAssert.assertThat(
125             "the prepared XMIR must have one object per name of a chain, but it doesnt",
126             new XMLDocument(
127                 maven.targetPath()
128                     .resolve(maven.dirName("pre-inference"))
129                     .resolve("foo")
130                     .resolve("x")
131                     .resolve("main.xmir")
132             ),
133             XhtmlMatchers.hasXPath(
134                 "//o[@base='.hinge']/o[@base='.lid']/o[@base='.v🌵x']/o[@base='ξ']"
135             )
136         );
137     }
