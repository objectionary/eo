16  /**
17   * Test cases for {@link Calls}.

21  final class CallsTest {
22  
23      @Test
24      void findsArgumentAtPositionOfVoid() {
25          final String arg = "Φ.app.φ.α1";
26          MatcherAssert.assertThat(
27              "a positional argument must go into the void at its position, but it doesnt",
28              new Calls(
29                  Map.of("Φ.status", List.of("cde", "msg")),
30                  Map.of("Φ.app.φ", Map.of(0, "Φ.app.φ.α0", 1, arg)),
31                  Map.of(),
32                  Map.of()
33              ).argument("Φ.app.φ", "Φ.status", "msg"),
34              Matchers.equalTo(Optional.of(arg))
35          );
36      }
37  
38      @Test
39      void findsArgumentBoundToVoidByName() {
40          final String arg = String.format("Φ.app.φ.x%d", new SecureRandom().nextInt(10_000));
41          MatcherAssert.assertThat(
42              "an argument bound by name must go into the void it names, but it doesnt",
43              new Calls(
44                  Map.of("Φ.status", List.of("cde", "msg")),
45                  Map.of(),
46                  Map.of("Φ.app.φ", Map.of("msg", "Φ.app.φ.y", "cde", arg)),
47                  Map.of()
48              ).argument("Φ.app.φ", "Φ.status", "cde"),
49              Matchers.equalTo(Optional.of(arg))
50          );
51      }
52  
53      @Test
54      void findsArgumentWhenCallLeavesLastVoidEmpty() {
55          final String arg = String.format("Φ.app.φ.α%d", new SecureRandom().nextInt(10_000));
56          MatcherAssert.assertThat(
57              "a call that leaves the last void empty must still put its argument in, but it doesnt",
58              new Calls(
59                  Map.of("Φ.padded", List.of("width", "fill", "cant-pad")),
60                  Map.of("Φ.app.φ", Map.of(0, arg, 1, "Φ.app.φ.α1")),
61                  Map.of(),
62                  Map.of()
63              ).argument("Φ.app.φ", "Φ.padded", "width"),
64              Matchers.equalTo(Optional.of(arg))
65          );
66      }
67  
68      @Test
69      void findsNoArgumentForVoidCallLeavesEmpty() {
70          MatcherAssert.assertThat(
71              "a call must put nothing into a void it leaves empty, but it does",
72              new Calls(
73                  Map.of("Φ.status", List.of("cde", "msg")),
74                  Map.of("Φ.app.φ", Map.of(0, "Φ.app.φ.α0")),
75                  Map.of(),

77              ).argument("Φ.app.φ", "Φ.status", "msg"),

79          );
80      }

82      @Test
83      void findsNoArgumentWhenPositionGoesPastLastVoid() {
84          MatcherAssert.assertThat(
85              "a call with an argument past the last void must put nothing in, but it does",
86              new Calls(
87                  Map.of("Φ.status", List.of("cde", "msg")),
88                  Map.of("Φ.app.φ", Map.of(0, "Φ.app.φ.α0", 2, "Φ.app.φ.α2")),

90                  Map.of()
91              ).argument("Φ.app.φ", "Φ.status", "cde"),

94      }

96      @Test
97      void findsNoArgumentWhenNameHitsFilledVoid() {
98          MatcherAssert.assertThat(
99              "a call that binds by name a void its positional argument fills must put nothing in",
100             new Calls(
101                 Map.of("Φ.status", List.of("cde", "msg")),
102                 Map.of("Φ.app.φ", Map.of(0, "Φ.app.φ.α0")),
103                 Map.of("Φ.app.φ", Map.of("cde", "Φ.app.φ.c")),
104                 Map.of()
105             ).argument("Φ.app.φ", "Φ.status", "cde"),
106             Matchers.equalTo(Optional.empty())
107         );
108     }

110     @Test
111     void leavesOutVoidPastLastPositionalArgument() {
112         MatcherAssert.assertThat(
113             "a call with fewer arguments than voids must leave the last void empty, but it doesnt",
114             new Calls(
115                 Map.of("Φ.string.at", List.of("i", "fallback")),
116                 Map.of("Φ.app.φ", Map.of(0, "Φ.app.φ.α0")),
117                 Map.of(),
118                 Map.of()
119             ).leaves("Φ.app.φ", "Φ.string.at", "fallback"),
120             Matchers.is(true)
121         );
122     }
123 
124     @Test
125     void doesNotLeaveOutVoidThatGetsArgument() {
126         MatcherAssert.assertThat(
127             "a call must not leave empty a void its positional argument goes into, but it does",
128             new Calls(
129                 Map.of("Φ.string.at", List.of("i", "fallback")),
130                 Map.of("Φ.app.φ", Map.of(0, "Φ.app.φ.α0")),
131                 Map.of(),
132                 Map.of()
133             ).leaves("Φ.app.φ", "Φ.string.at", "i"),
134             Matchers.is(false)
135         );
136     }
137 
138     @Test
139     void doesNotLeaveOutVoidBoundByName() {
140         MatcherAssert.assertThat(
141             "a call must not leave empty a void it binds by name, but it does",
142             new Calls(
143                 Map.of("Φ.string.at", List.of("i", "fallback")),
144                 Map.of("Φ.app.φ", Map.of(0, "Φ.app.φ.α0")),
145                 Map.of("Φ.app.φ", Map.of("fallback", "Φ.app.φ.f")),

147             ).leaves("Φ.app.φ", "Φ.string.at", "fallback"),

150     }
151 
152     @Test
153     void findsTypeBracketsGiveVoid() {
154         final String kind = String.format("Φ.kind%d", new SecureRandom().nextInt(10_000));
155         MatcherAssert.assertThat(
156             "a void must hold the type the brackets give at its position, but it doesnt",
157             new Calls(
158                 Map.of("Φ.status", List.of("cde", "msg")), Map.of(), Map.of(), Map.of()
159             ).member(new Annotation(String.format("Φ.status(Φ.number,%s)", kind)), "msg"),
160             Matchers.equalTo(Optional.of(new Annotation(kind)))
161         );
162     }
