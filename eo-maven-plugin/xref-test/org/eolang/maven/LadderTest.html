5   package org.eolang.maven;
6   
7   import java.util.LinkedHashMap;
8   import java.util.Map;
9   import org.hamcrest.MatcherAssert;
10  import org.hamcrest.Matchers;
11  import org.junit.jupiter.api.Test;
12  
13  /**
14   * Test case for {@link Ladder}.
15   *
16   * @since 0.76.0
17   */
18  final class LadderTest {
19  
20      @Test
21      void startsWithTheCountOfAllObjects() {
22          MatcherAssert.assertThat(
23              "the ladder must open with every object of the program, but it didnt",
24              new Ladder(LadderTest.tally(13, 39)).iterator().next(),
25              Matchers.equalTo("52 objects in all")
26          );
27      }
28  
29      @Test
30      void writesTheShareOfEachBand() {
31          MatcherAssert.assertThat(
32              "the ladder must write the share of a band out of a hundred, but it didnt",
33              new Ladder(LadderTest.tally(13, 39)),
34              Matchers.hasItem("75.0 % in gaps")
35          );
36      }
37  
38      @Test
39      void writesTheCountOfEachBand() {
40          MatcherAssert.assertThat(
41              "the ladder must write how many objects a band holds, but it didnt",
42              new Ladder(LadderTest.tally(13, 39)),
43              Matchers.hasItem("13 objects in named")
44          );
45      }
46  
47      @Test
48      void sharesNothingOfAnEmptyProgram() {
49          MatcherAssert.assertThat(
50              "the ladder of a program with no objects must share nothing, but it didnt",
51              new Ladder(LadderTest.tally(0, 0)),
52              Matchers.hasItem("0.0 % in named")
53          );
54      }
55  
56      @Test
57      void writesTheSharesBeforeTheCounts() {
58          MatcherAssert.assertThat(
59              "the ladder must write every share before any count, but it didnt",
60              new Ladder(LadderTest.tally(7, 2)),
61              Matchers.contains(
62                  "9 objects in all",
63                  "77.8 % in named",
64                  "22.2 % in gaps",
65                  "7 objects in named",
66                  "2 objects in gaps"
67              )
68          );
69      }
70  
71      private static Map<String, Integer> tally(final int named, final int gaps) {
72          final Map<String, Integer> tally = new LinkedHashMap<>(2);
73          tally.put("named", named);
74          tally.put("gaps", gaps);
75          return tally;
76      }
77  }
