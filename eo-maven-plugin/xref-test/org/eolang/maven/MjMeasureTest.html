29      @Test
30      void doesNothingWhenNoCoverageFileIsSet(@Mktmp final Path temp) throws IOException {
31          new FakeMaven(temp).withProgram(
32              String.join(System.lineSeparator(), "[] > x", "  42 > y")
33          ).execute(MjParse.class).execute(MjMeasure.class);
34          MatcherAssert.assertThat(
35              "no LCOV report must be written when eo.measure.hits is unset, but one was",
36              Files.exists(temp.resolve("target/coverage.info")),
37              Matchers.is(false)
38          );
39      }

41      @Test
42      void marksAHitLocationAsCoveredInTheLcovReport(@Mktmp final Path temp) throws Exception {
43          final FakeMaven maven = new FakeMaven(temp).withProgram(
44              String.join(System.lineSeparator(), "[] > x", "  42 > y")
45          ).execute(MjParse.class);
46          final String location = new CoverageManifest().locations(
47              new XMLDocument(maven.programTojo().xmir())
48          ).iterator().next();
49          final Path hits = temp.resolve("coverage.txt");
50          Files.writeString(hits, String.format("%s%n", location));
51          final Path lcov = temp.resolve("coverage.info");
52          maven.with("hits", hits.toFile())
53              .with("lcov", lcov.toFile())
54              .execute(MjMeasure.class);
55          MatcherAssert.assertThat(
56              "the LCOV report must record at least one hit line, but LH:0 everywhere",
57              Files.readString(lcov),
58              Matchers.not(Matchers.containsString("LH:0"))
59          );
60      }

62      @Test
63      void countsEveryRepeatOfTheSameLocation(@Mktmp final Path temp) throws Exception {
64          final FakeMaven maven = new FakeMaven(temp).withProgram(
65              String.join(System.lineSeparator(), "[] > x", "  42 > y")
66          ).execute(MjParse.class);
67          final String location = new CoverageManifest().locations(
68              new XMLDocument(maven.programTojo().xmir())
69          ).iterator().next();
70          final int last = location.lastIndexOf(':');
71          final String line = location.substring(
72              location.lastIndexOf(':', last - 1) + 1, last
73          );
74          final int repeats = 10_000;
75          final Path hits = temp.resolve("coverage.txt");
76          Files.write(hits, Collections.nCopies(repeats, location));
77          final Path lcov = temp.resolve("coverage.info");
78          maven.with("hits", hits.toFile())
79              .with("lcov", lcov.toFile())
80              .execute(MjMeasure.class);
81          MatcherAssert.assertThat(
82              "every repeat of a location must be counted, however many of them the raw hits file holds, but they werent",
83              Files.readString(lcov),
84              Matchers.containsString(
85                  String.format("DA:%s,%d", line, repeats)
86              )
87          );
88      }

161     @Test
162     void namesTheSourceAndLineWhenItIsNotUnicode(@Mktmp final Path temp) throws Exception {
163         final FakeMaven maven = new FakeMaven(temp).withProgram(
164             String.join(System.lineSeparator(), "[] > x", "  42 > y")
165         ).execute(MjParse.class);
166         Files.write(
167             temp.resolve("foo/x/main.eo"),
168             String.join(
169                 System.lineSeparator(),
170                 "[] > x",
171                 String.format("  \"%c\" > y", 0xE9),
172                 ""
173             ).getBytes(StandardCharsets.ISO_8859_1)
174         );
175         final Path hits = temp.resolve("coverage.txt");
176         Files.writeString(hits, "");
177         MatcherAssert.assertThat(
178             "a source that is not UTF-8 must be named with its line, but it wasnt",
179             String.join(
180                 System.lineSeparator(),
181                 new Causes(
182                     Assertions.assertThrows(
183                         IllegalStateException.class,
184                         () -> maven.with("hits", hits.toFile())
185                             .with("lcov", temp.resolve("coverage.info").toFile())
186                             .execute(MjMeasure.class),
187                         "measure must fail on a source that is not UTF-8, but it didnt"
188                     )
189                 )
190             ),
191             Matchers.allOf(
192                 Matchers.containsString("main.eo"),
193                 Matchers.containsString(":2:"),
194                 Matchers.containsString("not valid UTF-8")
195             )
196         );
197     }

241     @Test
242     void doesNotAttributeMergedMembersToThePackageFile(@Mktmp final Path temp) throws Exception {
243         final FakeMaven maven = new FakeMaven(temp).withProgram(
244             String.join(
245                 System.lineSeparator(),
246                 "+architect me@example.com",
247                 "+version 0.0.0",
248                 "",
249                 "[n] > foo",
250                 "  42 > x",
251                 "  n > @"
252             ),
253             "foo",
254             "foo.eo"
255         ).withProgram(
256             String.join(
257                 System.lineSeparator(),
258                 "+package foo",
259                 "",
260                 "[] > bar",
261                 "  42 > @"
262             ),
263             "foo.bar",
264             "foo/bar.eo"
265         ).execute(MjParse.class).execute(new PpMerge());
266         final Path hits = temp.resolve("coverage.txt");
267         Files.writeString(hits, "");
268         final Path lcov = temp.resolve("coverage.info");
269         maven.with("hits", hits.toFile())
270             .with("lcov", lcov.toFile())
271             .execute(MjMeasure.class);
272         MatcherAssert.assertThat(
273             "the meta directives of the package file must not be reported as DA misses, while its merged member must get a coverage block of its own",
274             Files.readString(lcov),
275             Matchers.allOf(
276                 Matchers.matchesPattern(
277                     "(?s).*SF:[^\\n]*foo\\.eo\\r?\\n(?:(?!DA:[123],0).)*end_of_record.*"
278                 ),
279                 Matchers.containsString("bar.eo")
280             )
281         );
