9   import java.io.IOException;
10  import java.nio.file.Files;
11  import java.nio.file.Path;
12  import java.nio.file.Paths;
13  import java.util.Collections;
14  import org.eolang.parser.EoSyntax;
15  import org.hamcrest.MatcherAssert;
16  import org.hamcrest.Matchers;
17  import org.junit.jupiter.api.Assertions;
18  import org.junit.jupiter.api.Test;
19  import org.junit.jupiter.api.extension.ExtendWith;
20  
21  /**
22   * Test case for {@link Transpilation}.
23   *
24   * @since 0.74
25   */
26  @ExtendWith(MktmpResolver.class)
27  final class TranspilationTest {
28  
29      @Test
30      void tellsTrackedStepsApartInTheCacheKey() {
31          MatcherAssert.assertThat(
32              "a build that writes the XMIRs of the train must not take the result of one that didnt",
33              this.transpilation(new Tracking(true, false)).version(),
34              Matchers.not(
35                  Matchers.equalTo(this.transpilation(new Tracking(false, false)).version())
36              )
37          );
38      }
39  
40      @Test
41      void tellsInferenceTablesApartInTheCacheKey(@Mktmp final Path temp) throws IOException {
42          final Path tables = Files.createDirectories(temp.resolve("tables"));
43          Files.writeString(
44              tables.resolve("answers.xml"),
45              "<answers><o loc='Q.f' copies='1'><is loc='Q.f'/></o></answers>"
46          );
47          MatcherAssert.assertThat(
48              "a build that read the rows of this object must not take one that had none",
49              this.transpilation(tables).version(Collections.singletonList("Q.f")),
50              Matchers.not(
51                  Matchers.equalTo(
52                      this.transpilation(temp.resolve("absent"))
53                          .version(Collections.singletonList("Q.f"))
54                  )
55              )
56          );
57      }
58  
59      @Test
60      void tellsDeadlinesApartInTheCacheKey() {
61          MatcherAssert.assertThat(
62              "a build whose tests run under one deadline took the result of one whose tests run under another",
63              this.transpilation(3L, "1G").version(),
64              Matchers.not(Matchers.equalTo(this.transpilation(11L, "1G").version()))
65          );
66      }
67  
68      @Test
69      void tellsMemoryBudgetsApartInTheCacheKey() {
70          MatcherAssert.assertThat(
71              "a build whose tests run under one memory budget took the result of one whose tests run under another",
72              this.transpilation(1L, "257M").version(),
73              Matchers.not(Matchers.equalTo(this.transpilation(1L, "3G").version()))
74          );
75      }
76  
77      @Test
78      void passesGivenDeadlineToGeneratedTimeout(@Mktmp final Path temp) throws IOException {
79          MatcherAssert.assertThat(
80              "The given deadline didnt reach the timeout of a generated test",
81              TranspilationTest.tested(this.transpilation(13L, "1G", temp)),
82              Matchers.containsString("@Timeout(value = 13, unit = TimeUnit.SECONDS)")
83          );
84      }
85  
86      @Test
87      void passesGivenMemoryToGeneratedBudget(@Mktmp final Path temp) throws IOException {
88          MatcherAssert.assertThat(
89              "The given memory budget didnt reach the budget of a generated test",
90              TranspilationTest.tested(this.transpilation(1L, "389M", temp)),
91              Matchers.containsString("@Budget(\"389M\")")
92          );
93      }
94  
95      @Test
96      void foldsInImportedXslLibrariesIntoVersion() {
97          MatcherAssert.assertThat(
98              "the cache-key version must differ from a fingerprint of the top-level XSLS alone, proving the xsl:import-ed libraries are actually folded in",
99              this.transpilation(new Tracking(false, false)).version(),
100             Matchers.not(
101                 Matchers.startsWith(new Fingerprint(Transpilation.XSLS).get())
102             )
103         );
104     }
105 
106     @Test
107     void buildsSourceFunctionForParentlessMeasuresPath() {
108         Assertions.assertDoesNotThrow(
109             () -> new Transpilation(
110                 new Tracking(false, false),
111                 false,
112                 "PhDefault",
113                 1L,
114                 "1G",
115                 Paths.get("xsl-measures.csv"),
116                 Paths.get("target"),
117                 Paths.get("target/eo/6-inference")
118             ).forSource("foo"),
119             "forSource() must not throw when eo.measures is a bare relative path with no parent directory"
120         );
121     }
122 
123     private Transpilation transpilation(final Tracking tracking) {
124         return new Transpilation(
125             tracking,
126             false,
127             "PhDefault",
128             1L,
129             "1G",
130             Paths.get("xsl-measures.csv"),
131             Paths.get("target"),
132             Paths.get("target/eo/6-inference")
133         );
134     }
135 
136     private Transpilation transpilation(final long deadline, final String memory) {
137         return new Transpilation(
138             new Tracking(false, false),
139             false,
140             "PhDefault",
141             deadline,
142             memory,
143             Paths.get("xsl-measures.csv"),
144             Paths.get("target"),
145             Paths.get("target/eo/6-inference")
146         );
147     }
148 
149     private Transpilation transpilation(
150         final long deadline, final String memory, final Path temp
151     ) {
152         return new Transpilation(
153             new Tracking(false, false),
154             false,
155             "PhDefault",
156             deadline,
157             memory,
158             temp.resolve("xsl-measures.csv"),
159             temp.resolve("target"),
160             temp.resolve("inference")
161         );
162     }
163 
164     private static String tested(final Transpilation train) throws IOException {
165         return String.join(
166             "",
167             train.forSource("foo").apply(
168                 new EoSyntax(
169                     String.join(
170                         System.lineSeparator(),
171                         "[] > foo", "  [] +> works", "    true > @", ""
172                     )
173                 ).parsed()
174             ).xpath("//tests/text()")
175         );
176     }
177 
178     private Transpilation transpilation(final Path tables) {
179         return new Transpilation(
180             new Tracking(false, false),
181             false,
182             "PhDefault",
183             1L,
184             "1G",
185             Paths.get("xsl-measures.csv"),
186             Paths.get("target"),
187             tables
188         );
189     }
