5   package org.eolang.maven;
6   
7   import com.yegor256.Together;
8   import com.yegor256.WeAreOnline;
9   import javax.net.ssl.SSLException;
10  import org.hamcrest.MatcherAssert;
11  import org.hamcrest.Matchers;
12  import org.junit.jupiter.api.Assertions;
13  import org.junit.jupiter.api.Test;
14  import org.junit.jupiter.api.extension.ExtendWith;
15  
16  /**
17   * Test case for {@link CommitHashesText}.
18   *
19   * @since 0.37.0
20   */
21  final class CommitHashesTextTest {
22  
23      @Test
24      void fallsBackOnATransportFailureThatIsNotUnknownHost() throws Exception {
25          MatcherAssert.assertThat(
26              "A transport failure other than an unknown host must still fall back to the built-in table, not propagate and fail the build",
27              new CommitHashesText(
28                  () -> {
29                      throw new SSLException("handshake failed");
30                  }
31              ).asString(),
32              Matchers.containsString("0.63.2")
33          );
34      }
35  
36      @Test
37      void findsEveryTagOfTheFallbackTable() {
38          MatcherAssert.assertThat(
39              "ChText must find a tag in the built-in fallback table, whatever the platform's line separator is",
40              new ChText(
41                  new CommitHashesText(
42                      () -> {
43                          throw new SSLException("handshake failed");
44                      }
45                  )::asString,
46                  "0.40.5"
47              ).value(),
48              Matchers.equalTo("ee14a1e30a9e0f8f64404e7a52f40c2a07f88359")
49          );
50      }
51  
52      @Test
53      void findsTheDefaultTagInTheFallbackTable() {
54          MatcherAssert.assertThat(
55              "The fallback table must hold a SHA for 'master', the default eo.tag, while it holds none",
56              new ChText(
57                  new CommitHashesText(
58                      () -> {
59                          throw new SSLException("handshake failed");
60                      }
61                  )::asString,
62                  "master"
63              ).value(),
64              Matchers.matchesRegex("[0-9a-f]{40}")
65          );
66      }
67  
68      @Test
69      void propagatesAFailureThatIsNotAnIoError() {
70          Assertions.assertThrows(
71              IllegalStateException.class,
72              () -> new CommitHashesText(
73                  () -> {
74                      throw new IllegalStateException("not an I/O failure");
75                  }
76              ).asString(),
77              "A non-I/O failure must not be swallowed into the fallback table"
78          );
79      }
80  
81      @Test
82      @ExtendWith(WeAreOnline.class)
83      void downloadsDefaultList() throws Exception {
84          MatcherAssert.assertThat(
85              "CommitHashesText downloads the default list of hashes from Objectionary",
86              new CommitHashesText().asString(),
87              Matchers.containsString("master")
88          );
89      }
90  
91      @Test
92      @ExtendWith(WeAreOnline.class)
93      void isThreadSafe() {
94          final CommitHashesText text = new CommitHashesText();
95          MatcherAssert.assertThat(
96              "Can be used in different threads without NPE",
97              new Together<>(
98                  thread -> text.asString() != null
99              ),
100             Matchers.not(Matchers.hasItems(false))
101         );
102     }
103 }
