20  /**
21   * Test cases for {@link Fillers}.
22   *
23   * <p>Every case is about {@code [x cant] > safe}, whose void {@code cant}
24   * one call leaves empty, as {@code safe 5} does, while others fill it. The
25   * call at {@code Φ.app.a} passes one argument, into {@code x}, and so
26   * leaves {@code cant} empty. Any other call passes two, and its second
27   * argument, the one in {@code cant}, is a terminator at {@code Φ.app.t}, a
28   * copy of {@code fix} at {@code Φ.app.f}, and unknown anywhere else.</p>
29   *
30   * @since 0.76.0
31   */
32  final class FillersTest {
33  
34      @Test
35      void holdsTerminatorWhenOneCallPassesItAndAnotherLeavesVoidEmpty() {
36          MatcherAssert.assertThat(
37              "a void left empty must hold the terminator when a call puts the terminator there, but it doesnt",
38              FillersTest.fillers(
39                  Set.of("Φ.app.a", "Φ.app.t"), Set.of()
40              ).held("Φ.safe.cant", Map.entry("Φ.safe", "cant")).getKey(),
41              Matchers.equalTo(Optional.of(Copy.terminator()))
42          );
43      }
44  
45      @Test
46      void keepsVoidUnknownWhenNoCallPassesTerminator() {
47          MatcherAssert.assertThat(
48              "a void left empty must stay unknown when no call puts the terminator there, but it doesnt",
49              FillersTest.fillers(
50                  Set.of("Φ.app.a", "Φ.app.f"), Set.of()
51              ).held("Φ.safe.cant", Map.entry("Φ.safe", "cant")).getKey(),
52              Matchers.equalTo(Optional.empty())
53          );
54      }
55  
56      @Test
57      void namesVoidThatLacksCallWithTerminator() {
58          final Fillers fillers = FillersTest.fillers(
59              Set.of("Φ.app.a", "Φ.app.f"), Set.of()
60          );
61          fillers.held("Φ.safe.cant", Map.entry("Φ.safe", "cant"));
62          MatcherAssert.assertThat(
63              "a void that a test with the terminator would make known must be named, but it isnt",
64              fillers.vacant(),
65              Matchers.contains(Map.entry("Φ.safe", "cant"))
66          );
67      }
68  
69      @Test
70      void holdsWhatOtherCallsPutWhenBareCopyLeavesVoidEmpty() {
71          MatcherAssert.assertThat(
72              "a void a bare copy leaves empty must hold what the other calls put there, when one of them puts the terminator there, but it doesnt",
73              FillersTest.fillers(
74                  Set.of("Φ.app.t", "Φ.app.f"),
75                  Set.of("Φ.safe")
76              ).held("Φ.safe.cant", Map.entry("Φ.safe", "cant")).getKey(),
77              Matchers.equalTo(Optional.of(new Copy(List.of("Φ.fix")).ending(true)))
78          );
79      }
80  
81      @Test
82      void keepsVoidUnknownWhenSomeArgumentIsUnknown() {
83          MatcherAssert.assertThat(
84              "a void must stay unknown when a call puts an unknown argument there, even with the terminator from another call, but it doesnt",
85              FillersTest.fillers(
86                  Set.of("Φ.app.t", "Φ.app.u"), Set.of()
87              ).held("Φ.safe.cant", Map.entry("Φ.safe", "cant")).getKey(),
88              Matchers.equalTo(Optional.empty())
89          );
90      }
91  
92      @Test
93      void keepsVoidUnknownWhenNothingFillsIt() {
94          MatcherAssert.assertThat(
95              "a void that no call fills must stay unknown, but it doesnt",
96              FillersTest.fillers(Set.of(), Set.of()).held(
97                  "Φ.safe.cant", Map.entry("Φ.safe", "cant")
98              ).getKey(),
99              Matchers.equalTo(Optional.empty())
100         );
101     }
102 
103     private static Fillers fillers(final Set<String> calls, final Set<String> bare) {
104         final Map<String, Map<Integer, String>> positional = new HashMap<>(0);
105         for (final String call : calls) {
106             final Map<Integer, String> args = new HashMap<>(0);
107             args.put(0, String.format("%s.α0", call));
108             if (!"Φ.app.a".equals(call)) {
109                 args.put(1, String.format("%s.α1", call));
110             }
111             positional.put(call, args);
112         }
113         return new Fillers(
114             Map.of(
115                 "Φ.safe", Map.of("x", "Φ.safe.x", "cant", "Φ.safe.cant"),
116                 "Φ.fix", Map.of()
117             ),
118             new Calls(
119                 Map.of("Φ.safe", List.of("x", "cant")),
120                 positional, Collections.emptyMap(), Collections.emptyMap()
121             ),
122             Map.of("Φ.safe", new ArrayList<>(new TreeSet<>(calls))),
123             bare,
124             (place, seen) -> {
125                 final Optional<Copy> found;
126                 if ("Φ.app.t.α1".equals(place)) {
127                     found = Optional.of(Copy.terminator());
128                 } else if ("Φ.app.f.α1".equals(place)) {
129                     found = Optional.of(new Copy(new Annotation("Φ.fix")));
130                 } else {
131                     found = Optional.empty();
132                 }
133                 return found;
134             }
135         );
136     }
