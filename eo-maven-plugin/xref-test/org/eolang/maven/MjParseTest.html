70      @Test
71      void failsOnInvalidUnicodeWithFileAndLine(@Mktmp final Path temp) throws IOException {
72          final FakeMaven maven = new FakeMaven(temp).withProgram("[] > a");
73          Files.write(
74              temp.resolve("foo/x/main.eo"),
75              new byte[] {
76                  '[', ']', ' ', '>', ' ', 'a', '\n',
77                  ' ', ' ', '"', (byte) 0xE9, '"', ' ', '>', ' ', 'x', '\n',
78              }
79          );
80          final Throwable thrown = Assertions.assertThrows(
81              IllegalStateException.class,
82              () -> maven.execute(new PpParse())
83          );
84          final StringBuilder chain = new StringBuilder();
85          for (Throwable cause = thrown; cause != null; cause = cause.getCause()) {
86              chain.append(cause.getMessage()).append('\n');
87          }
88          MatcherAssert.assertThat(
89              "a source that is not UTF-8 must fail with its file and line, not parse as U+FFFD",
90              chain.toString(),
91              Matchers.allOf(
92                  Matchers.containsString("main.eo"),
93                  Matchers.containsString(":2:"),
94                  Matchers.containsString("not valid UTF-8")
95              )
96          );
97      }

141     @Test
142     @ExtendWith(WeAreOnline.class)
143     void parsesWithCache(@Mktmp final Path temp) throws Exception {
144         final Path cache = temp.resolve("cache");
145         final FakeMaven maven = new FakeMaven(temp)
146             .withProgram("invalid content")
147             .with("cache", cache.toFile());
148         final String expected = new UncheckedText(
149             new TextOf(new ResourceOf("org/eolang/maven/main.xmir"))
150         ).asString();
151         final CommitHash hash = new ChCached(new ChNarrow(new ChRemote("0.40.5")));
152         final Path base = maven.targetPath().resolve(maven.dirName("parse"));
153         final Path target = new Place("foo.x.main").make(base, MjAssemble.XMIR);
154         new Cache(
155             cache.resolve(Parsing.CACHE)
156                 .resolve(MjParseTest.cacheVersion(maven.programTojo().identifier()))
157                 .resolve(hash.value()),
158             src -> expected
159         ).apply(maven.programTojo().source(), target, base.relativize(target));
160         target.toFile().delete();
161         final String actual = String.format(
162             "target/%s/foo/x/main.%s",
163             maven.dirName("parse"),
164             MjAssemble.XMIR
165         );
166         MatcherAssert.assertThat(
167             String.format("We expect that that %s is taken from the cache, but it didn't", actual),
168             new TextOf(
169                 maven.allTojosWithHash(hash)
170                     .execute(new PpParse())
171                     .result()
172                     .get(actual)
173             ).toString(),
174             Matchers.equalTo(expected)
175         );
176     }

325      * The test with high number of eo programs reveals concurrency problems of the ParseMojo.
326      * Since other tests works only with single program - it's hard to find concurrency mistakes.
327      *
328      * @param temp Test directory
329      * @throws IOException If problem with filesystem happened.
330      */
331     @Test
332     void parsesConcurrentlyWithLotsOfPrograms(@Mktmp final Path temp) throws IOException {
333         final FakeMaven maven = new FakeMaven(temp);
334         final int total = 50;
335         final Collection<String> xmirs = new ArrayList<>(total);
336         for (int program = 0; program < total; ++program) {
337             maven.withProgram(
338                 String.format("+package foo.x%n%n[] > main%s", FakeMaven.suffix(program))
339             );
340             xmirs.add(
341                 String.format(
342                     "target/%s/foo/x/main%s.%s",
343                     maven.dirName("parse"),
344                     FakeMaven.suffix(program),
345                     MjAssemble.XMIR
346                 )
347             );
348         }
349         MatcherAssert.assertThat(
350             "All programs must be parsed in a single concurrent run, but some XMIRs are absent",
351             maven.execute(new PpParse()).result().keySet(),
352             Matchers.hasItems(xmirs.toArray(new String[0]))
353         );
354     }

429     @Test
430     void logsParserWarningWithoutCallingItAFailure(@Mktmp final Path temp) throws Exception {
431         final List<String> messages = new ArrayList<>(0);
432         final String home = com.jcabi.log.Logger.format("%[file]s", temp);
433         final Appender appender = new AppenderSkeleton() {
434             @Override
435             protected void append(final LoggingEvent event) {
436                 if (event.getRenderedMessage().contains(home)) {
437                     messages.add(
438                         String.format("[%s] %s", event.getLevel(), event.getRenderedMessage())
439                     );
440                 }
441             }
442 
443             @Override
444             public void close() {
445                 // Nothing to release.
446             }
447 
448             @Override
449             public boolean requiresLayout() {
450                 return false;
451             }
452         };
453         final Logger logger = Logger.getLogger(Parsing.class);
454         final Level level = logger.getLevel();
455         logger.setLevel(Level.ALL);
456         logger.addAppender(appender);
457         try {
458             new FakeMaven(temp).withProgram(
459                 String.join(
460                     System.lineSeparator(),
461                     "+package foo.x",
462                     "",
463                     "[x] > main",
464                     "  x?.y.a > @"
465                 )
466             ).execute(new PpParse());
467         } finally {
468             logger.removeAppender(appender);
469             logger.setLevel(level);
470         }
471         MatcherAssert.assertThat(
472             "A parser warning must be logged as a warning, not as a parsing failure, but it was not",
473             messages.stream()
474                 .filter(line -> line.contains(temp.getFileName().toString()))
475                 .collect(Collectors.toList()),
476             Matchers.both(
477                 Matchers.<String>hasItem(
478                     Matchers.allOf(
479                         Matchers.containsString("[WARN]"),
480                         Matchers.containsString("fragile object requires")
481                     )
482                 )
483             ).and(
484                 Matchers.not(
485                     Matchers.<String>hasItem(Matchers.containsString("Failed to parse"))
486                 )
487             )
488         );
489     }

491     private static String cacheVersion(final String identifier) {
492         return String.format(
493             "%s-%s-%s",
494             FakeMaven.pluginVersion(),
495             new Fingerprint(
496                 Stream.concat(
497                     Canonical.XSLS.stream(), Canonical.IMPORTS.stream()
498                 ).toArray(String[]::new)
499             ).get(),
500             new UncheckedText(
501                 new HexOf(new Sha256DigestOf(new InputOf(identifier)))
502             ).asString()
503         );
504     }

506     /**
507      * The mojo that does nothing, but executes infinitely.
508      *
509      * @since 0.29
510      */
511     @Mojo(name = "infinite", defaultPhase = LifecyclePhase.VALIDATE)
512     private static final class Infinite extends MjSafe {
513 
514         @Override
515         public void exec() {
516             try {
517                 Thread.sleep(Long.MAX_VALUE);
518             } catch (final InterruptedException ex) {
519                 Thread.currentThread().interrupt();
520                 throw new IllegalStateException(ex);
521             }
522         }
523     }
