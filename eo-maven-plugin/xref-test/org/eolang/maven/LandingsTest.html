21  /**
22   * Test cases for {@link Landings}.
23   *
24   * @since 0.69.0
25   */
26  @ExtendWith(MktmpResolver.class)
27  final class LandingsTest {
28  
29      @Test
30      void namesArgumentAfterVoidItLandsIn(@Mktmp final Path temp) throws IOException {
31          final String hollow = String.format("lid%d", new SecureRandom().nextInt(10_000));
32          MatcherAssert.assertThat(
33              "the argument must take the name of the void it lands in, but it didnt",
34              new Landings(
35                  LandingsTest.tables(
36                      temp,
37                      LandingsTest.grounded("Φ.k.φ", "Φ.k.φ.α0"),
38                      String.format(
39                          "<o loc='Φ.k.φ.α0' copies='3' lands='%s'><is loc='Φ.number'/></o>",
40                          hollow
41                      )
42                  )
43              ).names(),
44              Matchers.hasEntry("Φ.k.φ.α0", hollow)
45          );
46      }
47  
48      @Test
49      void leavesArgumentThatLandsNowhereUnnamed(@Mktmp final Path temp) throws IOException {
50          MatcherAssert.assertThat(
51              "an argument whose row names no void must keep its place, but it was named",
52              new Landings(
53                  LandingsTest.tables(
54                      temp,
55                      LandingsTest.grounded("Φ.q.φ", "Φ.q.φ.α1"),
56                      "<o loc='Φ.q.φ.α1' copies='2'>",
57                      "<set copies='1'><terminator/></set>",
58                      "<set copies='1'><is loc='Φ.q.box'/></set>",
59                      "</o>"
60                  )
61              ).names(),
62              Matchers.not(Matchers.hasKey("Φ.q.φ.α1"))
63          );
64      }
65  
66      @Test
67      void tellsArgumentsOfOneCallApart(@Mktmp final Path temp) throws IOException {
68          MatcherAssert.assertThat(
69              "the second argument of a call must take the name of its own void, but it didnt",
70              new Landings(
71                  LandingsTest.tables(
72                      temp,
73                      LandingsTest.grounded("Φ.z.φ", "Φ.z.φ.α0", "Φ.z.φ.α1"),
74                      "<o loc='Φ.z.φ' copies='1'><is loc='Φ.z.pair'/></o>",
75                      "<o loc='Φ.z.φ.α0' copies='1' lands='left'><data/></o>",
76                      "<o loc='Φ.z.φ.α1' copies='1' lands='v🌵right'><data/></o>"
77                  )
78              ).names(),
79              Matchers.hasEntry("Φ.z.φ.α1", "v🌵right")
80          );
81      }
82  
83      @Test
84      void leavesArgumentOfCallOnVoidUnnamed(@Mktmp final Path temp) throws IOException {
85          MatcherAssert.assertThat(
86              "an argument of a call on a void must keep its place, but it was named",
87              new Landings(
88                  LandingsTest.tables(
89                      temp,
90                      String.join(
91                          "",
92                          "<facts><objects><type id='Φ.guard'>",
93                          "<attr name='fail' type='Φ.guard.fail' void='true'/>",
94                          "<attr name='φ' type='Φ.guard.φ'/>",
95                          "</type></objects>",
96                          "<reads><read id='Φ.guard.φ' name='fail' of='Φ.guard.φ.ρ'/></reads>",
97                          "<heads><type id='Φ.guard.φ.ρ'><ref loc='Φ.guard'/></type></heads>",
98                          "<calls><call loc='Φ.guard.φ'><arg loc='Φ.guard.φ.α0' n='0'/></call>",
99                          "</calls></facts>"
100                     ),
101                     "<o loc='Φ.guard.φ.α0' copies='1' lands='message'><data/></o>"
102                 )
103             ).names(),
104             Matchers.not(Matchers.hasKey("Φ.guard.φ.α0"))
105         );
106     }
107 
108     static Path tables(
109         final Path dir, final String facts, final String... rows
110     ) throws IOException {
111         Files.write(dir.resolve("facts.xml"), facts.getBytes(StandardCharsets.UTF_8));
112         Files.write(
113             dir.resolve("answers.xml"),
114             String.format("<answers>%s</answers>", String.join("", rows))
115                 .getBytes(StandardCharsets.UTF_8)
116         );
117         return dir;
118     }
119 
120     static String grounded(final String call, final String... args) {
121         return String.format(
122             "<facts><objects><type id='Φ.made'/></objects><reads/><heads><type id='%1$s'><ref loc='Φ.made'/></type></heads><calls><call loc='%1$s'>%2$s</call></calls></facts>",
123             call,
124             IntStream.range(0, args.length)
125                 .mapToObj(idx -> String.format("<arg loc='%s' n='%d'/>", args[idx], idx))
126                 .collect(Collectors.joining())
127         );
128     }
