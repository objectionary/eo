5   package org.eolang.maven;
6   
7   import com.yegor256.Mktmp;
8   import com.yegor256.MktmpResolver;
9   import java.io.IOException;
10  import java.nio.file.Files;
11  import java.nio.file.Path;
12  import org.hamcrest.MatcherAssert;
13  import org.hamcrest.Matchers;
14  import org.junit.jupiter.api.Test;
15  import org.junit.jupiter.api.extension.ExtendWith;
16  
17  /**
18   * Test cases for {@link MjRender}.
19   *
20   * @since 0.71.0
21   */
22  @ExtendWith(MktmpResolver.class)
23  final class MjRenderTest {
24  
25      @Test
26      void drawsAPagePerSourceFromTheAnswers(@Mktmp final Path temp) throws IOException {
27          final FakeMaven maven = new FakeMaven(temp).withProgram(
28              String.join(System.lineSeparator(), "[] > app", "  [] > t", "")
29          );
30          maven.execute(MjParse.class).execute(MjInfer.class);
31          Files.writeString(
32              maven.targetPath().resolve(maven.dirName("inference")).resolve("answers.xml"),
33              String.join(
34                  "",
35                  "<answers><o loc='Φ.app' copies='1'><is loc='Φ.app'/></o>",
36                  "<o loc='Φ.app.t' copies='3'><is loc='Φ.app.t'/></o></answers>"
37              )
38          );
39          maven.execute(MjRender.class);
40          MatcherAssert.assertThat(
41              "a page of the program must be drawn from answers.xml, but it wasnt",
42              Files.exists(
43                  maven.targetPath().getParent().resolve("site/inference/foo/x/main.eo.html")
44              ),
45              Matchers.is(true)
46          );
47      }
48  
49      @Test
50      void drawsNothingWithoutAnswers(@Mktmp final Path temp) throws IOException {
51          final FakeMaven maven = new FakeMaven(temp).withProgram(
52              String.join(System.lineSeparator(), "[] > app", "  [] > t", "")
53          );
54          maven.execute(MjParse.class).execute(MjInfer.class);
55          Files.delete(
56              maven.targetPath().resolve(maven.dirName("inference")).resolve("answers.xml")
57          );
58          maven.execute(MjRender.class);
59          MatcherAssert.assertThat(
60              "no page may be drawn when the engine wrote no answers, but one was",
61              Files.exists(maven.targetPath().getParent().resolve("site/inference/index.html")),
62              Matchers.is(false)
63          );
64      }
65  
66      @Test
67      void createsNoStageDirectoryWhenInferenceNeverRan(@Mktmp final Path temp) throws IOException {
68          new FakeMaven(temp).execute(MjRender.class);
69          MatcherAssert.assertThat(
70              "A goal that only reads must leave the target untouched, but it didnt",
71              Files.exists(temp.resolve("target")),
72              Matchers.is(false)
73          );
74      }
75  }
