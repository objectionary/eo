5   package org.eolang.maven;
6   
7   import com.yegor256.Mktmp;
8   import com.yegor256.MktmpResolver;
9   import java.io.IOException;
10  import java.nio.charset.StandardCharsets;
11  import java.nio.file.Files;
12  import java.nio.file.Path;
13  import java.util.zip.ZipEntry;
14  import java.util.zip.ZipOutputStream;
15  import org.hamcrest.MatcherAssert;
16  import org.hamcrest.Matchers;
17  import org.junit.jupiter.api.Assertions;
18  import org.junit.jupiter.api.Test;
19  import org.junit.jupiter.api.extension.ExtendWith;
20  
21  /**
22   * Test case for {@link UnpackedJar}.
23   *
24   * @since 0.64
25   */
26  @ExtendWith(MktmpResolver.class)
27  final class UnpackedJarTest {
28  
29      @Test
30      void unpacksAnEntryIntoTheDestination(@Mktmp final Path temp) throws IOException {
31          final Path dest = temp.resolve("into");
32          UnpackedJarTest.archived(temp.resolve("plain.jar"), "a/b.txt");
33          new UnpackedJar(temp.resolve("plain.jar"), dest).unpack();
34          MatcherAssert.assertThat(
35              "the entry must land under the destination, but it didnt",
36              Files.readString(dest.resolve("a/b.txt"), StandardCharsets.UTF_8),
37              Matchers.equalTo("hello")
38          );
39      }
40  
41      @Test
42      void refusesAnEntryBehindALink(@Mktmp final Path temp) throws IOException {
43          final Path dest = temp.resolve("guarded");
44          Files.createDirectories(dest);
45          Files.createDirectories(temp.resolve("elsewhere"));
46          Files.createSymbolicLink(dest.resolve("linked"), temp.resolve("elsewhere"));
47          UnpackedJarTest.archived(temp.resolve("sneaky.jar"), "linked/result.txt");
48          Assertions.assertThrows(
49              IOException.class,
50              () -> new UnpackedJar(temp.resolve("sneaky.jar"), dest).unpack(),
51              "an entry whose parent is a link out of the destination must be refused"
52          );
53      }
54  
55      @Test
56      void writesNothingBehindALink(@Mktmp final Path temp) throws IOException {
57          final Path dest = temp.resolve("guarded");
58          final Path outside = temp.resolve("elsewhere");
59          Files.createDirectories(dest);
60          Files.createDirectories(outside);
61          Files.createSymbolicLink(dest.resolve("linked"), outside);
62          UnpackedJarTest.archived(temp.resolve("sneaky.jar"), "linked/result.txt");
63          MatcherAssert.assertThat(
64              "the directory the link points at must stay untouched, but it didnt",
65              UnpackedJarTest.leaked(temp.resolve("sneaky.jar"), dest, outside),
66              Matchers.is(false)
67          );
68      }
69  
70      private static boolean leaked(final Path jar, final Path dest, final Path outside) {
71          Assertions.assertThrows(
72              IOException.class,
73              () -> new UnpackedJar(jar, dest).unpack(),
74              "the entry must be refused before anything is written"
75          );
76          return Files.exists(outside.resolve("result.txt"));
77      }
78  
79      private static void archived(final Path jar, final String entry) throws IOException {
80          try (ZipOutputStream zos = new ZipOutputStream(Files.newOutputStream(jar))) {
81              zos.putNextEntry(new ZipEntry(entry));
82              zos.write("hello".getBytes(StandardCharsets.UTF_8));
83              zos.closeEntry();
84          }
85      }
86  }
