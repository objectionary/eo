19  /**
20   * Test cases for {@link Merging}.
21   *
22   * @since 0.68.0
23   */
24  final class MergingTest {
25  
26      @Test
27      void writesTheMergedXmirOnlyWhenItsContentChanges(@TempDir final Path temp) throws Exception {
28          final Path pkg = temp.resolve("pkg.xmir");
29          Files.write(
30              pkg,
31              new EoSyntax("[] > foo").parsed().toString().getBytes(StandardCharsets.UTF_8)
32          );
33          final Path member = temp.resolve("member.xmir");
34          Files.write(
35              member,
36              new EoSyntax("[] > bar").parsed().toString().getBytes(StandardCharsets.UTF_8)
37          );
38          final Path target = new Place("foo").make(
39              new Seqdir(temp, 2).dirs().once("merge"), MjAssemble.XMIR
40          );
41          this.merge(pkg, member, temp);
42          final FileTime before = Files.getLastModifiedTime(target);
43          Thread.sleep(1_100L);
44          this.merge(pkg, member, temp);
45          MatcherAssert.assertThat(
46              "Merged XMIR should not be rewritten when its content hasn't changed",
47              Files.getLastModifiedTime(target),
48              Matchers.equalTo(before)
49          );
50      }
51  
52      private void merge(
53          final Path pkg, final Path member, final Path base
54      ) throws IOException {
55          final TjsForeign tojos = new TjsForeign();
56          tojos.add("foo").withXmir(pkg);
57          tojos.add("foo.bar").withXmir(member);
58          new Merging(tojos, base).exec();
59      }
60  }
