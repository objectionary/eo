11  import java.io.IOException;
12  import java.nio.charset.StandardCharsets;

15  import java.util.Collections;
16  import org.hamcrest.MatcherAssert;

21  /**
22   * Test case for {@link Rows}.
23   *
24   * @since 0.75.0
25   */
26  @ExtendWith(MktmpResolver.class)
27  final class RowsTest {
28  
29      @Test
30      void takesTheRowsOfWhatTheObjectHolds(@Mktmp final Path temp) throws IOException {
31          MatcherAssert.assertThat(
32              "a row of an attribute of the object must change the digest of the object, but it didnt",
33              this.digest(
34                  temp.resolve("one"), "<o loc=\"Q.foo.φ\" copies=\"2\"><is loc=\"Q.number\"/></o>"
35              ),
36              Matchers.not(
37                  Matchers.equalTo(
38                      this.digest(
39                          temp.resolve("two"), "<o loc=\"Q.foo.φ\" copies=\"2\"><data/></o>"
40                      )
41                  )
42              )
43          );
44      }
45  
46      @Test
47      void leavesTheRowsOfANeighbourOut(@Mktmp final Path temp) throws IOException {
48          MatcherAssert.assertThat(
49              "a row of an object whose locator merely starts with the same letters changed the digest",
50              this.digest(
51                  temp.resolve("with"), "<o loc=\"Q.foobar\" copies=\"5\"><terminator/></o>"
52              ),
53              Matchers.equalTo(this.digest(temp.resolve("without"), ""))
54          );
55      }
56  
57      @Test
58      void readsPastAHyphenatedSibling(@Mktmp final Path temp) throws IOException {
59          final String neighbours = String.join(
60              "",
61              "<o loc=\"Q.foo\" copies=\"1\"><is loc=\"Q.foo\"/></o>",
62              "<o loc=\"Q.foo-bar\" copies=\"1\"><is loc=\"Q.foo-bar\"/></o>"
63          );
64          MatcherAssert.assertThat(
65              "a change in a descendant row must reach the digest, a hyphenated sibling hid it",
66              this.digest(
67                  temp.resolve("one"),
68                  String.format(
69                      "%s<o loc=\"Q.foo.inner\" copies=\"1\" lands=\"x\"><data/></o>", neighbours
70                  )
71              ),
72              Matchers.not(
73                  Matchers.equalTo(
74                      this.digest(
75                          temp.resolve("two"),
76                          String.format(
77                              "%s<o loc=\"Q.foo.inner\" copies=\"1\" lands=\"y\"><data/></o>",
78                              neighbours
79                          )
80                      )
81                  )
82              )
83          );
84      }
85  
86      @Test
87      void takesTheVoidsOfTheFacts(@Mktmp final Path temp) throws IOException {
88          MatcherAssert.assertThat(
89              "a void the facts add to a formation must change the digest of the formation, but it didnt",
90              this.facts(
91                  temp.resolve("one"),
92                  String.join(
93                      "",
94                      "<facts><objects><type id=\"Q.foo\">",
95                      "<attr name=\"x\" type=\"Q.foo.x\" void=\"true\"/>",
96                      "</type></objects></facts>"
97                  )
98              ),
99              Matchers.not(
100                 Matchers.equalTo(
101                     this.facts(
102                         temp.resolve("two"),
103                         "<facts><objects><type id=\"Q.foo\"/></objects></facts>"
104                     )
105                 )
106             )
107         );
108     }
109 
110     @Test
111     void leavesTheHeadsOfTheFactsOut(@Mktmp final Path temp) throws IOException {
112         MatcherAssert.assertThat(
113             "a row of the facts outside their objects changed the digest",
114             this.facts(
115                 temp.resolve("heads"),
116                 "<facts><heads><type id=\"Q.foo.φ\"><ref loc=\"Q\"/></type></heads></facts>"
117             ),
118             Matchers.equalTo(this.facts(temp.resolve("none"), "<facts/>"))
119         );
120     }
121 
122     @Test
123     void framesEveryDigestField(@Mktmp final Path temp) throws IOException {
124         final String locator = "Q.φ";
125         final Path dir = temp.resolve("framed");
126         Files.createDirectories(dir);
127         final Path table = dir.resolve("answers.xml");
128         Files.writeString(
129             table,
130             String.format(
131                 "<answers><o loc=\"%s\" copies=\"4\"><is loc=\"Q.string\"/></o></answers>",
132                 locator
133             )
134         );
135         final String value = String.format(
136             "answers.xml:%s",
137             new XMLDocument(table).nodes("/answers/o").get(0)
138         );
139         MatcherAssert.assertThat(
140             "each locator and row value must carry its UTF-8 byte length into the digest, but one didnt",
141             new Rows(dir).digest(Collections.singletonList(locator)),
142             Matchers.equalTo(
143                 new Hashed(
144                     String.format(
145                         "%d\0%s%d\0%s",
146                         locator.getBytes(StandardCharsets.UTF_8).length,
147                         locator,
148                         value.getBytes(StandardCharsets.UTF_8).length,
149                         value
150                     )
151                 ).get()
152             )
153         );
154     }
155 
156     @Test
157     void answersEveryThreadThatAsksAtOnce(@Mktmp final Path temp) throws IOException {
158         Files.writeString(
159             temp.resolve("answers.xml"),
160             "<answers><o loc=\"Q.foo\" copies=\"7\"><is loc=\"Q.foo\"/></o></answers>"
161         );
162         final Rows rows = new Rows(temp);
163         MatcherAssert.assertThat(
164             "threads that ask for a digest at the same moment must all be answered, one wasnt",
165             new Together<>(30, thread -> rows.digest(Collections.singletonList("Q.foo"))).asList(),
166             Matchers.everyItem(
167                 Matchers.equalTo(new Rows(temp).digest(Collections.singletonList("Q.foo")))
168             )
169         );
170     }
171 
172     private String digest(final Path dir, final String rows) throws IOException {
173         Files.createDirectories(dir);
174         Files.writeString(
175             dir.resolve("answers.xml"), String.format("<answers>%s</answers>", rows)
176         );
177         return new Rows(dir).digest(Collections.singletonList("Q.foo"));
178     }
179 
180     private String facts(final Path dir, final String table) throws IOException {
181         Files.createDirectories(dir);
182         Files.writeString(dir.resolve("facts.xml"), table);
183         return new Rows(dir).digest(Collections.singletonList("Q.foo"));
184     }
