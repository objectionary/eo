27  /**
28   * Test case for {@link Resolving}.
29   *
30   * @since 0.61.0
31   */
32  @ExtendWith(MktmpResolver.class)
33  final class ResolvingTest {
34  
35      @Test
36      void retriesFailedDependencyAndMarksEveryRelatedTojo(@Mktmp final Path tmp) throws Exception {
37          final Path xmir = tmp.resolve("library.xmir");
38          Files.writeString(
39              xmir,
40              String.join(
41                  "", "<object><metas><meta><head>rt</head><part>jvm</part>",
42                  "<part>org.example:library:1.0.0</part></meta></metas></object>"
43              )
44          );
45          try (TjsForeign tojos = new TjsForeign()) {
46              tojos.add("first").withXmir(xmir).withVersion("1.0.0");
47              tojos.add("second").withXmir(xmir).withVersion("1.0.0");
48              final AtomicInteger attempts = new AtomicInteger();
49              final Resolving resolving = new Resolving(
50                  tojos, tmp.resolve("target"),
51                  (dep, place) -> {
52                      if (attempts.incrementAndGet() == 1) {
53                          throw new IllegalStateException("The first download failed");
54                      }
55                      try {
56                          Files.createDirectories(place);
57                          Files.writeString(place.resolve("library.class"), "downloaded");
58                      } catch (final IOException err) {
59                          throw new UncheckedIOException(err);
60                      }
61                  },
62                  false, false, false, true, (Scalar<Dep>) () -> null, false
63              );
64              final List<Integer> observed = new ArrayList<>(5);
65              int failed = 0;
66              try {
67                  resolving.exec();
68              } catch (final UncheckedIOException err) {
69                  failed = 1;
70              }
71              observed.add(failed);
72              observed.add(tojos.dependencies().size());
73              resolving.exec();
74              observed.add(attempts.get());
75              observed.add(tojos.dependencies().size());
76              resolving.exec();
77              observed.add(attempts.get());
78              MatcherAssert.assertThat(
79                  "A failed download must keep both tojos pending until a successful retry",
80                  observed, Matchers.equalTo(List.of(1, 2, 2, 0, 2))
81              );
82          }
83      }
84  
85      @Test
86      void cleansUpSharedPlaceConcurrentlyWithoutFailing(@Mktmp final Path tmp) throws Exception {
87          final Resolving resolving = new Resolving(
88              null, tmp, (dep, place) -> { }, false, false, false, false,
89              (Scalar<Dep>) () -> null, false
90          );
91          final int threads = 8;
92          final List<Integer> slots = new ArrayList<>(threads);
93          for (int idx = 0; idx < threads; ++idx) {
94              slots.add(idx);
95          }
96          for (int trial = 0; trial < 20; ++trial) {
97              final Path stale = tmp.resolve("stale-version");
98              Files.createDirectories(stale.resolve("nested"));
99              Files.write(
100                 stale.resolve("nested").resolve("file.txt"),
101                 "stale".getBytes(StandardCharsets.UTF_8)
102             );
103             for (final Path place : new Threads<Path>(
104                 threads,
105                 new Mapped<Scalar<Path>>(
106                     idx -> () -> resolving.cleanPlace(
107                         tmp, "1.0.0", new HashSet<>(Collections.emptySet())
108                     ),
109                     slots
110                 )
111             )) {
112                 MatcherAssert.assertThat(
113                     "cleanPlace must resolve to the requested version directory, but it didnt",
114                     place,
115                     Matchers.equalTo(tmp.resolve("1.0.0"))
116                 );
117             }
118         }
119     }
