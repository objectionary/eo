37  @ExtendWith(MktmpResolver.class)
38  final class DealphaingTest {
39  
40      @Test
41      void renamesEveryArgumentOfApplication(@Mktmp final Path temp) throws IOException {
42          MatcherAssert.assertThat(
43              "every argument of the application must take the name of its void, but it doesnt",
44              new XMLDocument(DealphaingTest.placed(temp, false, "bar", "baz")),
45              XhtmlMatchers.hasXPaths("/object/o/o[@as='bar']", "/object/o/o[@as='baz']")
46          );
47      }
48  
49      @Test
50      void keepsEveryAlphaOfHalfNamedApplication(@Mktmp final Path temp) throws IOException {
51          MatcherAssert.assertThat(
52              "an application with one argument unnamed must keep every αN, but it doesnt",
53              new XMLDocument(DealphaingTest.placed(temp, false, "bar", "")),
54              XhtmlMatchers.hasXPaths("/object/o/o[@as='α0']", "/object/o/o[@as='α1']")
55          );
56      }
57  
58      @Test
59      void countsEveryLostArgumentOfApplication(@Mktmp final Path temp) throws IOException {
60          MatcherAssert.assertThat(
61              "the log must count every argument of the application that found no void, but it doesnt",
62              DealphaingTest.logged(temp),
63              Matchers.hasItem(
64                  "Named 0 of 2 positional argument(s) in 1 XMIR(s), 2 found no void to be named after"
65              )
66          );
67      }
68  
69      @Test
70      void failsOnEveryLostArgumentWhenAsked(@Mktmp final Path temp) {
71          MatcherAssert.assertThat(
72              "the failure must count every argument of the application, but it doesnt",
73              Assertions.assertThrows(
74                  IllegalStateException.class,
75                  () -> DealphaingTest.placed(temp, true, "bar", ""),
76                  "a half named application must stop the build when asked to, but it didnt"
77              ).getMessage(),
78              Matchers.containsString("2 positional argument(s)")
79          );
80      }
81  
82      @Test
83      void namesMessageOfBottom(@Mktmp final Path temp) throws IOException {
84          MatcherAssert.assertThat(
85              "the argument of ⊥ must be named 'x' without failing the build, but it isnt",
86              new XMLDocument(
87                  DealphaingTest.dealphaed(temp, true, DealphaingTest.bottom(), Map.of())
88              ),
89              XhtmlMatchers.hasXPath("/object/o[@base='⊥']/o[@as='x']")
90          );
91      }
92  
93      @Test
94      void namesArgumentsInsideBottomMessage(@Mktmp final Path temp) throws IOException {
95          MatcherAssert.assertThat(
96              "the arguments of a call inside the message of ⊥ must be named, but they arent",
97              new XMLDocument(
98                  DealphaingTest.dealphaed(
99                      temp,
100                     true,
101                     "<object><o base='⊥' loc='Φ.k.φ'><o base='.printf' loc='Φ.k.φ.α0' as='α0'><o/><o loc='Φ.k.φ.α0.α1' as='α1'/></o></o></object>",
102                     Map.of("Φ.k.φ.α0.α1", "args")
103                 )
104             ),
105             XhtmlMatchers.hasXPaths(
106                 "/object/o[@base='⊥']/o[@as='x']",
107                 "/object/o/o[@base='.printf']/o[@as='args']"
108             )
109         );
110     }

127     private static String bottom() {
128         return "<object><o base='⊥' loc='Φ.k.φ'><o loc='Φ.k.φ.α0' as='α0'/></o></object>";
129     }

131     private static List<String> logged(final Path temp) {
132         return DealphaingTest.logged(() -> DealphaingTest.placed(temp, false, "bar", ""));
133     }
134 
135     private static List<String> logged(final Scalar<Path> run) {
136         final List<String> messages = new CopyOnWriteArrayList<>();
137         final Appender appender = new AppenderSkeleton() {
138             @Override
139             protected void append(final LoggingEvent event) {
140                 messages.add(String.valueOf(event.getRenderedMessage()));
141             }
142 
143             @Override
144             public void close() {
145                 // Nothing to release.
146             }
147 
148             @Override
149             public boolean requiresLayout() {
150                 return false;
151             }
152         };
153         final Logger logger = Logger.getLogger(Dealphaing.class);
154         final Level level = logger.getLevel();
155         logger.setLevel(Level.INFO);
156         logger.addAppender(appender);
157         try {
158             new Unchecked<>(run).value();
159         } finally {
160             logger.removeAppender(appender);
161             logger.setLevel(level);
162         }
163         return messages;
164     }
165 
166     private static Path placed(
167         final Path temp, final boolean strict, final String... lands
168     ) throws IOException {
169         final String call = "Φ.k.φ";
170         final Map<String, String> names = new HashMap<>(lands.length);
171         for (int idx = 0; idx < lands.length; ++idx) {
172             if (!lands[idx].isEmpty()) {
173                 names.put(String.format("%s.α%d", call, idx), lands[idx]);
174             }
175         }
176         return DealphaingTest.dealphaed(
177             temp,
178             strict,
179             String.format(
180                 "<object><o loc='%1$s'><o loc='%1$s.α0' as='α0'/><o loc='%1$s.α1' as='α1'/></o></object>",
181                 call
182             ),
183             names
184         );
185     }
186 
187     private static Path dealphaed(
188         final Path temp, final boolean strict, final String program,
189         final Map<String, String> names
190     ) throws IOException {
191         final Path xmir = temp.resolve("app.xmir");
192         Files.write(xmir, program.getBytes(StandardCharsets.UTF_8));
193         try (TjsForeign tojos = new TjsForeign()) {
194             final TjForeign tojo = tojos.add("app").withXmir(xmir);
195             new Dealphaing(List.of(tojo), names, temp.resolve("target"), strict).exec();
196             return tojo.xmir();
197         }
198     }
