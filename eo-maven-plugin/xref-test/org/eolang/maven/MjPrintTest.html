51      @Test
52      void declaresAParameterForEveryPenaltyWeight() {
53          final Set<String> declared = new MojoFields().all();
54          Assumptions.assumeFalse(declared.isEmpty());
55          MatcherAssert.assertThat(
56              "a weight a printer pack can vary must be a parameter a build can vary too",
57              Stream.of(PenaltyKey.values())
58                  .map(key -> MjPrintTest.param(key.name()))
59                  .filter(name -> name.isEmpty() || !declared.contains(name))
60                  .collect(Collectors.toList()),
61              Matchers.empty()
62          );
63      }

65      @Test
66      void printsSuccessfully(@Mktmp final Path temp) throws Exception {
67          final Path resources = new File(
68              "../eo-printer/src/test/resources/org/eolang/printer/print-packs/xmir"
69          ).toPath();
70          final Collection<Path> walk = new WkDefault(resources);
71          Assumptions.assumeTrue(!walk.isEmpty());
72          for (final Path source : walk) {
73              new Saved(new TextOf(source), temp.resolve(source)).value();
74          }
75          final Path output = temp.resolve("output");
76          new FakeMaven(temp)
77              .with("sources", temp.resolve(resources).toFile())
78              .with("output", output.toFile())
79              .execute(new PpPrint())
80              .result();
81          for (final Path source : walk) {
82              final String src = resources.relativize(source).toString()
83                  .replace(".xmir", ".eo");
84              MatcherAssert.assertThat(
85                  String.format(
86                      "File with name %s should have existed in output directory, but it didn't",
87                      src
88                  ),
89                  Files.exists(output.resolve(Paths.get(src))),
90                  Matchers.is(true)
91              );
92          }
93      }

95      @Test
96      void doesNotMangleADirectoryNameContainingTheXmirSubstring(@Mktmp final Path temp)

98          final Path source = temp.resolve("xmir/v1.xmir-legacy/main.xmir");
99          Files.createDirectories(source.getParent());

104                         System.lineSeparator(),
105                         "+package foo",
106                         "",
107                         "[] > main"
108                     )
109                 )
110             ).parsed().toString(),
111             source
112         ).value();
113         final Path output = temp.resolve("eo");
114         new FakeMaven(temp)
115             .with("sources", temp.resolve("xmir").toFile())
116             .with("output", output.toFile())
117             .execute(new PpPrint())
118             .result();

120             "only the trailing .xmir extension should be replaced, the .xmir substring in the directory name must survive untouched",
121             Files.exists(output.resolve("v1.xmir-legacy/main.eo")),

157     @ParameterizedTest
158     @ClasspathSource(value = "org/eolang/maven/print-packs", glob = "**.yaml")
159     void printsXmirToEo(final String pack) throws Exception {
160         final Xtory xtory = new XtSticky(new XtYaml(pack));
161         Assumptions.assumeTrue(xtory.map().get("skip") == null);
162         MatcherAssert.assertThat(
163             "PrintMojo should print EO in straight notation, but it didn't",
164             MjPrintTest.printed(xtory, this.dir).asString(),
165             Matchers.equalTo(this.expected(xtory))
166         );
167     }
168 
169     private String expected(final Xtory xtory) {
170         final String origin = (String) xtory.map().get("origin");
171         final String expected;
172         if (xtory.map().containsKey("printed")) {
173             expected = (String) xtory.map().get("printed");
174             MatcherAssert.assertThat(
175                 "The 'printed' section repeats 'origin' verbatim and must be deleted from the pack, since a pack without 'printed' already expects the printer to reproduce its 'origin'",
176                 expected,
177                 Matchers.not(Matchers.equalTo(origin))
178             );
179         } else {
180             expected = origin;
181         }
182         return expected;
183     }
184 
185     private static Text printed(final Xtory xtory, final Path temp)
186         throws Exception {
187         new Saved(
188             new EoSyntax(
189                 new InputOf(xtory.map().get("origin").toString())
190             ).parsed().toString(),
191             temp.resolve("xmir/foo/x/main.xmir")
192         ).value();
193         final FakeMaven maven = new FakeMaven(temp)
194             .with("sources", temp.resolve("xmir").toFile())
195             .with("output", temp.resolve("eo").toFile());
196         final Object pins = xtory.map().get("penalties");
197         if (pins != null) {
198             for (final Map.Entry<?, ?> pin : ((Map<?, ?>) pins).entrySet()) {
199                 final String param = MjPrintTest.param(pin.getKey().toString());
200                 if (!param.isEmpty()) {
201                     maven.with(param, ((Number) pin.getValue()).intValue());
202                 }
203             }
204         }
205         return new TextOf(
206             maven.execute(MjPrint.class).result().get("eo/foo/x/main.eo")
207         );
208     }

210     private static String param(final String key) {
211         return new MapOf<>(
212             new MapEntry<>("INDENT", "indent"),
213             new MapEntry<>("BRACKET", "bracket"),
214             new MapEntry<>("LEADING", "leading"),
215             new MapEntry<>("PHI", "phi"),
216             new MapEntry<>("IF", "conditional"),
217             new MapEntry<>("EXCESS", "excess"),
218             new MapEntry<>("SYMBOL", "symbol"),
219             new MapEntry<>("SPACE", "space"),
220             new MapEntry<>("WIDTH", "width"),
221             new MapEntry<>("STEP", "step")
222         ).getOrDefault(key, "");
223     }
