5   package org.eolang.maven;
6   
7   import com.jcabi.log.Logger;
8   import java.io.IOException;
9   import java.nio.file.Files;
10  import java.nio.file.Path;
11  import java.util.ArrayList;
12  import java.util.Collection;
13  import java.util.stream.Collectors;
14  import java.util.stream.Stream;
15  
16  /**
17   * How much a directory of tables holds, in words a log line can carry.
18   *
19   * <p>A path on its own says where to look and nothing about what is there.
20   * Whoever reads the log wants to know whether the tables were written at all
21   * and how big they came out, since a table that shrank by half between two
22   * builds is the first sign that a rule stopped seeing something.</p>
23   *
24   * @since 0.71.0
25   */
26  final class Tabled {
27  
28      /**
29       * The directory the tables are in.
30       */
31      private final Path dir;
32  
33      /**
34       * Ctor.
35       *
36       * @param tables The directory the tables are in
37       */
38      Tabled(final Path tables) {
39          this.dir = tables;
40      }
41  
42      /**
43       * How many tables there are and how much they weigh together.
44       *
45       * @return The description, for a log line
46       * @throws IOException If the directory cannot be read
47       */
48      String asString() throws IOException {
49          final Collection<Path> files = new ArrayList<>(0);
50          if (Files.exists(this.dir)) {
51              try (Stream<Path> all = Files.list(this.dir)) {
52                  files.addAll(all.filter(Files::isRegularFile).collect(Collectors.toList()));
53              }
54          }
55          long bytes = 0L;
56          for (final Path file : files) {
57              bytes = bytes + Files.size(file);
58          }
59          return Logger.format("%d table(s), %[size]s total", files.size(), bytes);
60      }
61  }
