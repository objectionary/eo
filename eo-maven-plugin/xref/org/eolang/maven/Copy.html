18  /**
19   * What a place copies: a formation, and where the objects in the voids of
20   * the copy come from, when the code alone says that.
21   *
22   * <p>Java makes the copy an atom returns, and the type the atom says it
23   * returns, such as {@code Φ.posix.status-return(Φ.number,Φ.string)}, gives
24   * in brackets what is in each void. A call written at one place of the
25   * program makes a copy whose voids hold the arguments written there, read in
26   * the copy the call is written in. A copy of anything else has voids nobody
27   * here can see into.</p>
28   *
29   * <p>A copy keeps only the type of the copy its call is written in. When that
30   * one was made by a call too, its type is just its formation, so an argument
31   * that reads one of its voids finds nothing. The program has finitely many
32   * such types, so a place reads into finitely many copies and every read
33   * ends.</p>
34   *
35   * <p>A copy may also be blurred: its formation stands for several formations,
36   * such as the two the program puts into the {@code fork} of {@code bool}. It
37   * keeps all those formations, its candidates, since each of them binds its
38   * attributes in its own way: a name read off it is looked up in every
39   * candidate, made by the same call, and copies a formation only when what
40   * they give agrees (#9973). The candidates do not have to have the same
41   * voids. When they do, the copy is uniform, and it tells where the arguments
42   * of a call on it land. When they do not, as with {@code Φ.number} and
43   * {@code Φ.number.plus} in a void filled by both, the copy may be read but
44   * not called, since an argument at one position lands in different voids of
45   * different candidates (#9978).</p>
46   *
47   * <p>A terminator makes a copy of no formation at all, as
48   * {@link #terminator()} says. A name read off it, or a call on it, is a
49   * terminator too, since the program never gets that far (#9982).</p>
50   *
51   * <p>A copy may also end: it is of its formations, or else it is the
52   * terminator, as a void is that one call fills with {@code T} and another
53   * with {@code [message]} (#9997). A name read off it is looked up in its
54   * formations, since the terminator never gets that far. A call on it, though,
55   * may land in the terminator, which takes its message by position only, so
56   * the code does not fix where its arguments land, and the copy a call on it
57   * makes ends too.</p>
58   *
59   * @since 0.76.0
60   */
61  final class Copy {
62  
63      /**
64       * The type: the formation, and the types an atom gives its voids in
65       * brackets.
66       */
67      private final Annotation type;
68  
69      /**
70       * The call that made the copy, when the code shows one: its place, and
71       * the type of the copy it is written in.
72       */
73      private final Optional<Map.Entry<String, Annotation>> call;
74  
75      /**
76       * The formations the copy may be of, in their natural order: just one
77       * when the copy is sharp, and several when it is blurred.
78       */
79      private final Set<String> forms;
80  
81      /**
82       * Whether it may be the terminator instead of any of its formations.
83       */
84      private final boolean end;
85  
86      /**
87       * Ctor.
88       *
89       * @param kind The type: the formation, and the types an atom gives its
90       *  voids in brackets
91       */
92      Copy(final Annotation kind) {
93          this(kind, Optional.empty());
94      }
95  
96      /**
97       * Ctor.
98       *
99       * @param candidates The formations the copy may be of: one for a sharp
100      *  copy, several for a blurred one, which may have different voids
101      */
102     Copy(final Collection<String> candidates) {
103         this(
104             new Annotation(new TreeSet<>(candidates).first()), Optional.empty(),
105             new TreeSet<>(candidates), false
106         );
107     }
108 
109     /**
110      * Ctor.
111      *
112      * @param kind The type: the formation, and the types an atom gives its
113      *  voids in brackets
114      * @param maker The call that made the copy, when the code shows one
115      */
116     Copy(final Annotation kind, final Optional<Map.Entry<String, Annotation>> maker) {
117         this(kind, maker, new TreeSet<>(Set.of(kind.head())), false);
118     }
119 
120     /**
121      * Ctor.
122      *
123      * @param kind The type: the formation, and the types an atom gives its
124      *  voids in brackets
125      * @param maker The call that made the copy, when the code shows one
126      * @param candidates The formations the copy may be of, which nobody
127      *  changes afterwards
128      * @param ends Whether it may be the terminator instead
129      */
130     private Copy(
131         final Annotation kind, final Optional<Map.Entry<String, Annotation>> maker,
132         final SortedSet<String> candidates, final boolean ends
133     ) {
134         this.type = kind;
135         this.call = maker;
136         this.forms = candidates;
137         this.end = ends;
138     }
139 
140     @Override
141     public boolean equals(final Object other) {
142         final boolean result;
143         if (this == other) {
144             result = true;
145         } else if (other == null || this.getClass() != other.getClass()) {
146             result = false;
147         } else {
148             result = this.type.equals(((Copy) other).type)
149                 && this.call.equals(((Copy) other).call)
150                 && this.forms.equals(((Copy) other).forms)
151                 && this.end == ((Copy) other).end;
152         }
153         return result;
154     }
155 
156     @Override
157     public int hashCode() {
158         return Objects.hash(this.type, this.call, this.forms, this.end);
159     }

161     /**
162      * The copy a terminator makes: it is of no formation at all, since the
163      * program stops there and nothing is ever read off it (#9982).
164      *
165      * @return The copy
166      */
167     static Copy terminator() {
168         return new Copy(new Annotation("⊥"), Optional.empty(), new TreeSet<>(), false);
169     }
170 
171     /**
172      * The copy a lookup gives when it is asked for again while it is still
173      * running: it stands for whatever that lookup finds, so a side that is
174      * this copy adds nothing to what the lookup finds (#9983).
175      *
176      * @return The copy
177      */
178     static Copy looped() {
179         return new Copy(new Annotation("↺"), Optional.empty(), new TreeSet<>(), false);
180     }
181 
182     /**
183      * Whether it is what a terminator makes, so that the program never gets
184      * past it.
185      *
186      * @return TRUE if it is
187      */
188     boolean stops() {
189         return this.forms.isEmpty() && !this.loops();
190     }
191 
192     /**
193      * Whether it stands for a lookup that is still running, as
194      * {@link #looped()} says.
195      *
196      * @return TRUE if it does
197      */
198     boolean loops() {
199         return this.forms.isEmpty() && "↺".equals(this.type.head());
200     }
201 
202     /**
203      * Whether it may be the terminator instead of any of its formations, so
204      * that a call on it fixes nothing about where its arguments land.
205      *
206      * @return TRUE if it may
207      */
208     boolean ends() {
209         return this.end;
210     }
211 
212     /**
213      * The same copy, which may be the terminator instead, or not.
214      *
215      * @param ends Whether it may be the terminator instead
216      * @return The copy
217      */
218     Copy ending(final boolean ends) {
219         return new Copy(this.type, this.call, new TreeSet<>(this.forms), ends);
220     }
221 
222     /**
223      * Whether its formation is the only one it may be of, so that its
224      * attributes may be read.
225      *
226      * @return TRUE if it is, FALSE if it is blurred
227      */
228     boolean sole() {
229         return this.forms.size() == 1;
230     }
231 
232     /**
233      * Whether all the formations it may be of have the same voids in the
234      * same order, so that the arguments of a call on it land in the same
235      * voids whichever formation it is. A sharp copy always is.
236      *
237      * @param calls What the calls put into the voids
238      * @return TRUE if they have
239      */
240     boolean uniform(final Calls calls) {
241         return this.forms.stream().map(calls::voids).distinct().count() == 1L;
242     }
243 
244     /**
245      * The formation it copies.
246      *
247      * @return The locator of the formation
248      */
249     String formation() {
250         return this.type.head();
251     }
252 
253     /**
254      * The sharp copies it may be, one for each formation it may be of, all
255      * made by the call that made it.
256      *
257      * @return The copies, in the order of their formations
258      */
259     Collection<Copy> candidates() {
260         return this.forms.stream().map(
261             form -> new Copy(
262                 new Annotation(form), this.call, new TreeSet<>(Set.of(form)), false
263             )
264         ).collect(Collectors.toList());
265     }
266 
267     /**
268      * The copy that a call, written at a place in another copy, makes of
269      * this one.
270      *
271      * @param place The place of the call
272      * @param outer The copy the call is written in
273      * @return The copy the call makes
274      */
275     Copy applied(final String place, final Copy outer) {
276         return new Copy(
277             new Annotation(this.formation()), Optional.of(Map.entry(place, outer.type)),
278             new TreeSet<>(this.forms), this.end
279         );
280     }

282     /**
283      * The type that Java puts into a void, as the brackets of the type say.
284      *
285      * @param name The name of the void
286      * @param calls What the calls put into the voids
287      * @return The type, or nothing when the type has no brackets for the void
288      */
289     Optional<Annotation> member(final String name, final Calls calls) {
290         return calls.member(this.type, name);
291     }
292 
293     /**
294      * The argument that the call which made the copy puts into a void: its
295      * place, and the copy that place is read in.
296      *
297      * @param name The name of the void
298      * @param calls What the calls put into the voids
299      * @return The place and the copy, or nothing when no call in the program
300      *  made the copy or the call does not show the argument
301      */
302     Optional<Map.Entry<String, Copy>> argument(final String name, final Calls calls) {
303         return this.call.flatMap(
304             made -> calls.argument(made.getKey(), this.formation(), name).map(
305                 arg -> Map.entry(arg, new Copy(made.getValue()))
306             )
307         );
308     }
309 
310     /**
311      * The copy that all of the given ones agree on.
312      *
313      * <p>A generic atom returns what one of several voids holds, and the code
314      * does not say which. A name read off a blurred copy is what one of its
315      * candidates gives, and the code does not say which either. So the
316      * result may be of any formation that any of the given copies may be of.
317      * When they are the same copy, it is that copy. When all those formations
318      * have the same voids in the same order, it is a copy of them, sharp when
319      * there is one and blurred when there are several, since the arguments of
320      * a call on it land in the same voids whichever it is. It is made by the
321      * call that made all the given copies, when there is one, and by no call
322      * in particular otherwise. When one of them is unknown, or the formations
323      * have different voids, nothing is said: {@code tuple.eq} and
324      * {@code bytes.eq} name their voids differently, so the argument of a
325      * call on one of the two lands nowhere in particular (#9973).</p>
326      *
327      * <p>A copy a terminator makes is left out, since the program stops
328      * there and never reads anything off it: {@code x.if T 42} gives what
329      * {@code 42} gives, and so does {@code recovered T 42}. When all of them
330      * are terminators, so is the result (#9982).</p>
331      *
332      * <p>A copy that stands for a lookup still running, as {@link #looped()}
333      * says, is left out too: it is the very lookup that asks, so it gives
334      * only what that lookup finds, and the lookup has only to agree with the
335      * others. When every one of them is such a copy, so is the result
336      * (#9983).</p>
337      *
338      * @param held What each void holds
339      * @param calls What the calls put into the voids
340      * @return The copy they agree on, or nothing
341      */
342     static Optional<Copy> agreed(final Collection<Optional<Copy>> held, final Calls calls) {
343         final List<Optional<Copy>> live = held.stream().filter(
344             each -> each.map(copy -> !copy.stops() && !copy.loops()).orElse(true)
345         ).collect(Collectors.toList());
346         final Optional<Copy> found;
347         if (!held.isEmpty() && live.isEmpty()
348             && held.stream().flatMap(Optional::stream).allMatch(Copy::loops)) {
349             found = Optional.of(Copy.looped());
350         } else if (!held.isEmpty() && live.isEmpty()) {
351             found = Optional.of(Copy.terminator());
352         } else {
353             found = Copy.joined(live, calls);
354         }
355         return found;
356     }
357 
358     private static Optional<Copy> joined(
359         final Collection<Optional<Copy>> held, final Calls calls
360     ) {
361         final Set<Copy> known = held.stream().flatMap(Optional::stream).collect(
362             Collectors.toSet()
363         );
364         final SortedSet<String> forms = known.stream().flatMap(copy -> copy.forms.stream())
365             .collect(Collectors.toCollection(TreeSet::new));
366         final Set<Optional<Map.Entry<String, Annotation>>> makers = known.stream()
367             .map(copy -> copy.call).collect(Collectors.toSet());
368         final Optional<Copy> found;
369         if (held.isEmpty() || held.contains(Optional.empty())
370             || forms.stream().map(calls::voids).distinct().count() > 1L) {
371             found = Optional.empty();
372         } else if (known.size() == 1) {
373             found = known.stream().findFirst().map(copy -> copy.ending(false));
374         } else if (makers.size() == 1) {
375             found = Optional.of(
376                 new Copy(new Annotation(forms.first()), makers.iterator().next(), forms, false)
377             );
378         } else {
379             found = Optional.of(new Copy(forms));
380         }
381         return found;
382     }
