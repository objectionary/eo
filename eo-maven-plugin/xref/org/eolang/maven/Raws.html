11  import java.nio.file.Path;
12  import java.util.function.UnaryOperator;
13  import javax.xml.transform.OutputKeys;
14  import javax.xml.transform.Transformer;
15  import javax.xml.transform.TransformerFactory;
16  import javax.xml.transform.dom.DOMSource;
17  import javax.xml.transform.stream.StreamResult;
18  import org.cactoos.io.InputOf;
19  import org.eolang.cache.GlobalCache;
20  import org.eolang.parser.EoSyntax;
21  import org.eolang.parser.Utf8;

23  /**
24   * The tree the EO parser makes of a source, before any XSL touches it.
25   *
26   * <p>Two readers of one {@code .eo} file put different XSL trains on top of
27   * the grammar, so what they share is the raw tree and nothing else. It is
28   * kept in the one place the rest of the cache gives that source, beside
29   * the hash of the text it was made of, so a rewritten source replaces the
30   * tree of its old text instead of adding one more next to it.</p>
31   *
32   * <p>The tree is written with no indentation of its own, because an
33   * indented tree is not the same tree: the newlines a pretty printer puts
34   * between elements come back as text nodes, and the XSL train the reader
35   * puts on top then sees children the parser never made.</p>
36   *
37   * @since 0.62.0
38   */
39  final class Raws {
40  
41      /**
42       * Where the results of earlier builds are looked for and kept.
43       */
44      private final GlobalCache cache;
45  
46      /**
47       * The directory the trees are written into.
48       */
49      private final Path base;
50  
51      /**
52       * Ctor.
53       *
54       * @param store Where the results of earlier builds are looked for and kept
55       * @param dir The directory the trees are written into
56       */
57      Raws(final GlobalCache store, final Path dir) {
58          this.cache = store;
59          this.base = dir;
60      }
61  
62      /**
63       * The raw tree of one source.
64       *
65       * @param tojo The source to read
66       * @return The tree the parser makes of it
67       * @throws IOException If fails to read or write
68       */
69      XML of(final TjForeign tojo) throws IOException {
70          final Path target = new Place(tojo.identifier()).make(this.base, MjAssemble.XMIR);
71          this.cache.footprint(
72              this.base.relativize(target),
73              new TojoHash(tojo),
74              src -> {
75                  final Transformer writer = TransformerFactory.newInstance().newTransformer();
76                  writer.setOutputProperty(OutputKeys.INDENT, "no");
77                  final StringWriter text = new StringWriter();
78                  writer.transform(
79                      new DOMSource(
80                          new EoSyntax(
81                              new InputOf(new Utf8(new InputOf(src), src.toString()).asString()),
82                              UnaryOperator.identity()
83                          ).parsed().inner()
84                      ),
85                      new StreamResult(text)
86                  );
87                  return text.toString();
88              }
89          ).apply(tojo.source(), target);
90          return new XMLDocument(target);
91      }
92  }
