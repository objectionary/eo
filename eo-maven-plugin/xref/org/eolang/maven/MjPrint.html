5   package org.eolang.maven;
6   
7   import com.jcabi.log.Logger;
8   import com.jcabi.xml.XMLDocument;
9   import java.io.File;
10  import java.io.IOException;
11  import java.nio.file.Path;
12  import java.nio.file.Paths;
13  import java.util.Set;
14  import java.util.regex.Pattern;
15  import org.apache.maven.plugins.annotations.LifecyclePhase;
16  import org.apache.maven.plugins.annotations.Mojo;
17  import org.apache.maven.plugins.annotations.Parameter;
18  import org.cactoos.text.TextOf;
19  import org.eolang.cache.Saved;
20  import org.eolang.printer.Xmir;
21  
22  /**
23   * Print XMIR to EO.
24   *
25   * <p>This goal goes through all XMIR sources found in the specified directory,
26   * converts them back to EO format, and saves the resulting EO files
27   * in the specified output directory, preserving the original directory structure.
28   * Input XMIR files are found in {@link #sources},
29   * output EO files are saved in {@link #output}.</p>
30   *
31   * @since 0.33.0
32   */
33  @Mojo(
34      name = "print",
35      defaultPhase = LifecyclePhase.PROCESS_SOURCES,
36      threadSafe = true
37  )
38  public final class MjPrint extends MjPenalties {
39  
40      /**
41       * Pattern to catch the trailing .xmir extension.
42       */
43      private static final Pattern XMIR = Pattern.compile("\\.xmir$");
44  
45      /**
46       * Directory with XMIR sources to print.
47       */
48      @Parameter(
49          alias = "print.sources",
50          property = "eo.print.sources",
51          required = true,
52          defaultValue = "${project.basedir}/src/main/xmir"
53      )
54      private File sources;
55  
56      /**
57       * Directory where printed EO files are placed.
58       */
59      @Parameter(
60          alias = "print.output",
61          property = "eo.print.output",
62          required = true,
63          defaultValue = "${project.build.directory}/generated-sources/eo"
64      )
65      private File output;
66  
67      /**
68       * Ctor.
69       */
70      public MjPrint() {
71          // nothing
72      }
73  
74      @Override
75      void exec() throws IOException {
76          final int total = new Threaded<>(
77              new WkDefault(this.sources.toPath()).includes(Set.of("**.xmir")),
78              this::print
79          ).total();
80          if (total == 0) {
81              Logger.info(this, "No XMIR sources found");
82          } else {
83              Logger.info(this, "Printed %d XMIR sources into EO", total);
84          }
85      }
86  
87      private int print(final Path source) throws Exception {
88          final Path home = this.output.toPath();
89          final Path relative = Paths.get(
90              MjPrint.XMIR.matcher(
91                  this.sources.toPath().relativize(source).toString()
92              ).replaceFirst(".eo")
93          );
94          new Saved(
95              new Xmir(
96                  new XMLDocument(new TextOf(source).asString()), this.weights()
97              ).toEO(),
98              home.resolve(relative)
99          ).value();
100         Logger.info(
101             this,
102             "Printed: %[file]s (%[size]s) => %[file]s (%[size]s)",
103             source,
104             source.toFile().length(),
105             home.resolve(relative),
106             home.resolve(relative).toFile().length()
107         );
108         return 1;
109     }
110 }
