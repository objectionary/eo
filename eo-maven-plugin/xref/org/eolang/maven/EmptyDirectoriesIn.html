5   package org.eolang.maven;
6   
7   import com.jcabi.log.Logger;
8   import java.io.File;
9   import java.nio.file.Files;
10  import java.nio.file.Path;
11  
12  /**
13   * Delete empty directories in provided root.
14   *
15   * <p>A symbolic link is a leaf here, whatever stands at the end of it. EO
16   * may remove a directory it made and then emptied, but the tree behind a
17   * link is somebody else's: descending through one would delete directories
18   * outside the build output that EO never created. A link is also something
19   * a directory holds, so a directory holding one is not empty.</p>
20   *
21   * <p>This holds for the root as well as for anything under it: when the
22   * root given is a link, the walk stops there and the tree it points at is
23   * left alone.</p>
24   *
25   * @since 0.55
26   */
27  final class EmptyDirectoriesIn {
28  
29      /**
30       * Root path.
31       */
32      private final File root;
33  
34      /**
35       * Ctor.
36       *
37       * @param root Root directory
38       */
39      EmptyDirectoriesIn(final Path root) {
40          this(root.toFile());
41      }
42  
43      /**
44       * Ctor.
45       *
46       * @param root Root directory
47       */
48      EmptyDirectoriesIn(final File root) {
49          this.root = root;
50      }
51  
52      /**
53       * Clear empty directories in {@code this.root}.
54       */
55      void clear() {
56          if (!this.root.isDirectory()) {
57              throw new IllegalStateException(
58                  Logger.format("Provided path %[file]s is not a directory", this.root)
59              );
60          }
61          this.delete(this.root);
62      }
63  
64      private void delete(final File dir) {
65          if (!dir.isDirectory() || Files.isSymbolicLink(dir.toPath())) {
66              return;
67          }
68          final File[] before = dir.listFiles();
69          if (before != null) {
70              for (final File file : before) {
71                  this.delete(file);
72              }
73          }
74          final File[] after = dir.listFiles();
75          if (after != null && after.length == 0 && !dir.equals(this.root) && dir.delete()) {
76              Logger.debug(EmptyDirectoriesIn.class, "Deleted empty directory %[file]s", dir);
77          }
78      }
79  }
