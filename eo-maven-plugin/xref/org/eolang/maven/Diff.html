5   package org.eolang.maven;
6   
7   import java.util.ArrayList;
8   import java.util.Arrays;
9   import java.util.List;
10  import java.util.regex.Pattern;
11  import java.util.stream.Collectors;
12  
13  /**
14   * A colored unified diff between two multi-line texts.
15   *
16   * <p>Both sides are split into lines and aligned by their longest
17   * common subsequence. The result is rendered in the familiar unified
18   * style: deletions are prefixed with {@code -} highlighted in red,
19   * additions with {@code +} highlighted in green, and each change is shown
20   * with a few unchanged lines around it, under an {@code @@} header that
21   * says where in either text the change sits. A line further away than
22   * that is not printed at all, so one wrong line in a long file prints
23   * nine lines and not the whole file. When the two texts are identical,
24   * {@link #colored()} returns an empty string and {@link #same()} returns
25   * {@code true}.</p>
26   *
27   * <p>Line terminators are part of what is compared, since two texts that
28   * differ only in them are not the same text and a diff that shows nothing
29   * changed would say the opposite. A carriage return is rendered as
30   * {@code \r}, and a text that does not end with a newline carries the
31   * note {@code git} prints for it.</p>
32   *
33   * @since 0.57.0
34   */
35  final class Diff {
36  
37      /**
38       * The line feed every line ends with.
39       */
40      private static final Pattern FEED = Pattern.compile("\\n");
41  
42      /**
43       * The carriage return a line may end with.
44       */
45      private static final Pattern RETURN = Pattern.compile("\\r");
46  
47      /**
48       * ANSI escape that resets all coloring.
49       */
50      private static final String RESET = "\033[0m";
51  
52      /**
53       * How many unchanged lines are shown on either side of a change.
54       */
55      private static final int CONTEXT = 3;
56  
57      /**
58       * The text before the change.
59       */
60      private final String before;
61  
62      /**
63       * The text after the change.
64       */
65      private final String after;
66  
67      /**
68       * Ctor.
69       *
70       * @param before The text before the change
71       * @param after The text after the change
72       */
73      Diff(final String before, final String after) {
74          this.before = before;
75          this.after = after;
76      }
77  
78      /**
79       * Are the two texts identical?
80       *
81       * @return TRUE if there is nothing to show
82       */
83      boolean same() {
84          return this.before.equals(this.after);
85      }
86  
87      /**
88       * Render the difference as a colored unified diff.
89       *
90       * @return The diff, or an empty string if the texts are identical
91       */
92      String colored() {
93          final String result;
94          if (this.same()) {
95              result = "";
96          } else {
97              result = Diff.render(Diff.lines(this.before), Diff.lines(this.after));
98          }
99          return result;
100     }
101 
102     private static List<String> lines(final String text) {
103         final List<String> out = new ArrayList<>(
104             Arrays.stream(Diff.FEED.split(text, -1))
105                 .map(line -> Diff.RETURN.matcher(line).replaceAll("\\\\r"))
106                 .collect(Collectors.toList())
107         );
108         if (out.get(out.size() - 1).isEmpty()) {
109             out.remove(out.size() - 1);
110         } else {
111             out.add("\\ No newline at end of file");
112         }
113         return out;
114     }
115 
116     private static String render(final List<String> before, final List<String> after) {
117         final List<String> script = Diff.script(before, after);
118         final StringBuilder out = new StringBuilder(0);
119         int idx = 0;
120         while (idx < script.size()) {
121             if (' ' == script.get(idx).charAt(0)) {
122                 idx += 1;
123             } else {
124                 final int last = Diff.closing(script, idx);
125                 final int till = Math.min(script.size(), last + 1 + Diff.CONTEXT);
126                 Diff.hunk(out, script, Math.max(0, idx - Diff.CONTEXT), till);
127                 idx = till;
128             }
129         }
130         return out.toString();
131     }
132 
133     private static List<String> script(final List<String> before, final List<String> after) {
134         final int[][] lcs = Diff.lcs(before, after);
135         final List<String> out = new ArrayList<>(0);
136         int row = 0;
137         int col = 0;
138         while (row < before.size() && col < after.size()) {
139             if (before.get(row).equals(after.get(col))) {
140                 out.add(String.join("", " ", before.get(row)));
141                 row += 1;
142                 col += 1;
143             } else if (lcs[row + 1][col] >= lcs[row][col + 1]) {
144                 out.add(String.join("", "-", before.get(row)));
145                 row += 1;
146             } else {
147                 out.add(String.join("", "+", after.get(col)));
148                 col += 1;
149             }
150         }
151         Diff.drain(out, before.subList(row, before.size()), "-");
152         Diff.drain(out, after.subList(col, after.size()), "+");
153         return out;
154     }
155 
156     private static int closing(final List<String> script, final int start) {
157         int last = start;
158         int idx = start;
159         while (idx < script.size() && idx - last <= 2 * Diff.CONTEXT) {
160             if (' ' != script.get(idx).charAt(0)) {
161                 last = idx;
162             }
163             idx += 1;
164         }
165         return last;
166     }
167 
168     private static void hunk(
169         final StringBuilder out, final List<String> script, final int from, final int till
170     ) {
171         out.append(
172             String.format(
173                 "@@ -%d,%d +%d,%d @@",
174                 Diff.counted(script, 0, from, '+') + 1,
175                 Diff.counted(script, from, till, '+'),
176                 Diff.counted(script, 0, from, '-') + 1,
177                 Diff.counted(script, from, till, '-')
178             )
179         ).append('\n');
180         for (final String line : script.subList(from, till)) {
181             if ('-' == line.charAt(0)) {
182                 Diff.deleted(out, line.substring(1));
183             } else if ('+' == line.charAt(0)) {
184                 Diff.added(out, line.substring(1));
185             } else {
186                 Diff.common(out, line.substring(1));
187             }
188         }
189     }
190 
191     private static int counted(
192         final List<String> script, final int from, final int till, final char skip
193     ) {
194         int count = 0;
195         for (final String line : script.subList(from, till)) {
196             if (skip != line.charAt(0)) {
197                 count += 1;
198             }
199         }
200         return count;
201     }
202 
203     private static int[][] lcs(final List<String> before, final List<String> after) {
204         final int rows = before.size();
205         final int cols = after.size();
206         final int[][] table = new int[rows + 1][cols + 1];
207         for (int row = rows - 1; row >= 0; row -= 1) {
208             for (int col = cols - 1; col >= 0; col -= 1) {
209                 if (before.get(row).equals(after.get(col))) {
210                     table[row][col] = table[row + 1][col + 1] + 1;
211                 } else {
212                     table[row][col] = Math.max(table[row + 1][col], table[row][col + 1]);
213                 }
214             }
215         }
216         return table;
217     }
218 
219     private static void drain(
220         final List<String> out, final List<String> tail, final String mark
221     ) {
222         for (final String line : tail) {
223             out.add(String.join("", mark, line));
224         }
225     }
226 
227     private static void common(final StringBuilder out, final String line) {
228         out.append(' ').append(line).append('\n');
229     }
230 
231     private static void deleted(final StringBuilder out, final String line) {
232         out.append("\033[31m-").append(line).append(Diff.RESET).append('\n');
233     }
234 
235     private static void added(final StringBuilder out, final String line) {
236         out.append("\033[32m+").append(line).append(Diff.RESET).append('\n');
237     }
238 }
