5   package org.eolang.maven;
6   
7   import com.jcabi.log.Logger;
8   import java.util.concurrent.Callable;
9   import java.util.concurrent.ExecutionException;
10  import java.util.concurrent.FutureTask;
11  import java.util.concurrent.TimeUnit;
12  import java.util.concurrent.TimeoutException;
13  import org.apache.maven.plugin.MojoFailureException;
14  
15  /**
16   * A body of a Mojo that must finish before its deadline.
17   *
18   * <p>The body runs in a thread of its own, so that one Mojo which never
19   * returns does not hang the whole build for good. When the deadline passes,
20   * or the body fails, the problem becomes a Maven failure, with the chain of
21   * causes in the log if the build asks for it.</p>
22   *
23   * @since 0.62.0
24   */
25  final class Deadline {
26  
27      /**
28       * The Mojo the body belongs to, also the source of the log records.
29       */
30      private final Object mojo;
31  
32      /**
33       * How many seconds the body may take.
34       */
35      private final long seconds;
36  
37      /**
38       * Whether the chain of causes goes to the log.
39       */
40      private final boolean unroll;
41  
42      /**
43       * Ctor.
44       *
45       * @param mojo The Mojo the body belongs to
46       * @param seconds How many seconds the body may take
47       * @param unroll Whether the chain of causes goes to the log
48       */
49      Deadline(final Object mojo, final long seconds, final boolean unroll) {
50          this.mojo = mojo;
51          this.seconds = seconds;
52          this.unroll = unroll;
53      }
54  
55      /**
56       * Run the body and wait for it, but no longer than the deadline.
57       *
58       * <p>The body goes into a {@link FutureTask} carried by a daemon
59       * thread of its own, rather than into an executor service. A cancel
60       * only interrupts the running task, and a task blocked on something
61       * that ignores interruption (a tight CPU loop, a blocking socket read)
62       * may never terminate. An executor would then have to be shut down and
63       * waited for, and that wait would either hang this method forever -
64       * silently swallowing whatever {@link MojoFailureException} the
65       * deadline itself just raised - or give up after a bounded time and
66       * leave the thread running anyway. A daemon thread needs neither: it
67       * is abandoned where it stands and never holds the JVM back from
68       * exiting.</p>
69       *
70       * <p>A deadline of zero seconds is no deadline: the body is waited for
71       * as long as it takes. Anything else would leave a build no way to say
72       * "do not time me out" other than naming a number large enough to never
73       * arrive, which is the default this class was given to replace.</p>
74       *
75       * @param body The body of the Mojo
76       * @throws MojoFailureException If the deadline passes or the body fails
77       */
78      void spent(final Callable<?> body) throws MojoFailureException {
79          if (this.seconds < 0L) {
80              throw new IllegalArgumentException(
81                  String.format(
82                      "The timeout must not be negative, while %d was given through eo.timeout",
83                      this.seconds
84                  )
85              );
86          }
87          final FutureTask<?> task = new FutureTask<>(body);
88          final Thread thread = new Thread(
89              task,
90              String.format("%s-deadline", this.mojo.getClass().getSimpleName())
91          );
92          thread.setDaemon(true);
93          thread.start();
94          try {
95              if (this.seconds == 0L) {
96                  task.get();
97              } else {
98                  task.get(this.seconds, TimeUnit.SECONDS);
99              }
100         } catch (final TimeoutException ex) {
101             this.reported(
102                 Logger.format(
103                     "Timeout %[ms]s for Mojo execution is reached",
104                     TimeUnit.SECONDS.toMillis(this.seconds)
105                 ),
106                 ex
107             );
108         } catch (final ExecutionException ex) {
109             this.reported(String.format("'%s' execution failed", this.mojo), ex);
110         } catch (final InterruptedException ex) {
111             Thread.currentThread().interrupt();
112             throw new IllegalStateException(
113                 Logger.format(
114                     "Timeout %[ms]s thread was interrupted",
115                     TimeUnit.SECONDS.toMillis(this.seconds)
116                 ),
117                 ex
118             );
119         } finally {
120             task.cancel(true);
121         }
122     }
123 
124     private void reported(final String msg, final Throwable problem)
125         throws MojoFailureException {
126         if (this.unroll) {
127             for (final String cause : new Causes(problem)) {
128                 Logger.error(this.mojo, cause);
129             }
130         }
131         throw new MojoFailureException(msg, problem);
132     }
133 }
