5   package org.eolang.maven;
6   
7   import io.github.sekator778.seqdir.Seqdir;
8   import java.io.IOException;
9   import org.apache.maven.plugins.annotations.LifecyclePhase;
10  import org.apache.maven.plugins.annotations.Mojo;
11  
12  /**
13   * Compile and lint all EO files.
14   *
15   * <p>This goal combines {@link MjAssemble}, {@link MjLint}, {@link MjMerge},
16   * {@link MjResolve} and {@link MjPlace} goals.
17   * See their documentation to find out more details.
18   * The {@link MjCompile} is useful to run the whole compilation process in one go without
19   * the need to call each goal separately.</p>
20   *
21   * <p>The merge happens right after the lint, so that every goal between this
22   * one and {@link MjTranspile} that reads an object through its tojo, such as
23   * {@link MjDealpha} and {@link MjLower}, reads it in the shape it will be
24   * compiled in and not in the shape the parser left. A project therefore
25   * never has to name {@code merge} between them, and naming it changes
26   * nothing: a member already inside its object is not moved again.</p>
27   *
28   * <p>{@link MjInfer} reads the files the parser wrote, one per source,
29   * and not the merged ones, for the reasons its own documentation gives. Its
30   * tables still fit the merged objects: the parser gives every object the
31   * locator it will carry once it is inside its package object, and the merge
32   * moves none of them.</p>
33   *
34   * @since 0.52
35   */
36  @Mojo(
37      name = "compile",
38      defaultPhase = LifecyclePhase.PROCESS_SOURCES,
39      threadSafe = true
40  )
41  public final class MjCompile extends MjSafe {
42  
43      /**
44       * Ctor.
45       */
46      public MjCompile() {
47          // nothing
48      }
49  
50      @Override
51      public void exec() throws IOException {
52          try (
53              TjsForeign tojos = this.tojos();
54              TjsForeign compile = this.compileTojos();
55              TjsPlaced placed = this.placed()
56          ) {
57              new Compiling(
58                  this.assembling(tojos),
59                  new Linting(
60                      tojos,
61                      compile,
62                      this.target.toPath(),
63                      this.cache.toPath(),
64                      !this.uncached,
65                      this.plugin.getVersion(),
66                      this.skipSourceLints,
67                      this.skipProgramLints,
68                      this.experimental,
69                      this.failOnWarning,
70                      this.lintAsPackage,
71                      this.skipLinting
72                  ),
73                  new Merging(
74                      tojos,
75                      this.target.toPath()
76                  ),
77                  new Resolving(
78                      tojos,
79                      this.target.toPath(),
80                      new CentralMaven(this.system, this.session, this.repositories),
81                      this.discover,
82                      this.zeros,
83                      this.jna,
84                      this.ignoreRuntime,
85                      this.runtime(),
86                      this.conflicts
87                  ),
88                  new Placing(
89                      placed,
90                      new Seqdir(this.target.toPath(), 2).dirs().once("resolve"),
91                      this.classes.toPath(),
92                      this.placeBinaries,
93                      this.skipBinaries,
94                      !this.preserved
95                  )
96              ).exec();
97          }
98      }
99  }
