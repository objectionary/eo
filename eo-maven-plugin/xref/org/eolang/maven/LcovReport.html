5   package org.eolang.maven;
6   
7   import java.util.ArrayList;
8   import java.util.List;
9   import java.util.Map;
10  import java.util.SortedMap;
11  import java.util.TreeMap;
12  
13  /**
14   * A coverage report rendered in the LCOV tracefile format, one
15   * {@code SF}/{@code DA}/{@code LH}/{@code LF} block per source file.
16   *
17   * <p>LCOV is chosen because Codecov and Coveralls both consume it
18   * directly. Format reference:
19   * https://github.com/linux-test-project/lcov/blob/master/man/geninfo.1</p>
20   *
21   * @since 0.75.0
22   */
23  final class LcovReport {
24  
25      /**
26       * Per source file, per line number, how many times it was hit.
27       */
28      private final Map<String, Map<Integer, Integer>> files;
29  
30      /**
31       * Ctor.
32       *
33       * @param hits Per source file, per line number, how many times it was hit
34       */
35      LcovReport(final Map<String, Map<Integer, Integer>> hits) {
36          this.files = hits;
37      }
38  
39      /**
40       * Render the whole tracefile.
41       *
42       * @return The LCOV text
43       */
44      String text() {
45          final StringBuilder out = new StringBuilder(32);
46          final SortedMap<String, Map<Integer, Integer>> byfile = new TreeMap<>(this.files);
47          for (final Map.Entry<String, Map<Integer, Integer>> file : byfile.entrySet()) {
48              final SortedMap<Integer, Integer> byline = new TreeMap<>(file.getValue());
49              final List<String> block = new ArrayList<>(byline.size() + 4);
50              block.add(String.format("SF:%s", file.getKey()));
51              int hit = 0;
52              for (final Map.Entry<Integer, Integer> line : byline.entrySet()) {
53                  block.add(String.format("DA:%d,%d", line.getKey(), line.getValue()));
54                  if (line.getValue() > 0) {
55                      hit += 1;
56                  }
57              }
58              block.add(String.format("LH:%d", hit));
59              block.add(String.format("LF:%d", byline.size()));
60              block.add("end_of_record");
61              out.append(String.join(String.valueOf('\n'), block)).append('\n');
62          }
63          return out.toString();
64      }
65  
66      /**
67       * The percentage of lines with at least one hit, across every file.
68       *
69       * @return A number from 0 to 100
70       */
71      double covered() {
72          int hit = 0;
73          int total = 0;
74          for (final Map<Integer, Integer> lines : this.files.values()) {
75              for (final int count : lines.values()) {
76                  total += 1;
77                  if (count > 0) {
78                      hit += 1;
79                  }
80              }
81          }
82          final double percentage;
83          if (total == 0) {
84              percentage = 100.0;
85          } else {
86              percentage = 100.0 * hit / total;
87          }
88          return percentage;
89      }
90  }
