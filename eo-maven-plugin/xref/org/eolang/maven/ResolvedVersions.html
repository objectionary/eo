5   package org.eolang.maven;
6   
7   import java.util.Collection;
8   import java.util.HashMap;
9   import java.util.HashSet;
10  import java.util.Map;
11  import java.util.Set;
12  import org.apache.maven.model.Dependency;
13  
14  /**
15   * Every version legitimately resolved in a run, grouped by coordinate, so
16   * {@link Resolving#cleanPlace} can tell a stale version (absent from the
17   * whole set) from a sibling version resolved alongside it in the same run.
18   *
19   * @since 0.61.0
20   */
21  final class ResolvedVersions {
22  
23      /**
24       * Dependencies resolved in this run.
25       */
26      private final Collection<Dep> deps;
27  
28      /**
29       * Ctor.
30       *
31       * @param resolved Dependencies resolved in this run
32       */
33      ResolvedVersions(final Collection<Dep> resolved) {
34          this.deps = resolved;
35      }
36  
37      /**
38       * Every version, by coordinate.
39       *
40       * @return Versions, by coordinate
41       */
42      Map<String, Set<String>> byCoordinate() {
43          final Map<String, Set<String>> result = new HashMap<>(0);
44          for (final Dep dep : this.deps) {
45              final Dependency dependency = dep.get();
46              result.computeIfAbsent(
47                  new DepCoordinate(dependency).value(),
48                  key -> new HashSet<>(0)
49              ).add(dependency.getVersion());
50          }
51          return result;
52      }
53  }
