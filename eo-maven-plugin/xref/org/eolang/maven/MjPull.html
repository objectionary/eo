5   package org.eolang.maven;
6   
7   import java.io.IOException;
8   import org.apache.maven.plugins.annotations.LifecyclePhase;
9   import org.apache.maven.plugins.annotations.Mojo;
10  
11  /**
12   * Pull EO files from Objectionary.
13   *
14   * <p>This goal goes through all objects from "foreign" catalog and looks for those without
15   * sources and pulls them from Objectionary remote repository.
16   * The pulled sources are stored in a directory {@code Seqdir} numbers "pull".</p>
17   *
18   * @since 0.1
19   */
20  @Mojo(
21      name = "pull",
22      defaultPhase = LifecyclePhase.PROCESS_SOURCES,
23      threadSafe = true
24  )
25  public final class MjPull extends MjSafe {
26  
27      /**
28       * Ctor.
29       */
30      public MjPull() {
31          // nothing
32      }
33  
34      @Override
35      public void exec() throws IOException {
36          try (TjsForeign tojos = this.tojos()) {
37              new Pulling(
38                  tojos,
39                  this.target.toPath(),
40                  this.hash,
41                  this.objectionary(),
42                  this.cache.toPath().resolve(Pulling.CACHE),
43                  this.plugin.getVersion(),
44                  this.overwrite,
45                  !this.uncached,
46                  this.offline
47              ).exec();
48          }
49      }
50  }
