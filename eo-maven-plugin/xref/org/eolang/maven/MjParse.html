5   package org.eolang.maven;
6   
7   import java.io.IOException;
8   import org.apache.maven.plugins.annotations.LifecyclePhase;
9   import org.apache.maven.plugins.annotations.Mojo;
10  import org.apache.maven.plugins.annotations.ResolutionScope;
11  
12  /**
13   * Parse EO to XML.
14   *
15   * <p>This is the initial goal that parses all found EO sources to XMIRs.
16   * You can read more about XMIR format
17   * <a href="https://www.eolang.org/XMIR.html">here</a></p>
18   *
19   * <p>The goal scans all the EO sources registered in the foreign file catalog
20   * (see {@link MjRegister} and {@link MjPull}) and then parses those that were not parsed
21   * before (i.e. do not have XMIRs yet) to XMIR format.
22   * The resulting XMIR files are stored in a directory {@code Seqdir} numbers "parse".</p>
23   *
24   * @since 0.1
25   */
26  @Mojo(
27      name = "parse",
28      defaultPhase = LifecyclePhase.GENERATE_SOURCES,
29      threadSafe = true,
30      requiresDependencyResolution = ResolutionScope.COMPILE
31  )
32  public final class MjParse extends MjSafe {
33  
34      /**
35       * Ctor.
36       */
37      public MjParse() {
38          // nothing
39      }
40  
41      @Override
42      public void exec() throws IOException {
43          try (TjsForeign tojos = this.tojos()) {
44              new Parsing(
45                  tojos,
46                  this.target.toPath(),
47                  this.sourcesDir.toPath(),
48                  this.caching(Parsing.CACHE)
49              ).exec();
50          }
51      }
52  }
