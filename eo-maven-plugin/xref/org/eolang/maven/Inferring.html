5   package org.eolang.maven;
6   
7   import com.jcabi.log.Logger;
8   import com.jcabi.xml.XMLDocument;
9   import java.io.IOException;
10  import java.nio.charset.StandardCharsets;
11  import java.nio.file.Files;
12  import java.nio.file.Path;
13  import org.cactoos.iterable.Filtered;
14  import org.cactoos.iterable.Sorted;
15  import org.eolang.inference.Answers;
16  import org.eolang.inference.Facts;
17  import org.eolang.inference.Names;
18  import org.eolang.inference.Prepared;
19  import org.eolang.inference.Walk;
20  
21  /**
22   * Read what a program says about its own objects and write it down.
23   *
24   * <p>The facts of a whole program are read at once, because an object in
25   * one file is used by objects in others and no single file can be read on
26   * its own. The XMIR files are named by the caller, prepared and left in the
27   * second directory, while {@code facts.xml} goes to the third.
28   * {@link MjInfer} names them from the registered tojos, and not from
29   * whatever the parse directory still holds: a source that was deleted
30   * leaves its XMIR behind, and reading that XMIR would put the facts of an
31   * object nobody wrote any more into the document (#9271).</p>
32   *
33   * <p>Every object is named by the locator the parser gives it,
34   * {@code Φ.app.inc.φ.ρ}, which is unique across the whole program and needs
35   * no dictionary to be read. It also makes the facts of many files one
36   * document, since no two files can name the same locator.</p>
37   *
38   * <p>Each file is made ready by {@link Prepared} before {@link Facts} reads
39   * it: every chain of names is split into one object per name, and the
40   * locators are set again on the objects the split made. That is what the
41   * prepared files hold, and they are worth keeping, since every row of
42   * {@code facts.xml} points into them.</p>
43   *
44   * <p>The engine then reads {@code facts.xml} alone and writes
45   * {@code answers.xml} beside it: for every place, which objects it can be
46   * when Java runs the program. It knows a few names of the program, the
47   * {@link Names}, and nothing else of it.</p>
48   *
49   * @since 0.67.0
50   */
51  final class Inferring implements Step {
52  
53      /**
54       * The XMIR files the facts are read from.
55       */
56      private final Iterable<Path> sources;
57  
58      /**
59       * The directory the XMIR files come from, which says where under
60       * {@link #prepared} each of them lands.
61       */
62      private final Path input;
63  
64      /**
65       * The directory for the prepared XMIR files, which keep the names they
66       * had. It is emptied first, because {@link Facts} reads it back as the whole
67       * program: a file left there by an earlier run would put the facts of a
68       * deleted source into the document.
69       */
70      private final Path prepared;
71  
72      /**
73       * The directory for {@code facts.xml} and {@code answers.xml}.
74       */
75      private final Path tables;
76  
77      /**
78       * The names of the program the engine has to know.
79       */
80      private final Names names;
81  
82      /**
83       * Ctor.
84       *
85       * @param xmirs The XMIR files the facts are read from
86       * @param parsed The directory those files come from
87       * @param pre The directory for the prepared XMIR files
88       * @param rows The directory for {@code facts.xml} and {@code answers.xml}
89       * @param known The names of the program the engine has to know
90       */
91      Inferring(
92          final Iterable<Path> xmirs, final Path parsed, final Path pre, final Path rows,
93          final Names known
94      ) {
95          this.sources = xmirs;
96          this.input = parsed;
97          this.prepared = pre;
98          this.tables = rows;
99          this.names = known;
100     }
101 
102     @Override
103     public void exec() throws IOException {
104         if (Files.exists(this.input)) {
105             new Deleted(this.prepared.toFile()).get();
106             if (Files.exists(this.prepared)) {
107                 throw new IOException(
108                     Logger.format(
109                         "Can't clean up %[file]s: prepared XMIRs of an earlier run are still there, and inferring them together with the current ones would describe sources that no longer exist",
110                         this.prepared
111                     )
112                 );
113             }
114             Files.createDirectories(this.prepared);
115             this.ready();
116             new Facts().follow(this.prepared, this.tables);
117             final long start = System.currentTimeMillis();
118             new Answers(new Walk(this.tables.resolve("facts.xml"), this.names))
119                 .write(this.tables.resolve("answers.xml"));
120             Logger.info(
121                 this,
122                 "The facts and the answers of the program are in %[file]s (%s), the answers took %[ms]s",
123                 this.tables, new Tabled(this.tables).asString(),
124                 System.currentTimeMillis() - start
125             );
126         } else {
127             Logger.info(
128                 this, "The directory %[file]s is absent, nothing to infer from it",
129                 this.input
130             );
131         }
132     }
133 
134     private void ready() throws IOException {
135         final Prepared train = new Prepared();
136         for (final Path source : new Sorted<>(
137             new Filtered<>(
138                 path -> path.toString().endsWith(".xmir") && Files.isRegularFile(path),
139                 this.sources
140             )
141         )) {
142             final Path target = this.prepared.resolve(this.input.relativize(source));
143             Files.createDirectories(target.getParent());
144             Files.write(
145                 target,
146                 train.apply(new XMLDocument(source)).toString().getBytes(StandardCharsets.UTF_8)
147             );
148         }
149     }
150 }
