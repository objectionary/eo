5   package org.eolang.maven;
6   
7   /**
8    * A commit hash in its short form ({@link ChNarrow}), read once ({@link ChCached}).
9    *
10   * @since 0.73.3
11   */
12  final class ChBrief implements CommitHash {
13  
14      /**
15       * The short hash.
16       */
17      private final CommitHash origin;
18  
19      /**
20       * Ctor.
21       *
22       * @param tag The Git tag in objectionary to take the hash of
23       */
24      ChBrief(final String tag) {
25          this(new ChRemote(tag));
26      }
27  
28      /**
29       * Ctor.
30       *
31       * @param full The hash to shorten
32       */
33      ChBrief(final CommitHash full) {
34          this.origin = new ChCached(new ChNarrow(full));
35      }
36  
37      @Override
38      public String value() {
39          return this.origin.value();
40      }
41  }
