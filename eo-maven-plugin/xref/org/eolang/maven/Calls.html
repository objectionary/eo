7   import java.util.Collections;
8   import java.util.List;
9   import java.util.Map;
10  import java.util.Optional;
11  import java.util.Set;
12  import java.util.TreeSet;
13  import java.util.stream.Collectors;
14  import org.eolang.inference.Annotation;

16  /**
17   * What the calls of the program put into the voids of the copies they make.
18   *
19   * <p>A call puts its arguments into the voids of the formation its head
20   * copies, counting the voids other than {@code ρ} in the order they are
21   * written. With {@code [cde msg] > status}, the call {@code status 0 "done"}
22   * at {@code Φ.app.φ} puts the object at {@code Φ.app.φ.α0} into
23   * {@code cde} and the one at {@code Φ.app.φ.α1} into {@code msg}. An
24   * argument bound by name goes into the void it names. A call may leave some
25   * voids empty, and then its arguments still go where their positions and
26   * names say, but a call with an argument past the last void, or with two
27   * arguments for one void, tells nothing here. A
28   * head whose copy already holds something in a void has no room for that
29   * many arguments, so such a call fails before anyone reads its voids.</p>
30   *
31   * <p>An atom is a call that Java makes. The type it says it returns, such as
32   * {@code Φ.posix.status-return(Φ.number,Φ.string)}, names the formation Java
33   * copies and lists in brackets what Java puts into each void, in the same
34   * order.</p>
35   *
36   * @since 0.76.0
37   */
38  final class Calls {
39  
40      /**
41       * The voids of each formation other than {@code ρ}, in the order they
42       * are written, by the formation.
43       */
44      private final Map<String, List<String>> voids;
45  
46      /**
47       * The arguments each call passes by position: the place of each
48       * argument, by its position, by the place of the call.
49       */
50      private final Map<String, Map<Integer, String>> slots;
51  
52      /**
53       * The arguments each call binds by name: the place of each argument, by
54       * the name of the void it is bound to, by the place of the call.
55       */
56      private final Map<String, Map<String, String>> names;
57  
58      /**
59       * The type each atom says it returns, by the atom.
60       */
61      private final Map<String, Annotation> returns;
62  
63      /**
64       * The voids each generic atom returns, by the atom.
65       *
66       * <p>A generic atom, such as {@code [] > recovered /A} with
67       * {@code ? > value /A?} and {@code ? > alternative /A}, returns
68       * whatever was put into one of its voids that carry the same type
69       * variable, and the code does not say which one. Here, the atom
70       * {@code Φ.recovered} has two such voids,
71       * {@code Φ.recovered.v🌵value} and
72       * {@code Φ.recovered.v🌵alternative}.</p>
73       */
74      private final Map<String, List<String>> generic;
75  
76      /**
77       * Ctor.
78       *
79       * @param holes The voids of each formation other than {@code ρ}, in order
80       * @param positional The arguments each call passes by position
81       * @param named The arguments each call binds by name
82       * @param types The type each atom says it returns
83       */
84      Calls(
85          final Map<String, List<String>> holes,
86          final Map<String, Map<Integer, String>> positional,
87          final Map<String, Map<String, String>> named,
88          final Map<String, Annotation> types
89      ) {
90          this(holes, positional, named, types, Collections.emptyMap());
91      }
92  
93      /**
94       * Ctor.
95       *
96       * @param holes The voids of each formation other than {@code ρ}, in order
97       * @param positional The arguments each call passes by position
98       * @param named The arguments each call binds by name
99       * @param types The type each atom says it returns
100      * @param variable The voids each generic atom returns, by the atom
101      */
102     Calls(
103         final Map<String, List<String>> holes,
104         final Map<String, Map<Integer, String>> positional,
105         final Map<String, Map<String, String>> named,
106         final Map<String, Annotation> types,
107         final Map<String, List<String>> variable
108     ) {
109         this.voids = holes;
110         this.slots = positional;
111         this.names = named;
112         this.returns = types;
113         this.generic = variable;
114     }
115 
116     /**
117      * Whether a call is written at the place.
118      *
119      * @param place The place
120      * @return TRUE if it is
121      */
122     boolean contains(final String place) {
123         return this.slots.containsKey(place) || this.names.containsKey(place);
124     }
125 
126     /**
127      * The places of all calls of the program.
128      *
129      * @return The places
130      */
131     Set<String> places() {
132         final Set<String> all = new TreeSet<>(this.slots.keySet());
133         all.addAll(this.names.keySet());
134         return all;
135     }
136 
137     /**
138      * The voids of a formation other than {@code ρ}, in the order they are
139      * written.
140      *
141      * @param formation The formation
142      * @return The names of the voids, empty when the formation has none
143      */
144     List<String> voids(final String formation) {
145         return this.voids.getOrDefault(formation, Collections.emptyList());
146     }
147 
148     /**
149      * The place of the argument that a call puts into a void.
150      *
151      * <p>The arguments of the call must fit into the voids: every positional
152      * argument has a void at its position, and every argument bound by name
153      * goes into a void of the formation that no positional argument fills.
154      * They may leave some voids empty: with
155      * {@code [text width fill cant-pad] > padded-left}, the call
156      * {@code "42".padded-left 5 " "} puts {@code 5} into {@code width} and
157      * leaves {@code cant-pad} empty (#10010).</p>
158      *
159      * @param call The place of the call
160      * @param formation The formation the call copies
161      * @param name The name of the void
162      * @return The place of the argument, or nothing when the call leaves the
163      *  void empty or its arguments do not fit into the voids
164      */
165     Optional<String> argument(final String call, final String formation, final String name) {
166         final List<String> holes = this.voids.getOrDefault(formation, Collections.emptyList());
167         final Map<Integer, String> placed = this.slots.getOrDefault(call, Collections.emptyMap());
168         final Map<String, String> bound = this.names.getOrDefault(call, Collections.emptyMap());
169         final int pos = holes.indexOf(name);
170         final Optional<String> found;
171         if (pos < 0 || !Calls.fitting(holes, placed, bound)) {
172             found = Optional.empty();
173         } else {
174             found = Optional.ofNullable(placed.get(pos)).or(
175                 () -> Optional.ofNullable(bound.get(name))
176             );
177         }
178         return found;
179     }
180 
181     /**
182      * Whether a call leaves a void of the formation it copies empty.
183      *
184      * <p>It does when it passes fewer arguments by position than it takes to
185      * reach the void, and binds nothing to it by name. With
186      * {@code [i fallback] > at}, the call {@code s.at 3} leaves out
187      * {@code fallback}, but not {@code i} (#9997).</p>
188      *
189      * @param call The place of the call
190      * @param formation The formation the call copies
191      * @param name The name of the void
192      * @return TRUE if it leaves the void empty
193      */
194     boolean leaves(final String call, final String formation, final String name) {
195         final int pos = this.voids.getOrDefault(formation, Collections.emptyList())
196             .indexOf(name);
197         return pos >= 0
198             && !this.slots.getOrDefault(call, Collections.emptyMap()).containsKey(pos)
199             && !this.names.getOrDefault(call, Collections.emptyMap()).containsKey(name);
200     }
201 
202     /**
203      * The type an atom says it returns.
204      *
205      * @param atom The formation of the atom
206      * @return The type, or nothing when the formation is no atom
207      */
208     Optional<Annotation> returned(final String atom) {
209         return Optional.ofNullable(this.returns.get(atom));
210     }
211 
212     /**
213      * The voids a generic atom returns: the places of all its voids that
214      * carry the type variable the atom says it returns.
215      *
216      * @param atom The formation of the atom
217      * @return The places of the voids, empty when the atom is not generic
218      */
219     List<String> slots(final String atom) {
220         return this.generic.getOrDefault(atom, Collections.emptyList());
221     }
222 
223     /**
224      * The types with brackets that the atoms say they return, which copy a
225      * formation.
226      *
227      * <p>Only the brackets tell what Java puts into the voids of the copy
228      * itself. An atom that returns a type with no brackets, such as
229      * {@code Φ.bool}, builds its copy the way {@code Data.ToPhi} does, out of
230      * the objects of the program, such as {@code Φ.true}, so the program
231      * fills the voids of that copy, not Java.</p>
232      *
233      * @param formation The formation
234      * @return The types, one for each atom that returns a copy of it with
235      *  brackets
236      */
237     List<Annotation> made(final String formation) {
238         return this.returns.values().stream().filter(
239             type -> type.head().equals(formation) && !type.members().isEmpty()
240         ).collect(Collectors.toList());
241     }
242 
243     /**
244      * The type that Java puts into a void of a copy an atom returns.
245      *
246      * @param type The type the atom says it returns
247      * @param name The name of the void
248      * @return The type in the brackets, or nothing when they leave the void
249      *  out
250      */
251     Optional<Annotation> member(final Annotation type, final String name) {
252         final List<Annotation> members = type.members();
253         final int pos = this.voids.getOrDefault(
254             type.head(), Collections.emptyList()
255         ).indexOf(name);
256         final Optional<Annotation> found;
257         if (pos >= 0 && pos < members.size()) {
258             found = Optional.of(members.get(pos));
259         } else {
260             found = Optional.empty();
261         }
262         return found;
263     }
264 
265     private static boolean fitting(
266         final List<String> holes, final Map<Integer, String> placed,
267         final Map<String, String> bound
268     ) {
269         boolean fit = true;
270         for (final int pos : placed.keySet()) {
271             if (pos < 0 || pos >= holes.size()) {
272                 fit = false;
273                 break;
274             }
275         }
276         for (final String name : bound.keySet()) {
277             final int pos = holes.indexOf(name);
278             if (pos < 0 || placed.containsKey(pos)) {
279                 fit = false;
280                 break;
281             }
282         }
283         return fit;
284     }
