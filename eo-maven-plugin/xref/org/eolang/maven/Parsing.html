34  /**
35   * Parse EO to XML.
36   *
37   * <p>This class parses all found EO sources to XMIRs.
38   * You can read more about XMIR format
39   * <a href="https://www.eolang.org/XMIR.html">here</a></p>
40   *
41   * <p>The class scans all the EO sources registered in the foreign file catalog
42   * and then parses those that were not parsed before (i.e. do not have XMIRs yet)
43   * to XMIR format. The resulting XMIR files are stored in a directory
44   * {@code Seqdir} numbers "parse".</p>
45   *
46   * @since 0.1
47   */
48  final class Parsing implements Step {
49  
50      /**
51       * Zero version.
52       */
53      static final String ZERO = "0.0.0";
54  
55      /**
56       * Directory for parsed cache.
57       */
58      static final String CACHE = "parsed";
59  
60      /**
61       * Foreign tojos catalog.
62       */
63      private final TjsForeign tojos;
64  
65      /**
66       * Target directory.
67       */
68      private final Path target;
69  
70      /**
71       * EO sources directory (used for logging).
72       */
73      private final Path home;
74  
75      /**
76       * Where the results of earlier builds are looked for and kept.
77       */
78      private final GlobalCache cache;
79  
80      /**
81       * Constructor.
82       *
83       * @param srcs Foreign tojos catalog
84       * @param target Target directory
85       * @param sources EO sources directory
86       * @param store Where the results of earlier builds are looked for and kept
87       */
88      Parsing(
89          final TjsForeign srcs,
90          final Path target,
91          final Path sources,
92          final GlobalCache store
93      ) {
94          this.tojos = srcs;
95          this.target = target;
96          this.home = sources;
97          this.cache = store;
98      }
99  
100     @Override
101     public void exec() {
102         final Collection<TjForeign> sources = this.tojos.withSources();
103         final String objects = sources.stream()
104             .map(TjForeign::identifier)
105             .filter(id -> id.contains("."))
106             .distinct()
107             .sorted()
108             .collect(Collectors.joining(" "));
109         final int total = this.parsed(
110             sources,
111             new Canonical(objects),
112             this.cache.with(
113                 new Fingerprint(
114                     Stream.concat(
115                         Canonical.XSLS.stream(), Canonical.IMPORTS.stream()
116                     ).toArray(String[]::new)
117                 ).get()
118             ).with(
119                 new UncheckedText(
120                     new HexOf(new Sha256DigestOf(new InputOf(objects)))
121                 ).asString()
122             )
123         );
124         if (0 == total) {
125             if (sources.isEmpty()) {
126                 Logger.info(
127                     this,
128                     "No .eo sources registered, nothing to be parsed to XMIRs (maybe you forgot to execute the \"register\" goal?)"
129                 );
130             } else {
131                 Logger.info(
132                     this,
133                     "No new .eo sources out of %d parsed to XMIRs",
134                     sources.size()
135                 );
136             }
137         } else {
138             Logger.info(
139                 this, "Parsed %d new .eo sources out of %d to XMIRs",
140                 total, sources.size()
141             );
142         }
143     }
144 
145     private int parsed(
146         final Collection<TjForeign> sources,
147         final UnaryOperator<XML> pipeline,
148         final GlobalCache store
149     ) {
150         return new Threaded<>(
151             new Filtered<>(this::unparsed, sources),
152             tojo -> this.parsed(tojo, pipeline, store)
153         ).total();
154     }
155 
156     private boolean unparsed(final TjForeign tojo) throws IOException {
157         return tojo.notParsed()
158             || !tojo.xmir().startsWith(
159                 new Seqdir(this.target, 2).dirs().once("parse")
160             );
161     }
162 
163     private int parsed(
164         final TjForeign tojo, final UnaryOperator<XML> pipeline, final GlobalCache store
165     ) throws Exception {
166         final Path source = tojo.source();
167         final String name = tojo.identifier();
168         final Path base = new Seqdir(this.target, 2).dirs().once("parse");
169         final Path xmir = new Place(name).make(base, MjAssemble.XMIR);
170         final List<Node> refs = new ArrayList<>(1);
171         store.footprint(
172             base.relativize(xmir),
173             new TojoHash(tojo),
174             src -> {
175                 final Node node = this.parsed(tojo, pipeline);
176                 refs.add(node);
177                 return new XMLDocument(node).toString();
178             }
179         ).apply(source, xmir);
180         tojo.withXmir(xmir)
181             .withVersion(Parsing.tojoVersion(xmir, refs))
182             .withDigest(new Sha(source).toString());
183         final List<Xnav> errors = new Xnav(xmir)
184             .element("object")
185             .element("errors")
186             .elements(Filter.withName("error"))
187             .collect(Collectors.toList());
188         if (errors.isEmpty()) {
189             Logger.debug(this, "Parsed %[file]s to %[file]s", source, xmir);
190         } else {
191             for (final Xnav error : errors) {
192                 this.logged(source, error);
193             }
194         }
195         return 1;
196     }
197 
198     private void logged(final Path source, final Xnav error) {
199         final String location = Logger.format(
200             "'%[file]s:%s': %s",
201             source,
202             error.attribute("line").text().orElse("0"),
203             error.text().orElse("")
204         );
205         if ("warning".equals(error.attribute("severity").text().orElse("error"))) {
206             Logger.warn(this, "Warning while parsing %s", location);
207         } else {
208             Logger.error(this, "Failed to parse %s", location);
209         }
210     }
211 
212     private Node parsed(
213         final TjForeign tojo, final UnaryOperator<XML> pipeline
214     ) throws IOException {
215         final Xmir xmir = new EoSource(tojo.identifier(), tojo.source(), pipeline).parsed(
216             pipeline.apply(
217                 new Raws(
218                     this.cache.with("raws"), new Seqdir(this.target, 2).dirs().once("raw")
219                 ).of(tojo)
220             )
221         );
222         Logger.debug(
223             Parsing.class,
224             "Parsed program '%s' from %[file]s:%n %s",
225             tojo.identifier(), this.home.relativize(tojo.source().toAbsolutePath()), xmir
226         );
227         if (xmir.broken()) {
228             new Saved(
229                 new TextOf(xmir.xml().toString()),
230                 this.target.resolve(
231                     String.format("broken-%x.xmir", System.nanoTime())
232                 )
233             ).value();
234         }
235         return xmir.xml().inner();
236     }
237 
238     private static String tojoVersion(
239         final Path target,
240         final List<Node> parsed
241     ) throws FileNotFoundException {
242         final Node node;
243         if (parsed.isEmpty()) {
244             node = new XMLDocument(target).inner();
245         } else {
246             node = parsed.get(0);
247         }
248         return new Xnav(node)
249             .element("object")
250             .element("metas").elements(
251                 Filter.all(
252                     Filter.withName("meta"),
253                     meta -> new Xnav(meta).elements(
254                         Filter.all(
255                             Filter.withName("head"),
256                             head -> head.text().map("version"::equals).orElse(false)
257                         )
258                     )
259                     .findAny()
260                     .isPresent()
261                 )
262             )
263             .findFirst()
264             .map(meta -> meta.element("tail").text().orElse(Parsing.ZERO))
265             .orElse(Parsing.ZERO);
266     }
