5   package org.eolang.maven;
6   
7   import java.util.Collection;
8   import java.util.stream.Collectors;
9   import org.apache.maven.settings.Proxy;
10  import org.apache.maven.settings.Settings;
11  import org.cactoos.Scalar;
12  
13  /**
14   * Active proxies of Maven settings, translated to the ones of Java.
15   *
16   * <p>Only an HTTP proxy is translated. {@code java.net.http} speaks to no
17   * other kind, so a {@code socks5} one would be taken from the settings and
18   * then quietly stepped around, which is worse than saying it cannot be
19   * used.</p>
20   *
21   * @since 0.62.0
22   */
23  final class Proxies implements Scalar<MvnProxy[]> {
24  
25      /**
26       * Maven settings.
27       */
28      private final Settings settings;
29  
30      /**
31       * Ctor.
32       *
33       * @param settings Maven settings
34       */
35      Proxies(final Settings settings) {
36          this.settings = settings;
37      }
38  
39      @Override
40      public MvnProxy[] value() {
41          final Collection<Proxy> active =
42              this.settings.getProxies()
43                  .stream()
44                  .filter(Proxy::isActive)
45                  .collect(Collectors.toList());
46          for (final Proxy proxy : active) {
47              final String protocol = proxy.getProtocol();
48              if (protocol != null && !protocol.isEmpty()
49                  && !"http".equalsIgnoreCase(protocol) && !"https".equalsIgnoreCase(protocol)) {
50                  throw new IllegalStateException(
51                      String.format(
52                          "The proxy %s:%d speaks %s, and only an HTTP proxy can be used here",
53                          proxy.getHost(), proxy.getPort(), protocol
54                      )
55                  );
56              }
57          }
58          return active.stream().map(MvnProxy::new).toArray(MvnProxy[]::new);
59      }
60  }
