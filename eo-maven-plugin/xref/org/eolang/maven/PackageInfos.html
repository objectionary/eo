5   package org.eolang.maven;
6   
7   import com.jcabi.log.Logger;
8   import java.io.File;
9   import java.io.IOException;
10  import java.nio.file.Files;
11  import java.nio.file.Path;
12  import java.util.Collection;
13  import java.util.List;
14  import java.util.regex.Pattern;
15  import java.util.stream.Collectors;
16  import org.eolang.cache.Saved;
17  
18  /**
19   * Package info classes.
20   *
21   * @since 0.60
22   */
23  final class PackageInfos {
24  
25      /**
26       * Pattern for stripping the leading EO (object) or EO_ (package) prefix
27       * off each dot-separated segment of a package, anchored to the start of
28       * the string or right after a dot so an EO occurring inside a segment's
29       * own name survives.
30       */
31      private static final Pattern PACKAGE = Pattern.compile("(^|\\.)EO_?");
32  
33      /**
34       * Pattern for replacing first default org.eolang package, with the dot
35       * that follows it optional, so that the root package itself is left with
36       * an empty EO name rather than with the Java name of the base.
37       */
38      private static final Pattern BASE = Pattern.compile("^org\\.eolang\\.?");
39  
40      /**
41       * Not allowed characters in package names.
42       */
43      private static final Pattern NOT_ALLOWED = Pattern.compile("[^a-zA-Z0-9_.$]");
44  
45      /**
46       * Pattern for package names beginning with a number.
47       */
48      private static final Pattern NUMBER_BEGINNING = Pattern.compile("^\\d.*");
49  
50      /**
51       * Directory where create package info files.
52       */
53      private final Path root;
54  
55      /**
56       * Directories with the Java sources a human wrote.
57       */
58      private final Collection<Path> sources;
59  
60      /**
61       * The directories a transpile of this run wrote a class into.
62       */
63      private final Collection<Path> dirs;
64  
65      /**
66       * Constructor.
67       *
68       * @param root In which directory create files
69       * @param sources Where the hand-written Java sources are
70       * @param dirs Where this run's own transpiled classes landed
71       */
72      PackageInfos(final Path root, final Collection<Path> sources, final Collection<Path> dirs) {
73          this.root = root;
74          this.sources = sources;
75          this.dirs = dirs;
76      }
77  
78      /**
79       * Create {@code package-info.java} files in the directories this run
80       * wrote a transpiled class into.
81       *
82       * <p>Never in every directory under the {@link #root}:
83       * {@code generated-sources} is the standard Maven convention
84       * directory, and other generators write their own output there, which
85       * a package annotation of ours has no business claiming. This is the
86       * same rule {@link JavaFiles#removeStale()} follows on the way
87       * out.</p>
88       *
89       * @return Amount of created files
90       * @throws IOException If fails to create a file
91       */
92      int create() throws IOException {
93          final List<Path> made = this.dirs.stream()
94              .filter(dir -> !dir.equals(this.root))
95              .filter(dir -> !dir.equals(this.root.resolve("org")))
96              .filter(dir -> !this.taken(dir))
97              .sorted()
98              .collect(Collectors.toList());
99          for (final Path dir : made) {
100             Logger.debug(
101                 this,
102                 "Created %s",
103                 new Saved(
104                     PackageInfos.content(
105                         this.root.relativize(dir)
106                             .toString()
107                             .replace(File.separator, ".")
108                     ), dir.resolve("package-info.java")
109                 ).value()
110             );
111         }
112         return made.size();
113     }
114 
115     private boolean taken(final Path dir) {
116         final Path pkg = this.root.relativize(dir);
117         return this.sources.stream().anyMatch(
118             src -> Files.exists(src.resolve(pkg).resolve("package-info.java"))
119         );
120     }
121 
122     private static String content(final String pkg) {
123         return String.join(
124             System.lineSeparator(),
125             "/**",
126             " * This file was auto-generated by eo-maven-plugin,",
127             " * don't modify it, all changes will be lost anyway.",
128             " */",
129             String.format(
130                 "@org.eolang.XmirPackage(\"%s\")",
131                 PackageInfos.BASE.matcher(
132                     PackageInfos.PACKAGE.matcher(pkg).replaceAll("$1")
133                 ).replaceFirst("")
134             ),
135             String.format("package %s;", PackageInfos.escaped(pkg))
136         );
137     }
138 
139     private static String escaped(final String pkg) {
140         String escaped = PackageInfos.NOT_ALLOWED.matcher(pkg).replaceAll("_");
141         if (PackageInfos.NUMBER_BEGINNING.matcher(escaped).matches()) {
142             escaped = String.format("_%s", escaped);
143         }
144         return escaped;
145     }
146 }
