23  /**
24   * Mojo that gives the positional arguments of a program the names of the
25   * voids they land in, right after the {@code inference} goal.
26   *
27   * <p>The parser has no way to know which void an argument fills, so it names
28   * them by their places: in {@code foo k}, the {@code k} is {@code α0}. The
29   * {@code answers.xml} of {@link MjInfer} knows better, since the row of
30   * an argument names the void it lands in whenever there is one. This goal
31   * reads it and writes {@code foo (bar: k)} instead, wherever the answer is
32   * certain, as described in #8301:</p>
33   *
34   * <pre> &lt;o loc="Φ.app.φ.α0" copies="1" lands="bar"&gt;
35   *   &lt;is loc="Φ.number"/&gt;
36   * &lt;/o&gt;</pre>
37   *
38   * <p>The row says that {@code Φ.app.φ.α0} is {@code bar}. The engine writes
39   * the name only when every application it saw puts the argument into one
40   * void and nowhere else. An argument has no {@code lands} when it lands in
41   * two voids, or when the engine cannot tell which formation the application
42   * is made on, such as an unknown or an object made in Java.</p>
43   *
44   * <p>The engine sees only the applications written in the program, and that
45   * is enough: EO always compiles the whole program, so there is no other
46   * program that could attach something else to a void. When another program
47   * appears, the compiler and the engine run again (#10056).</p>
48   *
49   * <p>An argument keeps its place where no single void is known to take it,
50   * and the number of those is written to the log. With {@link #rigid} set,
51   * the build fails when there is one.</p>
52   *
53   * <p>The XMIR goes to a directory {@code Seqdir} numbers "dealpha", and
54   * {@link MjTranspile} reads it from there.</p>
55   *
56   * @since 0.69.0
57   */
58  @Mojo(
59      name = "dealpha",
60      defaultPhase = LifecyclePhase.PROCESS_SOURCES,
61      threadSafe = true
62  )
63  public final class MjDealpha extends MjSafe {
64  
65      /**
66       * The directory with the {@code answers.xml} that {@link MjInfer}
67       * writes, falling back to a directory {@code Seqdir} numbers "inference"
68       * when unset.
69       */
70      @Parameter(alias = "dealpha.tables", property = "eo.dealpha.tables")
71      private File tables;
72  
73      /**
74       * Whether to fail the build when a positional argument is left without
75       * the name of its void.
76       */
77      @Parameter(
78          alias = "dealpha.rigid",
79          property = "eo.dealpha.rigid",
80          defaultValue = "false"
81      )
82      private boolean rigid;
83  
84      /**
85       * Ctor.
86       */
87      public MjDealpha() {
88          // nothing
89      }
90  
91      @Override
92      void exec() throws IOException {
93          final Path inference;
94          if (this.tables == null) {
95              inference = new Seqdir(this.target.toPath(), 2).dirs().once("inference");
96          } else {
97              inference = this.tables.toPath();
98          }
99          try (TjsForeign tojos = this.tojos()) {
100             final Collection<TjForeign> objects = tojos.standalone();
101             if (objects.isEmpty()) {
102                 Logger.debug(this, "No XMIR to name the arguments in");
103             } else {
104                 new Dealphaing(
105                     objects,
106                     MjDealpha.names(inference.resolve("answers.xml")),
107                     this.target.toPath(),
108                     this.rigid
109                 ).exec();
110             }
111         }
112     }
113 
114     private static Map<String, String> names(final Path answers) throws IOException {
115         if (!Files.exists(answers)) {
116             throw new IOException(
117                 Logger.format(
118                     "The table %[file]s is absent, while the arguments can be named only after the 'infer' goal has written it",
119                     answers
120                 )
121             );
122         }
123         return new Xnav(new XMLDocument(answers).inner())
124             .element("answers")
125             .elements(Filter.withName("o"))
126             .filter(row -> row.attribute("lands").text().isPresent()).collect(
127                 Collectors.toMap(
128                     row -> row.attribute("loc").text().get(),
129                     row -> row.attribute("lands").text().get()
130                 )
131             );
132     }
