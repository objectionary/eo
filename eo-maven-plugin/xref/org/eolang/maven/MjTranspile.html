22  /**
23   * Transpile.
24   *
25   * <p>This Maven Mojo transpiles XMIR files into Java source files.
26   * The goal of this class is to automate the process of transforming EO
27   * intermediate representations (XMIRs) into Java code, optimize the output,
28   * and integrate the results into the Maven build lifecycle for further
29   * processing.</p>
30   *
31   * <p>This is one of the final goals that transpiles XMIRs to Java files generated by
32   * {@link MjAssemble}, {@link MjParse} or {@link MjPull} goals.
33   * The resulting Java files are stored in a directory {@code Seqdir} numbers
34   * "transpile". The intermediate optimized XMIRs are stored in a directory
35   * {@code Seqdir} numbers "pre-transpile".</p>
36   *
37   * <p>Before anything is written, the members of every package this build
38   * compiles an object for are put inside that object, the way {@link MjMerge}
39   * does it. The two are the same step and running both changes nothing, since
40   * {@link Merging} leaves a merged member alone; it is done here as well
41   * because the shape of an object decides the name of the Java class of every
42   * atom it holds, and a build that skipped the goal would name classes that
43   * the library it compiles against does not carry. An atom
44   * {@code string.foo.bar} ships as {@code EOstring$EOfoo$EObar}, which is
45   * where it lands once {@code foo} is an attribute of {@code string}, and
46   * nowhere near the {@code org.eolang.EO_string} package an unmerged
47   * {@code +package string} would compile it into (#8295).</p>
48   *
49   * @since 0.1
50   */
51  @Mojo(
52      name = "transpile",
53      defaultPhase = LifecyclePhase.PROCESS_SOURCES,
54      threadSafe = true,
55      requiresDependencyResolution = ResolutionScope.COMPILE
56  )
57  public final class MjTranspile extends MjSafe {
58  
59      /**
60       * Java class name, simple or fully qualified.
61       */
62      private static final Pattern CLASS = Pattern.compile(
63          "^[A-Za-z_$][A-Za-z0-9_$]*+(?:\\.[A-Za-z_$][A-Za-z0-9_$]*+)*+$"
64      );

67       * Add to source root.
68       */
69      @Parameter(
70          alias = "transpile.attach",
71          property = "eo.transpile.attach",
72          defaultValue = "true"

76      /**
77       * Whether to transpile tests.
78       */
79      @Parameter(
80          alias = "transpile.tests",
81          property = "eo.transpile.tests",
82          defaultValue = "true"
83      )
84      private boolean tests;
85  
86      /**
87       * Whether to wrap every dispatched object with a location-carrying
88       * {@code PhSafe}, so a panic reports its {@code .eo} source position.
89       * Off by default, so a production build stays lean; turn it on (as
90       * {@code eo-runtime} does for its tests) to keep precise {@code .eo}
91       * locations in panics.
92       */
93      @Parameter(alias = "transpile.located", property = "eo.transpile.located")
94      private boolean located;
95  
96      /**
97       * Whether to wrap every located object in the generated Java into
98       * {@code PhCoverage}, so it records a coverage hit the first time it
99       * is touched at runtime. Off by default, so a production build stays
100      * lean. The hits are appended to the file named by the
101      * {@code eo.coverageFile} system property of the JVM that runs the
102      * compiled program; when that property is absent, every hit is a
103      * silent no-op. In {@code eo-runtime} the {@code coverage}
104      * profile turns this on and forwards that property to surefire in
105      * one step (see its {@code pom.xml}). The {@code measure}
106      * Maven goal ({@link MjMeasure}) turns the raw hits into an
107      * LCOV tracefile, bound to the {@code verify} phase since the hits
108      * only exist after tests actually run. {@code eo-runtime} pairs it
109      * with {@code measure.minimum} on that same goal, failing the build when
110      * the covered percentage of dataized {@code .eo} objects drops below
111      * a threshold, mirroring how the {@code jacoco} profile binds a
112      * {@code check} goal with its own per-metric thresholds.
113      */
114     @Parameter(alias = "transpile.coverage", property = "eo.transpile.coverage")
115     private boolean coverage;
116 
117     /**
118      * The class that a generated class extends instead of {@code PhDefault},
119      * set through {@code eo.transpile.superclass}. A tool that needs control over
120      * the objects of a program can name its own subclass of
121      * {@code PhDefault} here and see every {@code add()} and {@code take()}
122      * the program makes. The name is written into the generated Java
123      * verbatim, so a class outside {@code org.eolang} has to be named in
124      * full, since the generated files import nothing but
125      * {@code org.eolang.*}.
126      *
127      * <p>The substitution reaches every class the generated Java declares
128      * with an {@code extends} clause of its own, and every object it
129      * builds with {@code new}: the context of a generated
130      * {@code apply()}, an anonymous abstract object, and the argument
131      * of a {@code PhApplication}. An object with a {@code @base}
132      * extends {@code PhOnce} instead, and keeps doing so:
133      * {@code PhOnce} is a decorator, and the object it wraps is itself
134      * substituted, so nothing is lost.</p>
135      *
136      * <p>Since the generated Java builds objects with {@code new}, the
137      * named class has to offer the same constructors {@code PhDefault}
138      * does where the transpiler calls them: the one taking nothing,
139      * the one taking a location {@code String}, and the one taking a
140      * {@code byte[]}. A class that leaves one out fails to compile in
141      * generated sources.</p>
142      */
143     @Parameter(
144         alias = "transpile.superclass",
145         property = "eo.transpile.superclass",
146         defaultValue = "PhDefault"
147     )
148     private String superclass;
149 
150     /**
151      * How many seconds a single test made of an {@code ++>} or {@code -->}
152      * attribute may run before it is stopped and reported as skipped. The
153      * number is written into the {@code @Timeout} of every test, so the tests
154      * keep it in any build that compiles them, even one that configures
155      * nothing for JUnit. A test that outlives it is reported as skipped by
156      * {@code org.eolang.Deadline}, rather than left to run until the heap
157      * of the whole build is gone. Zero means no deadline, and no
158      * {@code @Timeout} is written; a negative number is refused. The property
159      * has the same name as the one {@code eo-runtime} gives its own tests, so
160      * one setting covers both.
161      */
162     @Parameter(
163         alias = "transpile.deadline",
164         property = "eo.transpile.deadline",
165         defaultValue = "1"
166     )
167     private long deadline;
168 
169     /**
170      * How much memory a single test made of an {@code ++>} or {@code -->}
171      * attribute may allocate before it is stopped and reported as skipped.
172      * It is written the way {@code -Xmx} is: {@code 1G}, {@code 512M},
173      * {@code 65536K}, or plain bytes; zero or nothing means no limit. The
174      * value goes into the {@code @Budget} of every test, which
175      * {@code org.eolang.Maxmem} reads while the test runs.
176      */
177     @Parameter(
178         alias = "transpile.memory",
179         property = "eo.transpile.memory",
180         defaultValue = "1G"
181     )
182     private String memory;
183 
184     /**
185      * The directory with the tables that {@link MjInfer} writes, read by
186      * {@code purify.xsl} to find out which formations are safe to cache. The
187      * default is the one {@link MjInfer} saves them to, so a build that
188      * runs the goals in their usual order needs to say nothing here. A build
189      * that skips {@code eo:infer} leaves the directory absent, and then
190      * nothing is marked. Falls back to a directory {@code Seqdir} numbers
191      * "inference" when unset.
192      */
193     @Parameter(alias = "transpile.tables", property = "eo.transpile.tables")
194     private File tables;
195 
196     /**
197      * Ctor.
198      */
199     public MjTranspile() {
200         // nothing
201     }

203     @Override
204     public void exec() throws IOException {
205         try (TjsForeign tojos = this.tojos()) {
206             new Merging(tojos, this.target.toPath()).exec();
207             final Path inference;
208             if (this.tables == null) {
209                 inference = new Seqdir(this.target.toPath(), 2).dirs().once("inference");
210             } else {
211                 inference = this.tables.toPath();
212             }
213             new Transpiling(
214                 tojos.standalone(),
215                 this.target.toPath(),
216                 new Written(
217                     this.generated.toPath(), this.tests, this.roots(),
218                     this.project.getTestCompileSourceRoots().stream()
219                         .map(Paths::get).toArray(Path[]::new)
220                 ),
221                 new Transpilation(
222                     new Tracking(this.tracking, this.located),
223                     this.coverage,
224                     this.base(),
225                     this.seconds(),
226                     this.memory,
227                     this.measures.toPath(),
228                     this.target.toPath(),
229                     inference
230                 ),
231                 this.stored()
232             ).exec();
233         }
234         if (this.attach) {
235             this.project.addCompileSourceRoot(
236                 this.generated.toPath().toAbsolutePath().toString()
237             );
238             Logger.info(
239                 this, "The directory added to Maven 'compile-source-root': %[file]s",
240                 this.generated
241             );
242             final String gtests = this.generated.toPath().getParent().resolve(
243                 "generated-test-sources"
244             ).toAbsolutePath().toString();
245             this.project.addTestCompileSourceRoot(gtests);
246             Logger.info(
247                 this, "The directory added to Maven 'test-compile-source-root': %[file]s",
248                 gtests
249             );
250         }
251     }
252 
253     private Collection<Path> roots() {
254         final Path build = this.target.toPath().getParent();
255         return this.project.getCompileSourceRoots().stream()
256             .map(Paths::get)
257             .filter(root -> !root.startsWith(build))
258             .collect(Collectors.toList());
259     }
260 
261     private long seconds() {
262         if (this.deadline < 0L) {
263             throw new IllegalArgumentException(
264                 String.format(
265                     "The value of eo.transpile.deadline is %d, while zero or a positive number of seconds is expected",
266                     this.deadline
267                 )
268             );
269         }
270         return this.deadline;
271     }
272 
273     private String base() {
274         if (!MjTranspile.CLASS.matcher(this.superclass).matches()) {
275             throw new IllegalArgumentException(
276                 String.format(
277                     "The value of eo.transpile.superclass is '%s', while a Java class name matching '%s' is expected",
278                     this.superclass,
279                     MjTranspile.CLASS
280                 )
281             );
282         }
283         return this.superclass;
284     }
285 
286     // The cache is skipped when the XMIR of every step is asked for: those
287     // dumps are made by the train, and a cache hit hands back a stored
288     // answer without running it (#7724). The version segment is not folded
289     // in here, because it differs from one source to the next: the rows of
290     // the inference tables an object is named in go into it (#7945), so
291     // `Transpiling` folds it in per file.
292     private GlobalCache stored() {
293         final GlobalCache store;
294         if (this.tracking) {
295             store = new GlobalCache.GcFresh();
296         } else {
297             store = this.caching(Transpiling.CACHE);
298         }
299         return store;
300     }
