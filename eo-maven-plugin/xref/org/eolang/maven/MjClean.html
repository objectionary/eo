5   package org.eolang.maven;
6   
7   import com.jcabi.log.Logger;
8   import org.apache.maven.plugins.annotations.LifecyclePhase;
9   import org.apache.maven.plugins.annotations.Mojo;
10  
11  /**
12   * Implementation of maven clean plugin,
13   * just deleting target/eo directory.
14   *
15   * <p>This goal simply deletes the directory containing all the files used for compilation.</p>
16   *
17   * @since 0.28.6
18   */
19  @Mojo(
20      name = "clean",
21      defaultPhase = LifecyclePhase.CLEAN,
22      threadSafe = true
23  )
24  public class MjClean extends MjSafe {
25  
26      /**
27       * Ctor.
28       */
29      public MjClean() {
30          // nothing
31      }
32  
33      @Override
34      final void exec() {
35          if (!this.target.exists()) {
36              Logger.debug(
37                  this,
38                  "The directory %[file]s doesn't exist",
39                  this.target
40              );
41              return;
42          }
43          if (new Deleted(this.target).get()) {
44              Logger.info(
45                  this,
46                  "Deleted all files in the %[file]s directory",
47                  this.target
48              );
49          } else {
50              throw new IllegalStateException(
51                  String.format(
52                      "Failed to delete the directory %s, some files are still there",
53                      this.target
54                  )
55              );
56          }
57      }
58  }
