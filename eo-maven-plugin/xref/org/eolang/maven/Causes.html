5   package org.eolang.maven;
6   
7   import java.util.ArrayList;
8   import java.util.Iterator;
9   import java.util.LinkedHashSet;
10  import java.util.List;
11  import java.util.Objects;
12  
13  /**
14   * The chain of causes of a problem, ready for the log.
15   *
16   * <p>The chain is flattened, and a cause that merely repeats a part of an
17   * earlier one is dropped, so that a reader of the log sees every reason
18   * once and nothing else.</p>
19   *
20   * @since 0.62.0
21   */
22  final class Causes implements Iterable<String> {
23  
24      /**
25       * The problem to explain.
26       */
27      private final Throwable problem;
28  
29      /**
30       * Ctor.
31       *
32       * @param problem The problem to explain
33       */
34      Causes(final Throwable problem) {
35          this.problem = problem;
36      }
37  
38      @Override
39      public Iterator<String> iterator() {
40          final List<String> causes = Causes.all(this.problem);
41          causes.removeIf(Objects::isNull);
42          int idx = 0;
43          while (idx < causes.size()) {
44              final String cause = causes.get(idx);
45              for (int later = idx + 1; later < causes.size(); ++later) {
46                  final String another = causes.get(later);
47                  if (another != null && cause.contains(another)) {
48                      causes.remove(idx);
49                      idx -= 1;
50                      break;
51                  }
52              }
53              idx += 1;
54          }
55          return new LinkedHashSet<>(causes).iterator();
56      }
57  
58      private static List<String> all(final Throwable problem) {
59          final List<String> causes = new ArrayList<>(0);
60          causes.add(problem.getMessage());
61          final Throwable cause = problem.getCause();
62          if (cause != null) {
63              causes.addAll(Causes.all(cause));
64          }
65          return causes;
66      }
67  }
