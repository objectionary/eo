25  /**
26   * Fold the formations of a program into Java atoms.
27   *
28   * <p>A formation whose behaviour is decided at compile time pays the full
29   * cost of the object graph at runtime, for an answer the compiler could
30   * have known. Lowering computes such a formation through the external
31   * {@code phino} binary, which is the only thing here that knows the
32   * calculus, and keeps the answer as the body of a Java atom, so that the
33   * graph is never built while the program runs.</p>
34   *
35   * <p>This goal is the entry point of that pipeline. It runs after
36   * {@code render}, by which time a program has said everything
37   * the lowering needs to hear about it, and before {@code transpile}, the
38   * last moment at which the Java of a program can still be changed.</p>
39   *
40   * <p>The goal folds by default; {@code -Deo.lower.bypass=true} turns it off.
41   * When phino cannot be started on this computer, the goal prints a warning
42   * and does nothing, so the build goes on without atoms. With
43   * {@code -Deo.lower.optional=false} the goal fails the build instead.
44   * When it runs it makes sure that the binary on this machine is the one
45   * the pin names, since an answer of another version cannot be trusted,
46   * and then plants the entries of the build. It hands the lowering the
47   * XMIR of every standalone object, the directory where {@code eo:infer}
48   * left its tables, because an entry is a formation applied to what the
49   * tables say its voids hold, and one home directory, which {@code Seqdir}
50   * numbers as {@code NN-lowering}. There the lowering keeps the world, the
51   * sources with their tests cut out in {@code 1-planting}, and the protocol
52   * of every object morphed in {@code 2-protocols}. The atoms the lowering
53   * renders land in the numbered {@code atoms} directory, which this goal finds and hands to javac
54   * as a source root, and every XMIR file with an atom in the place of a body
55   * lands in the numbered {@code patched} directory, where this goal points
56   * the tojo of that object, so the transpiler reads the patched copy and every other object
57   * stays where it was. The copies stay
58   * from build to build, so only a copy listed in {@code patched.tsv}, one
59   * patched in this very build, is handed to the transpiler.</p>
60   *
61   * <p>With {@code -Deo.lower.minutes=true} the lowering also keeps the
62   * protocol of every entry as indented text, called its "minutes", in the
63   * numbered {@code minutes} directory, beside {@code 2-protocols}, so a person
64   * can read why an entry stopped where it did without any tool. The same run
65   * of phino writes both, so no entry is computed twice.</p>
66   *
67   * @since 0.64.0
68   */
69  @Mojo(
70      name = "lower",
71      defaultPhase = LifecyclePhase.PROCESS_SOURCES,
72      threadSafe = true
73  )
74  public final class MjLower extends MjSafe {

215     @Override
216     void exec() throws IOException {
217         if (this.bypass) {
218             Logger.info(
219                 this,
220                 "Lowering is bypassed with -Deo.lower.bypass=true"
221             );
222         } else {
223             final Path home = new Seqdir(this.target.toPath(), 2)
224                 .dirs().once("lowering").toAbsolutePath();
225             final Path inference;
226             if (this.tables == null) {
227                 inference = new Seqdir(this.target.toPath(), 2).dirs().once("inference");
228             } else {
229                 inference = this.tables.toPath();
230             }
231             try (TjsForeign tojos = this.tojos()) {
232                 final Lowering pipeline = new Lowering(
233                     new ListOf<>(new Mapped<>(TjForeign::xmir, tojos.standalone())),
234                     inference,
235                     home,
236                     this.binary,
237                     this.pin,
238                     this.memory,
239                     this.caching("lowered"),
240                     new Scope(this.only, this.never),
241                     this.steps,
242                     this.loops,
243                     Duration.ofSeconds(this.budget),
244                     this.minutes
245                 );
246                 if (this.optional && !pipeline.available()) {
247                     Logger.warn(
248                         this,
249                         "Lowering is skipped, since phino '%s' cannot be started, set -Deo.lower.optional=false to fail instead",
250                         this.binary
251                     );
252                 } else {
253                     pipeline.exec();
254                     final Path atoms = new Seqdir(home, 1).dirs().find("atoms").orElseThrow(
255                         () -> new IllegalStateException(
256                             String.format(
257                                 "There are no atoms in '%s', while the lowering always makes them",
258                                 home
259                             )
260                         )
261                     );
262                     MjLower.repoint(tojos, home);
263                     this.project.addCompileSourceRoot(atoms.toString());
264                     Logger.info(
265                         this, "The directory added to Maven 'compile-source-root': %[file]s", atoms
266                     );
267                 }
268             }
269         }
270     }
271 
272     private static void repoint(final TjsForeign tojos, final Path home) throws IOException {
273         final Path patched = new Seqdir(home, 1).dirs().find("patched").orElseThrow();
274         final Collection<String> fresh = new ListOf<>(
275             new Mapped<>(
276                 Text::asString,
277                 new Split(new TextOf(home.resolve("patched.tsv")), "\\R")
278             )
279         );
280         for (final TjForeign tojo : tojos.standalone()) {
281             final String name = tojo.xmir().getFileName().toString();
282             if (fresh.contains(name)) {
283                 tojo.withXmir(patched.resolve(name));
284             }
285         }
286     }
