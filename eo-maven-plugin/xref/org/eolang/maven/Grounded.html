21  /**
22   * The arguments of the calls whose head the code alone fixes.
23   *
24   * <p>Which void an argument lands in depends on what its call copies, and
25   * the engine answers that from what the program was seen to put there. In
26   * this program</p>
27   *
28   * <pre> [fail] &gt; guard
29   *   fail &gt; @
30   *     "it broke"</pre>
31   *
32   * <p>every caller puts into {@code fail} a formation whose void is
33   * {@code message}, so the engine says the string lands in {@code message}.
34   * It knows only the callers it saw: a caller in Java, or in a program built
35   * later, may put there a formation whose void has another name, and an
36   * argument passed by name then finds no void to fill. An argument is
37   * grounded only when {@link Origins} follows the head of its call to one
38   * formation from {@code facts.xml} alone, since then no caller can change the
39   * void it lands in. There are two exceptions, and {@link Origins} explains
40   * both. A void that every call of the program fills with a formation written
41   * right there, all with the same voids in the same order, holds one of them,
42   * as {@code true} and {@code false} fill the {@code fork} of {@code bool}. A
43   * void {@code φ} that the engine, in {@code answers.xml} beside
44   * {@code facts.xml}, saw hold one formation and nothing else holds that
45   * formation, as the {@code φ} of a {@code number} holds {@code Φ.bytes}.</p>
46   *
47   * <p>A head that {@code facts.xml} marks with {@code <terminator/>}, such as
48   * {@code T "it broke"}, is where the program stops. {@link Origins} leaves it
49   * out when it agrees on the sides of an {@code .if}, so a name read off
50   * {@code x.if (T "too big") 42} is found in {@code 42} (#9982).</p>
51   *
52   * <p>A void that some places leave empty is unknown, unless a call of the
53   * program, such as a test, puts {@code T} into it, as {@link Fillers} says.
54   * The voids that would be known with such a test are written to the log, so
55   * that the missing tests may be added (#9997).</p>
56   *
57   * <p>What an atom returns is a type, such as
58   * {@code Φ.posix.status-return(Φ.number,Φ.string)}. The formation it fixes is
59   * the name before the brackets, and the brackets fix what Java puts into the
60   * voids of that formation. A call written in the program fixes the voids of
61   * the copy it makes in the same way, with the arguments written there. So
62   * {@link Calls} gets each type whole, the voids of every formation in the
63   * order they are written, and the arguments of every call, split into those
64   * passed by position and those bound by name.</p>
65   *
66   * @since 0.76.0
67   */
68  final class Grounded {
69  
70      /**
71       * The path of {@code facts.xml}.
72       */
73      private final Path facts;
74  
75      /**
76       * The path of {@code answers.xml}, where a row of a place that holds a
77       * single {@code is} and nothing else, no terminator, no datum, no
78       * unknown and no second option, tells the one formation the engine saw
79       * the place hold.
80       */
81      private final Path answers;
82  
83      /**
84       * Ctor.
85       *
86       * @param path The path of {@code facts.xml}
87       */
88      Grounded(final Path path) {
89          this(path, path.resolveSibling("answers.xml"));
90      }
91  
92      /**
93       * Ctor.
94       *
95       * @param path The path of {@code facts.xml}
96       * @param rows The path of {@code answers.xml}, which may be absent
97       */
98      Grounded(final Path path, final Path rows) {
99          this.facts = path;
100         this.answers = rows;
101     }
102 
103     /**
104      * The locators of the arguments whose call has a head the code fixes.
105      *
106      * @return The locators
107      * @throws IOException If the table can't be read
108      */
109     Collection<String> arguments() throws IOException {
110         if (!Files.exists(this.facts)) {
111             throw new IOException(
112                 Logger.format(
113                     "The table %[file]s is absent, while the heads of calls can be followed only after the 'infer' goal has written it",
114                     this.facts
115                 )
116             );
117         }
118         final Xnav table = new Xnav(new XMLDocument(this.facts).inner()).element("facts");
119         final Origins origins = new Origins(
120             table.element("objects").elements(Filter.withName("type")).collect(
121                 Collectors.toMap(
122                     type -> type.attribute("id").text().get(),
123                     Grounded::targets,
124                     (first, second) -> first
125                 )
126             ),
127             new Calls(
128                 table.element("objects").elements(Filter.withName("type")).collect(
129                     Collectors.toMap(
130                         type -> type.attribute("id").text().get(),
131                         type -> type.elements(Filter.withName("attr")).filter(
132                             attr -> "true".equals(attr.attribute("void").text().orElse(""))
133                         ).map(attr -> attr.attribute("name").text().get()).filter(
134                             name -> !"ρ".equals(name)
135                         ).collect(Collectors.toList()),
136                         (first, second) -> first
137                     )
138                 ),
139                 table.element("calls").elements(Filter.withName("call")).collect(
140                     Collectors.toMap(
141                         call -> call.attribute("loc").text().get(),
142                         call -> call.elements(Filter.withName("arg")).filter(
143                             arg -> arg.attribute("n").text().isPresent()
144                         ).collect(
145                             Collectors.toMap(
146                                 arg -> Integer.parseInt(arg.attribute("n").text().get()),
147                                 arg -> arg.attribute("loc").text().get(),
148                                 (first, second) -> first
149                             )
150                         ),
151                         (first, second) -> first
152                     )
153                 ),
154                 table.element("calls").elements(Filter.withName("call")).collect(
155                     Collectors.toMap(
156                         call -> call.attribute("loc").text().get(),
157                         call -> call.elements(Filter.withName("arg")).filter(
158                             arg -> arg.attribute("n").text().isEmpty()
159                         ).collect(
160                             Collectors.toMap(
161                                 arg -> arg.attribute("name").text().get(),
162                                 arg -> arg.attribute("loc").text().get(),
163                                 (first, second) -> first
164                             )
165                         ),
166                         (first, second) -> first
167                     )
168                 ),
169                 table.element("objects").elements(Filter.withName("type")).filter(
170                     type -> type.attribute("returns").text().isPresent()
171                 ).collect(
172                     Collectors.toMap(
173                         type -> type.attribute("id").text().get(),
174                         type -> new Annotation(type.attribute("returns").text().get()),
175                         (first, second) -> first
176                     )
177                 ),
178                 table.element("objects").elements(Filter.withName("type")).filter(
179                     type -> type.elements(Filter.withName("return")).findAny().isPresent()
180                 ).collect(
181                     Collectors.toMap(
182                         type -> type.attribute("id").text().get(),
183                         type -> type.elements(Filter.withName("return")).map(
184                             slot -> slot.attribute("loc").text().get()
185                         ).collect(Collectors.toList()),
186                         (first, second) -> first
187                     )
188                 )
189             ),
190             table.element("reads").elements(Filter.withName("read")).collect(
191                 Collectors.toMap(
192                     read -> read.attribute("id").text().get(),
193                     read -> Map.entry(
194                         read.attribute("of").text().get(),
195                         read.attribute("name").text().get()
196                     ),
197                     (first, second) -> first
198                 )
199             ),
200             table.element("heads").elements(Filter.withName("type")).filter(
201                 type -> type.elements(Filter.withName("ref")).findAny().isPresent()
202             ).collect(
203                 Collectors.toMap(
204                     type -> type.attribute("id").text().get(),
205                     type -> type.element("ref").attribute("loc").text().get(),
206                     (first, second) -> first
207                 )
208             ),
209             this.sole(),
210             table.element("heads").elements(Filter.withName("type")).filter(
211                 type -> type.elements(Filter.withName("terminator")).findAny().isPresent()
212             ).map(type -> type.attribute("id").text().get()).collect(Collectors.toSet())
213         );
214         final Collection<String> found = table.element("calls")
215             .elements(Filter.withName("call"))
216             .filter(call -> origins.formation(call.attribute("loc").text().get()).isPresent())
217             .flatMap(call -> call.elements(Filter.withName("arg")))
218             .map(arg -> arg.attribute("loc").text().get())
219             .collect(Collectors.toSet());
220         if (!origins.vacant().isEmpty()) {
221             Logger.info(
222                 this,
223                 "%d void(s) are left empty by some of their callers, while no call puts T into them, so the arguments read through them may find no void to be named after: %s; if such a void may indeed be left empty, add a test that passes T to it",
224                 origins.vacant().size(),
225                 origins.vacant().stream().map(
226                     hole -> String.format("'%s' of '%s'", hole.getValue(), hole.getKey())
227                 ).sorted().collect(Collectors.joining(", "))
228             );
229         }
230         return found;
231     }
232 
233     private Map<String, String> sole() throws IOException {
234         final Map<String, String> found;
235         if (Files.exists(this.answers)) {
236             found = new Xnav(new XMLDocument(this.answers).inner())
237                 .element("answers")
238                 .elements(Filter.withName("o"))
239                 .filter(Grounded::single)
240                 .filter(row -> row.elements(Filter.withName("is")).count() == 1L).collect(
241                     Collectors.toMap(
242                         row -> row.attribute("loc").text().get(),
243                         row -> row.element("is").attribute("loc").text().get(),
244                         (first, second) -> first
245                     )
246                 );
247         } else {
248             found = Collections.emptyMap();
249         }
250         return found;
251     }
252 
253     private static boolean single(final Xnav row) {
254         return row.elements(node -> node.node().getNodeType() == Node.ELEMENT_NODE)
255             .count() == 1L;
256     }
257 
258     private static Map<String, String> targets(final Xnav type) {
259         return type.elements(Filter.withName("attr")).collect(
260             Collectors.toMap(
261                 attr -> attr.attribute("name").text().get(),
262                 attr -> attr.attribute("rho").text().orElseGet(
263                     () -> attr.attribute("type").text().get()
264                 ),
265                 (first, second) -> second
266             )
267         );
268     }
