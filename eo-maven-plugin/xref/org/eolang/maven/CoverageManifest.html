12  import java.util.Arrays;
13  import java.util.Collection;
14  import java.util.LinkedHashSet;

16  /**
17   * The locations a transpiled XMIR would have {@code PhCoverage} wrapped
18   * around, derived the same way {@code to-java.xsl} itself decides it,
19   * but without deriving it from generated Java text.
20   *
21   * <p>{@code to-java.xsl} (the last shift of {@link Transpilation#XSLS})
22   * wraps a location into {@code PhCoverage} when it carries {@code @line}
23   * and {@code @pos} and its {@code @loc} has no {@code +} and no {@code .-}
24   * in it. Those
25   * attributes are already present by the time every shift except the last
26   * one has run, so running that same prefix here and reading the result
27   * gets the exact same set {@code to-java.xsl} would act on, structurally,
28   * rather than by pattern-matching the Java it emits. Four shapes never get
29   * a hit and are left out here: {@code classes.xsl} marks a whole atomic
30   * class {@code @skip-java}, and {@code to-java.xsl} then writes no
31   * {@code <java>} for it at all; the lambda marker of any atom (the
32   * {@code o[@name='λ']} carrying {@code @atom}, whether the atom is a whole
33   * class or a single attribute) is never the argument {@code to-java.xsl}
34   * runs through {@code located} mode, only its parent is; a free attribute
35   * {@code attrs.xsl} wraps into a {@code <void>}, since {@code to-java.xsl}
36   * compiles it to a bare {@code AtVoid} and never runs it through
37   * {@code located} mode either — the formal attributes a file's root
38   * object declares are free this same way, which is why its own
39   * declaration line never gets a hit (#6995); and, the same way, a void
40   * attribute of any other formation (bracket-declared or written as
41   * {@code ? > name}) parses to an {@code o} whose own {@code @base} is
42   * a void, the empty-set glyph — no body to dataize, so no {@code located} mode call and no
43   * hit ever lands on its declaration line either (#7377).</p>
44   *
45   * <p>An anonymous formation is left out too. {@code anonymous-to-nested.xsl}
46   * turns one into a nested class, and {@code to-java.xsl} never runs a
47   * nested class through {@code located} mode, so no hit can ever land on
48   * the line the {@code []} sits on. Its body still gets its own locations
49   * — only the formation's own declaration line is left out (#7941).</p>
50   *
51   * <p>An atom attribute is left out along with everything it declares,
52   * even though {@code to-java.xsl} does wrap it and its voids and the
53   * runtime does record hits for them (#7055). Such an attribute has no
54   * body in {@code .eo} at all — its body is the Java class the
55   * {@code @atom} names — so the lines it occupies say nothing about how
56   * much of the {@code .eo} source the tests exercised. Counting them
57   * reads as coverage of code that is not there: the declaration line of
58   * {@code [] > read /Q.bytes} shows up uncovered while the {@code ? >
59   * offset} lines under it show up covered, both of them merely
60   * declarations. The hits the runtime still appends for those locations
61   * find no place in the manifest and are dropped, the same way a hit of
62   * any location the manifest does not name is.</p>
63   *
64   * @since 0.75.0
65   */
66  final class CoverageManifest {
67  
68      /**
69       * Every shift of the transpile train except {@code to-java.xsl}.
70       */
71      private final Train<Shift> train = new TrClasspath<Shift>(
72          Arrays.copyOf(Transpilation.XSLS, Transpilation.XSLS.length - 1)
73      ).back();
74  
75      /**
76       * The locations a transpile of this XMIR would instrument, each as
77       * {@code loc:line:pos}, the same string {@code PhCoverage} records a
78       * hit under. An element with no {@code @loc} is not a location at
79       * all: a parser error element carries a line and a position and
80       * nothing else, and every other element of a parsed XMIR that
81       * carries both carries a locator too.
82       *
83       * @param xmir The XMIR a source was parsed and optimized into
84       * @return The locations, in document order
85       */
86      Collection<String> locations(final XML xmir) {
87          final XML passed = new Xsline(this.train).pass(xmir);
88          final Collection<String> found = new LinkedHashSet<>();
89          for (final XML located : passed.nodes(
90              "//*[@line and @pos and @loc and not(contains(@loc,'.p🌵')) and not(contains(@loc,'.n🌵')) and not(@atom) and not(@skip-java) and not(self::class) and not(@base=codepoints-to-string(8709)) and not(ancestor::void) and not(ancestor-or-self::*[o[@atom]]) and (@base or @name)]"
91          )) {
92              found.add(
93                  String.format(
94                      "%s:%s:%s",
95                      located.xpath("./@loc").get(0),
96                      located.xpath("./@line").get(0),
97                      located.xpath("./@pos").get(0)
98                  )
99              );
100         }
101         return found;
102     }
103 }
