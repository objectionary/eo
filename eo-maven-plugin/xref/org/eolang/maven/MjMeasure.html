26  /**
27   * Turns the raw {@code loc:line:pos} coverage hits {@code PhCoverage}
28   * appended at runtime into an LCOV tracefile.
29   *
30   * <p>Bound to the {@code verify} phase, after the tests that actually
31   * populate the raw hits file have run (unlike {@code transpile}, which
32   * only decides whether locations get wrapped in {@code PhCoverage} in
33   * the first place, long before any test executes). Does nothing when
34   * {@code eo.measure.hits} is not set or the file does not exist yet, the
35   * same "no-op when disabled" contract {@code PhCoverage} itself follows,
36   * unless {@code eo.measure.minimum} is above zero: then the build fails,
37   * because a gate that measured nothing must not pass.</p>
38   *
39   * <p>Every {@code .eo} file is re-parsed from its own source instead of
40   * reading the {@code xmir} stored in its tojo, because that xmir is the
41   * {@code MjMerge} one for a package root and mixes the members it spliced
42   * in, whose lines refer to their own files. Reporting those members'
43   * locations against the package root puts lines of unrelated files where
44   * the package's meta directives are, so each file is covered under its
45   * own name (they all carry {@code @loc} with the {@code Φ} prefix of the
46   * object they name, so the runtime hits still line up).</p>
47   *
48   * <p>Every registered {@code .eo} file gets a record of its own, even one
49   * the manifest names no location in: a file of nothing but atom
50   * declarations and unit tests, like {@code bytes.eo}, has no dataizable
51   * body at all. Such a file used to be absent from the tracefile
52   * altogether, which left the report unable to tell it apart from a file
53   * that was never compiled (#7057).</p>
54   *
55   * <p>The raw hits are read one line at a time and never held together in
56   * memory. What the report keeps out of them is one counter per line of
57   * source, so its own size is bounded by the manifest, but the file the
58   * runtime appends to is not: a location is written once per generated
59   * class per JVM, so every surefire fork appends its own copy of the few
60   * thousand distinct locations, and while every {@code PhCoverage} wrapper
61   * still started a dedup set of its own (#6508), a whole test suite left
62   * gigabytes of duplicates behind. Reading that file into a list first is
63   * what killed the {@code eo-runtime} coverage build with an
64   * {@code OutOfMemoryError} once it outgrew the 4Gb heap
65   * {@code .mvn/jvm.config} gives Maven.</p>
66   *
67   * @since 0.75.0
68   */
69  @Mojo(
70      name = "measure",
71      defaultPhase = LifecyclePhase.VERIFY,
72      threadSafe = true
73  )
74  public final class MjMeasure extends MjSafe {
75  
76      /**
77       * The raw {@code loc:line:pos} hits file {@code PhCoverage} appended to.
78       */
79      @Parameter(alias = "measure.hits", property = "eo.measure.hits")
80      private File hits;
81  
82      /**
83       * Where to write the LCOV tracefile.
84       */
85      @Parameter(
86          alias = "measure.lcov",
87          property = "eo.measure.lcov",
88          defaultValue = "${project.build.directory}/coverage.info"
89      )
90      private File lcov;
91  
92      /**
93       * The minimum percentage of dataized {@code .eo} objects that must be
94       * covered, or the build fails. Zero by default, so a build with
95       * coverage tracking off (the common case, see {@code transpile.coverage}
96       * on {@link MjTranspile}) never fails on a threshold it never opted
97       * into; {@code eo-runtime} raises it once tracking is on, mirroring
98       * how the {@code jacoco} profile binds a {@code check} goal with its
99       * own per-metric thresholds.
100      */
101     @Parameter(
102         alias = "measure.minimum",
103         property = "eo.measure.minimum",
104         defaultValue = "0"
105     )
106     private double minimum;
107 
108     /**
109      * Ctor.
110      */
111     public MjMeasure() {
112         // nothing
113     }
114 
115     @Override
116     public void exec() throws IOException {
117         if (this.hits == null || !this.hits.exists()) {
118             this.skipped();
119         } else {
120             this.report();
121         }
122     }
123 
124     private void skipped() throws IOException {
125         if (this.minimum > 0.0d) {
126             throw new IOException(
127                 String.format(
128                     "Can't check the %.1f%% coverage minimum, no hits file at %s",
129                     this.minimum,
130                     Objects.toString(
131                         this.hits, "eo.measure.hits (unset)"
132                     )
133                 )
134             );
135         }
136         Logger.info(
137             this,
138             "No coverage hits file at %[file]s, skipping the LCOV report",
139             this.hits
140         );
141     }
142 
143     private void report() throws IOException {
144         final Map<String, Map<Integer, Integer>> perfile = new LinkedHashMap<>(0);
145         final Map<String, Integer> lineof = new HashMap<>(0);
146         final Map<String, String> fileof = new HashMap<>(0);
147         final CoverageManifest manifest = new CoverageManifest();
148         try (TjsForeign tojos = this.tojos()) {
149             for (final TjForeign tojo : tojos.withXmir()) {
150                 final String source = tojo.source().toString();
151                 final Map<Integer, Integer> lines = perfile.computeIfAbsent(
152                     source, key -> new LinkedHashMap<>(0)
153                 );
154                 final XML xmir = new EoSyntax(
155                     new UncheckedText(
156                         new Utf8(new InputOf(tojo.source()), source)
157                     ).asString()
158                 ).parsed();
159                 if (!xmir.nodes(
160                     "/object/errors/error[@severity='error' or @severity='critical']"
161                 ).isEmpty()) {
162                     throw new IOException(
163                         String.format(
164                             "Can't build a coverage report, %s doesn't parse", source
165                         )
166                     );
167                 }
168                 for (final String location : manifest.locations(xmir)) {
169                     final int last = location.lastIndexOf(':');
170                     final int line = Integer.parseInt(
171                         location.substring(location.lastIndexOf(':', last - 1) + 1, last)
172                     );
173                     lines.putIfAbsent(line, 0);
174                     lineof.put(location, line);
175                     fileof.put(location, source);
176                 }
177             }
178         }
179         try (Stream<String> records = Files.lines(this.hits.toPath())) {
180             records.forEach(
181                 hit -> {
182                     final String source = fileof.get(hit);
183                     if (source != null) {
184                         perfile.get(source).merge(lineof.get(hit), 1, Integer::sum);
185                     }
186                 }
187             );
188         }
189         final LcovReport report = new LcovReport(perfile);
190         new Saved(report.text(), this.lcov.toPath()).value();
191         final double covered = report.covered();
192         Logger.info(
193             this, "EO object coverage: %.1f%%, LCOV report saved to %[file]s",
194             covered, this.lcov
195         );
196         if (covered < this.minimum) {
197             throw new IOException(
198                 String.format(
199                     "EO object coverage is %.1f%%, below the required %.1f%% minimum",
200                     covered, this.minimum
201                 )
202             );
203         }
204     }
205 }
