27  /**
28   * Put the members of a package inside the object that the package names.
29   *
30   * <p>The parsed XMIR of the object and of its members are all on disk by the
31   * time this runs, and the splice is a matter of moving elements: the
32   * top-level {@code <o>} of {@code number/lt.xmir} becomes a child of the
33   * top-level {@code <o>} of {@code number.xmir}. No name inside the moved tree
34   * is rewritten, because the parser leaves every name fully qualified and every
35   * {@code loc} already reads as the locator the node will carry once it is an
36   * attribute of {@code Φ.number}, so no reference can be captured by an
37   * attribute of the object it lands in.</p>
38   *
39   * <p>Every package this build compiles an object for is merged, and no other:
40   * a package whose name no object carries, as {@code examples} in a program
41   * that declares {@code +package examples} and nothing called {@code examples},
42   * keeps its members as objects of their own. An ancestor object must not
43   * hide such a package: without a standalone immediate package object, its
44   * members cannot be reached, so the merge refuses them (#9756).</p>
45   *
46   * <p>A member arrives after the attributes the object already had, so the
47   * places of the voids, and with them the meaning of applying the object to
48   * arguments, stay as they were.</p>
49   *
50   * <p>A void can be read only in the file that declares it (#9788). The
51   * parser hides every void of the object behind {@code v🌵}, together with
52   * every read of it in the object's own file, and nothing renames a name
53   * after the parser. A member is a file of its own, outside its object, so
54   * it reads what the object holds through getters. A member that reads a
55   * void {@code x} of the object finds no attribute {@code x} there and fails
56   * at runtime, like any read from outside.</p>
57   *
58   * <p>Merging twice does nothing the second time: a member that has already
59   * moved is marked as merged in the tojos and is not standalone any more, so
60   * the package it came from is no longer a package with members and nothing
61   * is spliced into the object again. That is what lets {@link MjTranspile}
62   * merge on its own without asking whether the {@link MjMerge} goal has
63   * already run.</p>
64   *
65   * <p>The tests a member declares do not travel with it. A test is legal only
66   * as a direct child of the top-level object of a file, which is what the
67   * parser demands and what the transpiler reads when it writes the test class,
68   * so each one is lifted out of the member and appended to the object beside
69   * the tests the object declares itself. The member is merged before the
70   * package that holds it, so a test written three files deep arrives at the top
71   * one level at a time and nothing has to look for it.</p>
72   *
73   * <p>A lifted test takes the name of the member it came from in front of its
74   * own, so {@code works} of {@code box/left.eo} arrives as
75   * {@code left-works}. Each file names its tests on its own, and two files of
76   * one package naming a test alike is as ordinary as two files naming an
77   * attribute alike, so the name a test arrives under has to say which member
78   * wrote it. Without the prefix the second {@code works} read as a name the
79   * object already held and stopped the build.</p>
80   *
81   * @since 0.68.0
82   */
83  final class Merging implements Step {
84  
85      /**
86       * The length of the mark the parser puts in front of a test: one letter,
87       * {@code p} for a test that must hold and {@code n} for one that must
88       * throw, and the cactus that follows it.
89       */
90      private static final int MARK = "p🌵".length();
91  
92      /**
93       * The tojos of everything this build compiles.
94       */
95      private final TjsForeign tojos;
96  
97      /**
98       * Target directory.
99       */
100     private final Path target;
101 
102     /**
103      * Ctor.
104      *
105      * @param foreign The tojos of everything this build compiles
106      * @param tgt Target directory
107      */
108     Merging(final TjsForeign foreign, final Path tgt) {
109         this.tojos = foreign;
110         this.target = tgt;
111     }
112 
113     @Override
114     public void exec() throws IOException {
115         final Map<String, TjForeign> all = this.indexed();
116         Merging.reachable(all);
117         final Collection<String> found = Merging.deepest(all);
118         int done = 0;
119         for (final String pkg : found) {
120             done = done + this.spliced(pkg, all);
121         }
122         if (done == 0) {
123             Logger.debug(this, "No package member to put inside its object");
124         } else {
125             Logger.info(
126                 this, "Put %d member(s) into %d package object(s), XMIR is in %[file]s",
127                 done, found.size(), this.dir()
128             );
129         }
130     }
131 
132     private static void reachable(final Map<String, TjForeign> all) {
133         for (final String member : all.keySet()) {
134             final int split = member.lastIndexOf('.');
135             if (split > 0 && !all.containsKey(member.substring(0, split))) {
136                 final String pkg = member.substring(0, split);
137                 String parent = pkg;
138                 while (parent.contains(".")) {
139                     parent = parent.substring(0, parent.lastIndexOf('.'));
140                     if (all.containsKey(parent)) {
141                         throw new IllegalStateException(
142                             String.format(
143                                 "The member '%s' is unreachable: package '%s' has no standalone object to merge into, while ancestor object '%s' hides its namespace",
144                                 member, pkg, parent
145                             )
146                         );
147                     }
148                 }
149             }
150         }
151     }
152 
153     private Path dir() throws IOException {
154         return new Seqdir(this.target, 2).dirs().once("merge");
155     }
156 
157     private static Collection<String> deepest(final Map<String, TjForeign> all) {
158         return all.keySet().stream()
159             .filter(name -> name.indexOf('.') > 0)
160             .map(name -> name.substring(0, name.lastIndexOf('.')))
161             .distinct()
162             .filter(all::containsKey)
163             .sorted(Merging.deeper())
164             .collect(Collectors.toList());
165     }
166 
167     private static Comparator<String> deeper() {
168         return Comparator.comparingInt((String pkg) -> pkg.split("\\.").length)
169             .reversed()
170             .thenComparing(Comparator.naturalOrder());
171     }
172 
173     private Map<String, TjForeign> indexed() {
174         final Map<String, TjForeign> all = new HashMap<>(0);
175         for (final TjForeign tojo : this.tojos.standalone()) {
176             all.put(tojo.identifier(), tojo);
177         }
178         return all;
179     }
180 
181     private int spliced(final String pkg, final Map<String, TjForeign> all) throws IOException {
182         final TjForeign object = all.get(pkg);
183         final Map<String, TjForeign> members = Merging.members(pkg, all);
184         final Node formation = Merging.formation(object.xmir());
185         final Collection<String> taken = Merging.names(formation);
186         for (final Map.Entry<String, TjForeign> member : members.entrySet()) {
187             final Node top = formation.getOwnerDocument().importNode(
188                 Merging.top(member.getValue().xmir()).node(), true
189             );
190             final String name = Merging.named(top);
191             if (name.isEmpty()) {
192                 throw new IllegalStateException(
193                     String.format(
194                         "The member '%s' has no name, while only a named object can become an attribute of '%s'",
195                         member.getKey(), pkg
196                     )
197                 );
198             }
199             Merging.claimed(taken, name, member.getKey(), pkg);
200             formation.appendChild(top);
201             for (final Node test : Merging.tests(top)) {
202                 final String lifted = Merging.prefixed(Merging.named(test), name);
203                 Merging.claimed(taken, lifted, member.getKey(), pkg);
204                 test.getAttributes().getNamedItem("name").setNodeValue(lifted);
205                 formation.appendChild(top.removeChild(test));
206             }
207         }
208         final Path dest = new Place(pkg).make(this.dir(), MjAssemble.XMIR);
209         final String merged = new XMLDocument(formation.getOwnerDocument()).toString();
210         if (!Files.exists(dest) || !new Diff(Files.readString(dest), merged).same()) {
211             new Saved(merged, dest).value();
212         }
213         object.withXmir(dest);
214         for (final TjForeign member : members.values()) {
215             member.withMerged(pkg);
216         }
217         Logger.debug(
218             this, "Put %d member(s) of '%s' into %[file]s",
219             members.size(), pkg, dest
220         );
221         return members.size();
222     }
223 
224     private static void claimed(
225         final Collection<String> taken, final String name, final String member, final String pkg
226     ) {
227         if (taken.contains(name)) {
228             throw new IllegalStateException(
229                 String.format(
230                     "The name '%s' arriving from the member '%s' is already an attribute of '%s', while one object cannot hold two attributes under one name",
231                     name, member, pkg
232                 )
233             );
234         }
235         taken.add(name);
236     }
237 
238     private static Collection<Node> tests(final Node object) {
239         final Collection<Node> found = new ArrayList<>(0);
240         final NodeList kids = object.getChildNodes();
241         for (int idx = 0; idx < kids.getLength(); ++idx) {
242             final Node kid = kids.item(idx);
243             final String name = Merging.named(kid);
244             if (name.startsWith("p🌵") || name.startsWith("n🌵")) {
245                 found.add(kid);
246             }
247         }
248         return found;
249     }
250 
251     private static String prefixed(final String test, final String owner) {
252         return String.format(
253             "%s%s-%s",
254             test.substring(0, Merging.MARK),
255             owner,
256             test.substring(Merging.MARK)
257         );
258     }
259 
260     private static String named(final Node object) {
261         return Optional.ofNullable(object.getAttributes())
262             .map(attrs -> attrs.getNamedItem("name"))
263             .map(Node::getNodeValue)
264             .orElse("");
265     }
266 
267     private static Map<String, TjForeign> members(
268         final String pkg, final Map<String, TjForeign> all
269     ) {
270         final String prefix = String.format("%s.", pkg);
271         final Map<String, TjForeign> found = new TreeMap<>();
272         for (final Map.Entry<String, TjForeign> tojo : all.entrySet()) {
273             final String name = tojo.getKey();
274             if (name.startsWith(prefix) && name.indexOf('.', prefix.length()) < 0) {
275                 found.put(name, tojo.getValue());
276             }
277         }
278         return found;
279     }
280 
281     private static Node formation(final Path xmir) throws IOException {
282         return Merging.top(xmir).node();
283     }
284 
285     private static Xnav top(final Path xmir) throws IOException {
286         return new Xnav(new XMLDocument(xmir).inner())
287             .element("object")
288             .element("o");
289     }
290 
291     private static Collection<String> names(final Node formation) {
292         final Collection<String> taken = new ArrayList<>(0);
293         new Xnav(formation).elements(Filter.withName("o")).forEach(
294             attr -> attr.attribute("name").text().ifPresent(taken::add)
295         );
296         return taken;
297     }
