101     /**
102      * Constructor.
103      */
104     CommitHashesText() {
105         this(CommitHashesText::fetch);
106     }
107 
108     /**
109      * Constructor.
110      *
111      * @param source Text source, retried on its own before this class falls
112      *  back to {@link CommitHashesText#FALLBACK}
113      */
114     CommitHashesText(final Text source) {
115         super(new Synced(new Sticky(() -> CommitHashesText.safe(source))));
116     }
117 
118     private static String safe(final Text source) throws Exception {
119         String hashes;
120         try {
121             hashes = source.asString();
122         } catch (final IOException ex) {
123             Logger.warn(
124                 CommitHashesText.class,
125                 "Can't load hashes: %[exception]s",
126                 ex
127             );
128             hashes = CommitHashesText.FALLBACK;
129         }
130         return hashes;
131     }
132 
133     @RetryOnFailure(delay = 1L, unit = TimeUnit.SECONDS)
134     private static String fetch() throws Exception {
135         return new TextOf(new URL("https://home.objectionary.com/tags.txt")).asString();
136     }
