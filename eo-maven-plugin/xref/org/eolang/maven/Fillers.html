18  /**
19   * What the calls of the program and the atoms put into a void, when the
20   * code alone says that.
21   *
22   * <p>Every call that copies the formation puts an argument into the void,
23   * and every atom that returns a copy of it with brackets puts there the type
24   * its brackets give. The void holds one of the formations these copy, when
25   * each of them copies a formation the code fixes, and no place copies the
26   * formation without a call. With one formation the copy is sharp, and with
27   * several it is blurred. Those formations do not have to have the same voids:
28   * a name read off the void is looked up in each of them, while a call on the
29   * void is refused when their voids differ, as {@link Copy#uniform} says
30   * (#9978).</p>
31   *
32   * <p>A call that puts a terminator into the void adds nothing to what the
33   * void holds, since the program stops before the copy is made (#9982).
34   * Neither does an argument that is the very lookup still running, as
35   * {@link Copy#looped()} says (#9983).</p>
36   *
37   * <p>A place that copies the formation without a call, or a call that leaves
38   * the void out, as {@link Calls#leaves} says, leaves the void empty. The code
39   * does not say what is there then, since an atom, or a helper the formation
40   * is handed to, may fill it, so such a place makes the void unknown. There is
41   * one exception: some call of the program puts a terminator into the void,
42   * such as {@code safe 5 T} in a test of {@code [x cant] > safe}. That call
43   * shows that the void may be left with the terminator in it, so a place that
44   * leaves it empty adds nothing more. Then the void holds the terminator, when
45   * no other call fills it, and what the other calls put there otherwise,
46   * in a copy that may be the terminator instead, as {@link Copy#ends()} says,
47   * since a call on the void may then land in the terminator (#9997). A void
48   * that would be known this way, if only some call put a terminator into it,
49   * is remembered, as {@link #vacant()} says, so that the
50   * test it lacks may be named.</p>
51   *
52   * <p>An argument may lead back to the void itself, when a helper passes its
53   * own void to itself. In the first round the void is taken to hold nothing,
54   * and such an argument adds nothing. In each next round the void is taken to
55   * hold what the round before found, and every argument is looked up again,
56   * until what the void holds stops changing.</p>
57   *
58   * @since 0.76.0
59   */
60  final class Fillers {
61  
62      /**
63       * The place each attribute of a formation leads to, by the name, by the
64       * formation.
65       */
66      private final Map<String, Map<String, String>> attributes;
67  
68      /**
69       * What the calls of the program put into the voids of the copies they
70       * make, and what the atoms return.
71       */
72      private final Calls calls;
73  
74      /**
75       * The calls that copy each formation, by the formation.
76       */
77      private final Map<String, List<String>> copies;
78  
79      /**
80       * The formations that some place copies without a call.
81       */
82      private final Set<String> bare;
83  
84      /**
85       * The voids whose fillers are being looked at right now.
86       */
87      private final Guesses guesses;
88  
89      /**
90       * The copy a place makes, looked up with the lookups of one round.
91       */
92      private final BiFunction<String, Lookups, Optional<Copy>> lookup;
93  
94      /**
95       * The voids that are left empty and would be known if some call put a
96       * terminator into them: the formation and the name of each.
97       */
98      private final Set<Map.Entry<String, String>> empties;
99  
100     /**
101      * Ctor.
102      *
103      * @param places The place each attribute of a formation leads to
104      * @param applications What the calls put into the voids
105      * @param made The calls that copy each formation
106      * @param empty The formations that some place copies without a call
107      * @param copy The copy a place makes
108      */
109     Fillers(
110         final Map<String, Map<String, String>> places,
111         final Calls applications,
112         final Map<String, List<String>> made,
113         final Set<String> empty,
114         final BiFunction<String, Lookups, Optional<Copy>> copy
115     ) {
116         this.attributes = places;
117         this.calls = applications;
118         this.copies = made;
119         this.bare = empty;
120         this.guesses = new Guesses();
121         this.lookup = copy;
122         this.empties = new HashSet<>(0);
123     }
124 
125     /**
126      * The voids that the places of the program leave empty, and that would be
127      * known if some call put a terminator into them, among the voids looked
128      * at so far.
129      *
130      * <p>Such a void needs a test that passes the terminator to it, when it
131      * may be left empty indeed: then the program itself says that it may hold
132      * the terminator. A void that an atom, or a helper, fills is listed too,
133      * since the code does not tell that from a void left empty (#9997).</p>
134      *
135      * @return The formation and the name of each void
136      */
137     Set<Map.Entry<String, String>> vacant() {
138         return this.empties;
139     }
140 
141     /**
142      * Whether the fillers of the void are being looked at right now.
143      *
144      * @param place The place of the void
145      * @return TRUE if they are
146      */
147     boolean open(final String place) {
148         return this.guesses.open(place);
149     }

151     /**
152      * What a void whose fillers are being looked at is taken to hold
153      * meanwhile.
154      *
155      * @param place The place of the void
156      * @return The guess, or nothing in the first round
157      */
158     Optional<Copy> hit(final String place) {
159         return this.guesses.hit(place);
160     }
161 
162     /**
163      * A fresh cache of lookups, aware of the voids whose fillers are being
164      * looked at.
165      *
166      * @return The lookups
167      */
168     Lookups lookups() {
169         return new Lookups(this.guesses);
170     }
171 
172     /**
173      * What a void holds, once what it is taken to hold stops changing.
174      *
175      * <p>It takes at most eight rounds, and after that the void is said to
176      * hold nothing the code fixes.</p>
177      *
178      * @param place The place of the void
179      * @param hole The formation of the void, and its name
180      * @return The copy, or nothing when the code does not fix it, and whether
181      *  it may be remembered, which it may not when it leans on what another
182      *  void, whose fillers are still being looked at, is taken to hold
183      */
184     Map.Entry<Optional<Copy>, Boolean> held(
185         final String place, final Map.Entry<String, String> hole
186     ) {
187         this.guesses.guess(place, Optional.empty());
188         final List<Boolean> lacking = new ArrayList<>(1);
189         final Map.Entry<Optional<Copy>, Set<String>> run = this.guesses.tracked(
190             () -> this.held(place, hole, Optional.empty(), 8, lacking)
191         );
192         this.guesses.close(place);
193         final boolean settled = !this.guesses.leaning(run.getValue());
194         if (settled && lacking.contains(true)) {
195             this.empties.add(hole);
196         }
197         return Map.entry(run.getKey(), settled);
198     }
199 
200     private Optional<Copy> held(
201         final String place, final Map.Entry<String, String> hole,
202         final Optional<Copy> guess, final int left, final List<Boolean> lacking
203     ) {
204         final Map.Entry<Optional<Copy>, Boolean> shaped = this.shape(hole);
205         final Optional<Copy> next = shaped.getKey();
206         final Optional<Copy> found;
207         if (next.isEmpty() || next.equals(guess)) {
208             found = next;
209             lacking.add(shaped.getValue());
210         } else if (left == 0) {
211             found = Optional.empty();
212         } else {
213             this.guesses.guess(place, next);
214             found = this.held(place, hole, next, left - 1, lacking);
215         }
216         return found;
217     }
218 
219     private Map.Entry<Optional<Copy>, Boolean> shape(final Map.Entry<String, String> hole) {
220         final List<Optional<Copy>> found = this.found(hole);
221         final boolean ends = found.stream().anyMatch(
222             each -> each.filter(Copy::stops).isPresent()
223         );
224         final List<Optional<Copy>> live = found.stream().filter(
225             each -> each.map(copy -> !copy.stops()).orElse(true)
226         ).collect(Collectors.toList());
227         final boolean vacant = this.vacant(hole);
228         final Optional<Copy> held;
229         if (!ends && (vacant || live.isEmpty())) {
230             held = Optional.empty();
231         } else if (live.isEmpty()) {
232             held = Optional.of(Copy.terminator());
233         } else {
234             held = this.known(live).map(copy -> copy.ending(copy.ends() || ends));
235         }
236         return Map.entry(
237             held, vacant && !ends && (live.isEmpty() || this.known(live).isPresent())
238         );
239     }
240 
241     private boolean vacant(final Map.Entry<String, String> hole) {
242         return this.bare.contains(hole.getKey())
243             || this.copies.getOrDefault(hole.getKey(), List.of()).stream().anyMatch(
244                 call -> this.calls.leaves(call, hole.getKey(), hole.getValue())
245             );
246     }
247 
248     private List<Optional<Copy>> found(final Map.Entry<String, String> hole) {
249         final Lookups seen = this.lookups();
250         final List<Optional<Copy>> found = new ArrayList<>(0);
251         for (final String call : this.copies.getOrDefault(hole.getKey(), List.of())) {
252             if (!this.calls.leaves(call, hole.getKey(), hole.getValue())) {
253                 this.filler(call, hole, seen).ifPresent(found::add);
254             }
255         }
256         for (final Annotation type : this.calls.made(hole.getKey())) {
257             found.add(this.calls.member(type, hole.getValue()).map(Copy::new));
258         }
259         return found;
260     }
261 
262     private Optional<Copy> known(final List<Optional<Copy>> live) {
263         final Set<String> forms = live.stream().flatMap(Optional::stream)
264             .flatMap(copy -> copy.candidates().stream())
265             .map(Copy::formation).collect(Collectors.toCollection(TreeSet::new));
266         final Optional<Copy> found;
267         if (!forms.isEmpty()
268             && live.stream().allMatch(Optional::isPresent)
269             && forms.stream().allMatch(this.attributes::containsKey)) {
270             found = Optional.of(
271                 new Copy(forms).ending(
272                     live.stream().flatMap(Optional::stream).anyMatch(Copy::ends)
273                 )
274             );
275         } else {
276             found = Optional.empty();
277         }
278         return found;
279     }
280 
281     private Optional<Optional<Copy>> filler(
282         final String call, final Map.Entry<String, String> hole,
283         final Lookups seen
284     ) {
285         final Optional<String> arg = this.calls.argument(call, hole.getKey(), hole.getValue());
286         final Map.Entry<Optional<Copy>, Set<String>> run = this.guesses.tracked(
287             () -> arg.flatMap(place -> this.lookup.apply(place, seen))
288         );
289         final Optional<Optional<Copy>> found;
290         if (run.getKey().filter(Copy::loops).isPresent()
291             || run.getKey().isEmpty() && arg.isPresent() && this.guesses.blind(run.getValue())) {
292             found = Optional.empty();
293         } else {
294             found = Optional.of(run.getKey());
295         }
296         return found;
297     }
