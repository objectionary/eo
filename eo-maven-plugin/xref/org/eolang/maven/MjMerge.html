5   package org.eolang.maven;
6   
7   import java.io.IOException;
8   import org.apache.maven.plugins.annotations.LifecyclePhase;
9   import org.apache.maven.plugins.annotations.Mojo;
10  
11  /**
12   * Put every member of a package inside the object that the package names.
13   *
14   * <p>An object and the package of the same name are two things today, and it
15   * takes the runtime to join them: asked for an attribute it does not hold,
16   * {@code Φ.number} goes looking for {@code Φ.number.lt} on the classpath and
17   * binds itself into the first void of what it finds. This goal does that
18   * joining at compile time instead, where the parts are already on disk as
19   * XMIR, so that {@code lt} is simply an attribute of {@code number} and
20   * nothing has to be searched for at runtime.</p>
21   *
22   * <p>It runs after {@link MjLint}, so that every member is read and reported
23   * on as the file a human wrote, and before {@link MjTranspile}, which is the
24   * first goal that cares about the shape of the object it compiles. The
25   * merged XMIR goes to a directory {@code Seqdir} numbers "merge".</p>
26   *
27   * <p>Every package this build compiles an object for is merged. A package
28   * whose name no object carries keeps its members as objects of their own,
29   * reached through the package namespace as before.</p>
30   *
31   * <p>{@link MjCompile} runs the same step right after its lint, and
32   * {@link MjTranspile} runs it again before it writes anything, so a build
33   * that never names this goal still compiles merged objects, and every goal
34   * between the two reads them merged. Naming it anyway costs nothing: a
35   * member already inside its object is not moved again.</p>
36   *
37   * @since 0.68.0
38   */
39  @Mojo(
40      name = "merge",
41      defaultPhase = LifecyclePhase.PROCESS_SOURCES,
42      threadSafe = true
43  )
44  public final class MjMerge extends MjSafe {
45  
46      /**
47       * Ctor.
48       */
49      public MjMerge() {
50          // nothing
51      }
52  
53      @Override
54      void exec() throws IOException {
55          try (TjsForeign tojos = this.tojos()) {
56              new Merging(
57                  tojos,
58                  this.target.toPath()
59              ).exec();
60          }
61      }
62  }
