15  import java.util.Set;
16  import java.util.concurrent.ConcurrentLinkedQueue;

18  import java.util.stream.Collectors;
19  import java.util.stream.Stream;
20  import org.eolang.cache.Footprint;
21  import org.eolang.cache.GlobalCache;

23  /**
24   * The Java files that a transpiled XMIR is written out as.
25   *
26   * <p>One XMIR carries a class per object, and every one of them becomes
27   * a Java file under the generated sources directory, taken from the
28   * global cache when the tojo is at a released hash and put there when
29   * it is not.</p>
30   *
31   * <p>The implementation of an atom is written by hand, so no Java file is
32   * made of it. Its JUnit companion is not: the class of an atom carries its
33   * own {@code <tests>}, and that body reaches {@link JavaPlaced} like any
34   * other. A class with no {@code <java>} body is not counted among the
35   * generated main sources, while the companion is still placed (#9390).</p>
36   *
37   * @since 0.74
38   */
39  final class JavaFiles {
40  
41      /**
42       * Java extension.
43       */
44      private static final String JAVA = "java";
45  
46      /**
47       * Generated sources directory.
48       */
49      private final Path generated;
50  
51      /**
52       * Java files generated during the current transpilation.
53       *
54       * <p>The collection is shared by parallel {@link #total(boolean, Path,
55       * String, boolean, GlobalCache)} calls. It is reconciled only after all XMIRs have
56       * been processed.</p>
57       */
58      private final Collection<Path> fresh;
59  
60      /**
61       * Where a class of the current run would have gone, written or not.
62       *
63       * <p>A class this run decided to skip, such as the implementation of an
64       * atom, names a directory all the same, and a file an earlier run left
65       * there is stale exactly as one beside a file we did write (#7763).</p>
66       */
67      private final Collection<Path> touched;
68  
69      /**
70       * Configured test source roots.
71       */
72      private final Path[] roots;
73  
74      /**
75       * Ctor.
76       *
77       * @param dir Generated sources directory
78       * @param roots Configured test source roots
79       */
80      JavaFiles(final Path dir, final Path... roots) {
81          this.generated = dir;
82          this.roots = roots.clone();
83          this.fresh = new ConcurrentLinkedQueue<>();
84          this.touched = new ConcurrentLinkedQueue<>();
85      }
86  
87      /**
88       * Generate java files and count them.
89       *
90       * @param rewrite Rewrite .java files even if they exist
91       * @param target Full target path to XMIR after transpilation optimizations
92       * @param hsh Tojo hash
93       * @param tests Whether to generate test sources for this tojo
94       * @param cache The cache of this XMIR, keyed by the objects it holds
95       * @return Amount of generated .java files
96       * @throws IOException If fails to save files
97       * @checkstyle ParameterNumberCheck (6 lines)
98       */
99      int total(
100         final boolean rewrite,
101         final Path target,
102         final String hsh,
103         final boolean tests,
104         final GlobalCache cache
105     ) throws IOException {
106         final AtomicInteger saved = new AtomicInteger(0);
107         if (Files.exists(target)) {
108             final Xnav object = new Xnav(target).element("object");
109             final Collection<Xnav> classes = object.elements(Filter.withName("class"))
110                 .collect(Collectors.toList());
111             for (final Xnav clazz : classes) {
112                 final String jname = clazz.attribute("java-name").text().get();
113                 final Path tgt = new Place(jname).make(this.generated, JavaFiles.JAVA);
114                 this.touched.add(tgt);
115                 if (clazz.element("java").text().isPresent()) {
116                     this.fresh.add(tgt);
117                 }
118                 final Footprint java = new FpJavaGenerated(
119                     clazz, new FileGenerationReport(saved, tgt, target)
120                 );
121                 new JavaPlaced(
122                     new FpIfReleased(
123                         hsh,
124                         cache.kept(
125                             this.generated.relativize(tgt),
126                             () -> hsh,
127                             new RewritePolicy(rewrite, tgt),
128                             java
129                         ),
130                         java
131                     ),
132                     tgt,
133                     this.generated, this.roots
134                 ).exec(clazz, tests);
135             }
136             Logger.debug(
137                 this,
138                 "Generated %d Java files from %[file]s",
139                 saved.get(), target
140             );
141         }
142         return saved.get();
143     }
144 
145     /**
146      * Delete generated Java files absent from the current XMIR collection.
147      *
148      * <p>Only considers files inside a directory this run's own {@link
149      * #fresh} output actually landed in, never the whole {@link #generated}
150      * tree: {@code generated-sources} is the standard Maven convention
151      * directory, and other generators (antlr4, protobuf, jaxb2, the
152      * annotation-processor output of maven-compiler-plugin) can write their
153      * own {@code .java} files there too, outside any directory eo itself
154      * touched this run.</p>
155      *
156      * @throws IOException If fails to inspect or remove a generated file
157      */
158     void removeStale() throws IOException {
159         if (Files.exists(this.generated)) {
160             final Set<Path> expected = new HashSet<>(this.fresh);
161             final Collection<Path> dirs = this.directories();
162             try (Stream<Path> walk = Files.walk(this.generated)) {
163                 for (final Path file : walk.filter(Files::isRegularFile)
164                     .filter(path -> path.toString().endsWith(JavaFiles.JAVA)).collect(
165                         Collectors.toList()
166                     )) {
167                     if (dirs.contains(file.getParent())
168                         && !expected.contains(file)
169                         && !"package-info.java".equals(file.getFileName().toString())) {
170                         Files.delete(file);
171                         Logger.debug(this, "Deleted stale generated Java file %[file]s", file);
172                     }
173                 }
174             }
175         }
176     }
177 
178     /**
179      * The directories this run's own output landed in, every parent up
180      * to and including the generated sources root.
181      *
182      * <p>A class this run decided to skip names a directory all the
183      * same, so the set covers what a transpile owns, and nothing another
184      * generator wrote into the same convention directory.</p>
185      *
186      * @return The directories
187      */
188     Collection<Path> directories() {
189         final Set<Path> dirs = new HashSet<>();
190         for (final Path file : this.touched) {
191             for (Path dir = file.getParent(); dir != null && dir.startsWith(this.generated);
192                 dir = dir.getParent()) {
193                 dirs.add(dir);
194             }
195         }
196         return dirs;
197     }
198 }
