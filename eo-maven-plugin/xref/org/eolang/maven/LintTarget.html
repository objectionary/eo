5   package org.eolang.maven;
6   
7   import com.github.lombrozo.xnav.Xnav;
8   import com.jcabi.xml.XML;
9   import java.nio.file.Path;
10  import org.eolang.parser.OnDefault;
11  import org.eolang.parser.OnDetailed;
12  
13  /**
14   * The path a per-file lint result belongs at, derived from the XMIR
15   * document's own object name.
16   *
17   * <p>A source that declares no object gives its XMIR no name at all, and
18   * that very absence is the defect the linting is about to report, so the
19   * name the catalog gives the file stands in for it instead of aborting the
20   * goal (#9325).</p>
21   *
22   * @since 0.64
23   */
24  final class LintTarget {
25  
26      /**
27       * The XMIR document being linted.
28       */
29      private final XML xmir;
30  
31      /**
32       * The document's own source path (for a detailed error location if the
33       * object name can't be read).
34       */
35      private final Path source;
36  
37      /**
38       * The name the catalog gives the file, used when the document names
39       * no object of its own.
40       */
41      private final String catalogued;
42  
43      /**
44       * Ctor.
45       *
46       * @param doc The XMIR document being linted
47       * @param src The document's own source path
48       * @param name The name the catalog gives the file
49       */
50      LintTarget(final XML doc, final Path src, final String name) {
51          this.xmir = doc;
52          this.source = src;
53          this.catalogued = name;
54      }
55  
56      /**
57       * The lint result's path, under the given base directory.
58       *
59       * @param base Base directory
60       * @return Full path
61       */
62      Path under(final Path base) {
63          return new Place(this.name()).make(base, MjAssemble.XMIR);
64      }
65  
66      private String name() {
67          final Xnav nav = new Xnav(this.xmir.inner());
68          final String name;
69          if (nav.path("/object/o[1]/@name").findAny().isPresent()
70              || nav.path("/object/class/@name").findAny().isPresent()) {
71              name = new OnDetailed(
72                  new OnDefault(new Xnav(this.xmir.inner())), this.source
73              ).get();
74          } else {
75              name = this.catalogued;
76          }
77          return name;
78      }
79  }
