5   package org.eolang.maven;
6   
7   import java.nio.file.FileSystem;
8   
9   /**
10   * The step that proves the file system can name a file after an EO object.
11   *
12   * @since 0.74
13   */
14  final class Naming implements Step {
15  
16      /**
17       * The file system to name files in.
18       */
19      private final FileSystem system;
20  
21      /**
22       * The name of an EO object to try.
23       */
24      private final String name;
25  
26      /**
27       * Ctor.
28       *
29       * @param fsys The file system to name files in
30       * @param obj The name of an EO object to try
31       */
32      Naming(final FileSystem fsys, final String obj) {
33          this.system = fsys;
34          this.name = obj;
35      }
36  
37      @Override
38      public void exec() {
39          try {
40              this.system.getPath(this.name);
41          } catch (final IllegalArgumentException ex) {
42              throw new IllegalStateException(
43                  String.format(
44                      "The file system cannot name a file \"%s\" after an EO object, run the build with a UTF-8 locale such as LANG=C.UTF-8",
45                      this.name
46                  ),
47                  ex
48              );
49          }
50      }
51  }
