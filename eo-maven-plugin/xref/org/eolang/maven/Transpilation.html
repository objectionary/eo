31  /**
32   * The XSL train that turns an XMIR into Java, and the cache key of its output.
33   *
34   * <p>The train is the same for every XMIR of a build, so it is compiled once
35   * per thread and reused. The key tells a cached result of an earlier build
36   * apart from the one this train would produce now.</p>
37   *
38   * @since 0.62.0
39   */
40  final class Transpilation {
41  
42      /**
43       * The XSL steps of the transpile train, in order, ending with
44       * {@code purify.xsl} and {@code to-java.xsl}, the two that take
45       * parameters. Kept as a single list so both the train in
46       * {@link #compiled(boolean, boolean, String, long, String, Path)} and the cache-key
47       * fingerprint in {@link #version()} are derived from the same source.
48       */
49      static final String[] XSLS = {
50          "/org/eolang/maven/transpile/recursion-to-cps.xsl",
51          "/org/eolang/parser/parse/set-locators.xsl",
52          "/org/eolang/maven/transpile/set-original-names.xsl",
53          "/org/eolang/maven/transpile/classes.xsl",
54          "/org/eolang/maven/transpile/tests.xsl",
55          "/org/eolang/maven/transpile/anonymous-to-nested.xsl",
56          "/org/eolang/maven/transpile/package.xsl",
57          "/org/eolang/maven/transpile/attrs.xsl",
58          "/org/eolang/maven/transpile/data.xsl",
59          "/org/eolang/maven/transpile/recursion-to-loop.xsl",
60          "/org/eolang/maven/transpile/purify.xsl",
61          "/org/eolang/maven/transpile/to-java.xsl",
62      };
63  
64      /**
65       * Classpath resources {@code xsl:import}-ed by one or more of
66       * {@link #XSLS}, so their content must also be folded into the
67       * cache-key fingerprint in {@link #version()} — editing one of these
68       * shared libraries changes the actual transpile output just as much
69       * as editing a top-level stylesheet does, but leaves {@link #XSLS}
70       * itself unchanged (see #6032). Not part of {@link #XSLS} itself
71       * because that array is also used verbatim to build the actual XSL
72       * train in {@link #compiled(boolean, boolean, String, long, String, Path)}, where its
73       * last two elements are special-cased as {@code purify.xsl} and
74       * {@code to-java.xsl}.
75       */
76      static final String[] IMPORTS = {
77          "/org/eolang/parser/_funcs.xsl",
78          "/org/eolang/parser/_specials.xsl",
79          "/org/eolang/maven/transpile/_recursion.xsl",
80          "/org/eolang/lowering/_java-names.xsl",
81      };
82  
83      /**
84       * Parsing trains with XSLs, one per thread, keyed by whether source
85       * locations are tracked.
86       *
87       * <p>A single shared instance is deliberately avoided: {@link TrClasspath}
88       * and {@link StClasspath} compile their XSL stylesheets lazily on
89       * first use, and that lazy compilation is not thread-safe. Sharing one
90       * instance across the {@link Threaded} worker pool races on the same
91       * underlying compilation cache and can produce truncated or garbled
92       * Java output. Keeping one instance per thread still lets each worker
93       * thread reuse its own compiled stylesheets across the sources it
94       * processes.</p>
95       */
96      private static final ThreadLocal<Map<String, Train<Shift>>> TRAINS =
97          ThreadLocal.withInitial(HashMap::new);

143     /**

150     /**
151      * Ctor.
152      *
153      * @param diagnostics Which diagnostic artifacts to emit while transpiling
154      * @param cvrg Whether located objects are wrapped into {@code PhCoverage}
155      * @param base The class that a generated class extends instead of {@code PhDefault}
156      * @param seconds How many seconds a single generated test may run
157      * @param size How much memory a single generated test may allocate
158      * @param measures Path to the file where XSL measurements are stored
159      * @param dir The target directory of the build
160      * @param tables The directory with the tables of {@link MjInfer}
161      */
162     Transpilation(
163         final Tracking diagnostics,
164         final boolean cvrg,
165         final String base,
166         final long seconds,
167         final String size,
168         final Path measures,
169         final Path dir,
170         final Path tables
171     ) {
172         this.tracking = diagnostics;
173         this.coverage = cvrg;
174         this.superclass = base;
175         this.deadline = seconds;
176         this.memory = size;
177         this.measures = measures;
178         this.target = dir;
179         this.inference = tables;
180         this.rows = new Rows(tables);

183     /**
184      * Cache-key version segment: a fingerprint of the bundled transpile
185      * XSLs and the libraries they {@code xsl:import}, plus the {@code trackLocations}/
186      * {@code tracking}/{@code coverageTracking} flags. The plugin version
187      * is not part of it: {@link Caching} already folds that into the key of
188      * every cache it makes. Folding the XSL content in means
189      * that a change in the transformation logic invalidates the global
190      * transpile cache even when the plugin version is unchanged (a
191      * constant {@code -SNAPSHOT} during development), see #5578; folding
192      * the imported libraries in too closes the gap where editing one of
193      * them changed the actual output without changing anything in
194      * {@link #XSLS} itself, see #6032. Folding the three flags, the name
195      * of the base class, and the limits of a test in means changing any of
196      * them also invalidates the cache, since all of them change what a build
197      * of the same source produces: the first two, the base class, and the
198      * limits change what {@code to-java.xsl} emits (see #6031, #5955, and
199      * #9074), and
200      * {@code tracking} decides whether the XMIRs of the train are written
201      * at all, which a cache hit would otherwise skip (see #7628).
202      * The tables belong to {@link #version(Collection)} instead.
203      *
204      * @return The version segment shared by every source
205      */
206     String version() {
207         return String.format(
208             "%s-%b-%b-%b-%s-%d-%s",
209             new Fingerprint(
210                 Stream.concat(
211                     Arrays.stream(Transpilation.XSLS), Arrays.stream(Transpilation.IMPORTS)
212                 ).toArray(String[]::new)
213             ).get(),
214             this.tracking.locations(), this.tracking.steps(), this.coverage, this.superclass,
215             this.deadline, this.memory
216         );
217     }
218 
219     /**
220      * Cache-key version segment for a file holding these objects.
221      *
222      * <p>{@code purify.xsl} reads the tables and stamps {@code @pure}, which
223      * {@code to-java.xsl} turns into {@code new PhSticky(...)}, so a source
224      * with different rows is different Java (#7627, #7945). The Java files
225      * of that source are keyed by the same segment: {@code Transpiling}
226      * derives the cache of one tojo from this and hands it to
227      * {@code JavaFiles}, so a class never comes back from a slot the rows
228      * of another build filled (#8001).</p>
229      *
230      * @param locators The locators of the objects the file holds
231      * @return The version segment for {@link CachePath}
232      */
233     String version(final Collection<String> locators) {
234         return String.format("%s-%s", this.version(), this.rows.digest(locators));
235     }
236 
237     /**
238      * Whether the XMIR of every step is written down.
239      *
240      * <p>A build that asks for them wants to read them, and a cached
241      * transpilation writes none: the dumps are made by the train and the
242      * cache hands back its stored answer without running it (#7724).</p>
243      *
244      * @return True if the steps are tracked
245      */
246     boolean steps() {
247         return this.tracking.steps();
248     }
249 
250     /**
251      * Build XSL transformation function for a source file.
252      * If transformation steps are tracked - creates a new {@link Xsline}
253      * for every XMIR in purpose of thread safety.
254      *
255      * @param name Name of the object the source XMIR holds
256      * @return XSL transformation function
257      * @throws IOException If the directory for the steps cannot be numbered
258      */
259     Function<XML, XML> forSource(final String name) throws IOException {
260         final Train<Shift> measured = this.measured(this.train());
261         final Function<XML, XML> func;
262         if (this.tracking.steps()) {
263             final Path dir = new Place(name).make(
264                 new Seqdir(this.target, 2).dirs().once("pre-transpile"), ""
265             );
266             func = xml -> new Xsline(new TrSpy(measured, dir)).pass(xml);
267         } else {
268             func = new Xsline(measured)::pass;
269         }
270         return func;
271     }

273     private Train<Shift> measured(final Train<Shift> base) {
274         final Path parent = this.measures.toAbsolutePath().getParent();
275         if (parent.toFile().mkdirs()) {
276             Logger.debug(this, "Directory created for %[file]s", this.measures);
277         }
278         if (!Files.exists(parent)) {
279             throw new IllegalArgumentException(
280                 String.format(
281                     "For some reason, the directory %s is absent, can't write measures to %s",
282                     parent,
283                     this.measures
284                 )
285             );
286         }
287         if (Files.isDirectory(this.measures)) {
288             throw new IllegalArgumentException(
289                 String.format(
290                     "This is not a file but a directory, can't write to it: %s",
291                     this.measures
292                 )
293             );
294         }
295         return new TrLambda(
296             base,
297             shift -> new StMeasured(shift, this.measures)
298         );
299     }
300 
301     private Train<Shift> train() {
302         final boolean track = this.tracking.locations();
303         final boolean instrument = this.coverage;
304         final String base = this.superclass;
305         final long seconds = this.deadline;
306         final String size = this.memory;
307         final Path tables = this.inference;
308         return Transpilation.TRAINS.get().computeIfAbsent(
309             String.format("%b|%b|%s|%d|%s|%s", track, instrument, base, seconds, size, tables),
310             ignored -> Transpilation.compiled(track, instrument, base, seconds, size, tables)
311         );
312     }
313 
314     private static Train<Shift> compiled(
315         final boolean track, final boolean instrument, final String base,
316         final long seconds, final String size, final Path tables
317     ) {
318         final int last = Transpilation.XSLS.length - 1;
319         return new TrFull(
320             new TrJoined<>(
321                 new TrClasspath<>(
322                     Arrays.copyOf(Transpilation.XSLS, last - 1)
323                 ).back(),
324                 new TrDefault<>(
325                     new StPure(Transpilation.XSLS[last - 1], tables),
326                     new StClasspath(
327                         Transpilation.XSLS[last],
328                         String.format("disclaimer %s", new Disclaimer()),
329                         String.format("trackLocations %b", track),
330                         String.format("coverage %b", instrument),
331                         String.format("phiDefaultClass %s", base),
332                         String.format("deadline %d", seconds),
333                         String.format("maxmem %s", size)
334                     )
335                 )
336             )
337         );
338     }
