32  /**
33   * The step of the transpile train that runs {@code purify.xsl}, which marks
34   * the formations that are safe to cache by reading the tables
35   * {@link MjInfer} leaves on disk.
36   *
37   * <p>It is not a {@link com.yegor256.xsline.StClasspath} only because of
38   * where those tables are. A stylesheet reads another document through the
39   * resolver of its own transformation, and the one that
40   * {@link com.yegor256.xsline.StClasspath} installs looks for every document
41   * on the classpath, where the tables of a build under way are not. So the
42   * resolver here answers a {@code file:} URI from the disk and leaves
43   * everything else — the libraries a stylesheet {@code xsl:import}-s — to the
44   * classpath, as before. A table that is not there stays not there: the
45   * resolver says so, {@code doc-available()} in the stylesheet hears it, and a
46   * build that skips {@code eo:infer} marks nothing.</p>
47   *
48   * <p>A table read once is kept, because the stylesheet asks for it again for
49   * every XMIR it stamps and the tables of {@code eo-runtime} are tens of
50   * megabytes — a second per file when every ask parses them afresh. One copy
51   * is kept instead, read by every thread of {@link Threaded} and written by
52   * none, and parsed with node expansion turned off so that no reader builds a
53   * part of it while another one walks past. It is held by a soft reference,
54   * since those megabytes are hundreds once parsed and reading a table again is
55   * better than running a build out of memory.</p>
56   *
57   * <p>What is kept is one state of the file, not its name: the time it was
58   * last written and its length go into the key. A build that runs
59   * {@code eo:infer} and then {@code eo:transpile} twice in one JVM
60   * rewrites the tables in between, and the second stamping must read what
61   * inference has just written rather than what the first one parsed
62   * (see #8225). A table that is not there has no state to tell, and an
63   * empty one stands for it: nothing is ever kept under such a key, since
64   * the reading throws before anything is put into the map.</p>
65   *
66   * @since 0.75.0
67   */
68  final class StPure extends StEnvelope {
69  
70      /**
71       * The tables read so far, by their URIs and the state of the files
72       * they were read from.
73       */
74      private static final Map<String, SoftReference<Node>> TABLES = new HashMap<>(0);
75  
76      /**
77       * The lock on {@link #TABLES}, held while a table is being read, so that
78       * the threads that want the same one wait for it instead of each reading
79       * a copy of its own.
80       */
81      private static final Lock LOCK = new ReentrantLock();
82  
83      /**
84       * Ctor.
85       *
86       * @param sheet The classpath path of the stylesheet to run
87       * @param tables The directory with the tables of {@link MjInfer},
88       *  which does not have to exist
89       */
90      StPure(final String sheet, final Path tables) {
91          super(new StXSL(StPure.compiled(sheet, tables)));
92      }
93  
94      private static XSL compiled(final String sheet, final Path tables) {
95          try {
96              return new XSLDocument(StPure.class.getResource(sheet), sheet)
97                  .with(StPure.sources())
98                  .with("inference", tables.toUri().toString());
99          } catch (final IOException ex) {
100             throw new IllegalStateException(
101                 String.format("Failed to read '%s' from classpath", sheet), ex
102             );
103         }
104     }
105 
106     private static Sources sources() {
107         final Sources classpath = new ClasspathSources();
108         return (href, base) -> {
109             final Source found;
110             if (href.startsWith("file:")) {
111                 found = new DOMSource(StPure.table(href), href);
112             } else {
113                 found = classpath.resolve(href, base);
114             }
115             return found;
116         };
117     }
118 
119     private static Node table(final String href) throws TransformerException {
120         final String key = String.format("%s#%s", href, StPure.stamp(href));
121         StPure.LOCK.lock();
122         try {
123             final SoftReference<Node> kept = StPure.TABLES.get(key);
124             Node found = null;
125             if (kept != null) {
126                 found = kept.get();
127             }
128             if (found == null) {
129                 found = StPure.parsed(href);
130                 StPure.TABLES.put(key, new SoftReference<>(found));
131             }
132             return found;
133         } finally {
134             StPure.LOCK.unlock();
135         }
136     }
137 
138     private static String stamp(final String href) {
139         String state;
140         try {
141             final BasicFileAttributes attrs = Files.readAttributes(
142                 Paths.get(URI.create(href)), BasicFileAttributes.class
143             );
144             state = String.format(
145                 "%d:%d", attrs.lastModifiedTime().toMillis(), attrs.size()
146             );
147         } catch (final IOException ex) {
148             state = "";
149         }
150         return state;
151     }
152 
153     private static Node parsed(final String href) throws TransformerException {
154         try {
155             final DocumentBuilderFactory factory = DocumentBuilderFactory.newInstance();
156             factory.setNamespaceAware(true);
157             factory.setFeature("http://apache.org/xml/features/dom/defer-node-expansion", false);
158             return factory.newDocumentBuilder().parse(href);
159         } catch (final ParserConfigurationException | SAXException | IOException ex) {
160             throw new TransformerException(
161                 String.format("Failed to read the table of types from '%s'", href), ex
162             );
163         }
164     }
165 }
